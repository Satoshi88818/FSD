FSD — Full Self-Driving Dual-Stream Model

Complete Codebase (Hybrid Edition)

This is the fully integrated codebase combining the original FSD geometric pipeline with the Tesla v14-inspired spatio-temporal transformer improvements. All three model variants are available: FSD (original), PureE2EFSD (pure end-to-end), and HybridFSD (best of both worlds).

""" def __init__(self, out_channels: int = 256, config: ModelConfig = CONFIG): super().__init__() self.config = config convnext = convnext_base(weights="DEFAULT") self.net = nn.Sequential( convnext.features, nn.Conv2d(1024, out_channels, 1), ) self.out_channels = out_channels def forward(self, rgb: torch.Tensor) -> torch.Tensor: """ Args: rgb: (B, N, 3, H, W) Returns: feats: (B, N, out_channels, H//32, W//32) """ B, N, C, H, W = rgb.shape out = self.net(rgb.flatten(0, 1)) return out.view(B, N, self.out_channels, H // 32, W // 32) class PhotonBackbone(nn.Module): """ Lightweight photon (single-channel) backbone. #!/usr/bin/env python3 """lss.py — Optimized Lift-Splat-Shoot with Gaussian depth and calibration-aware backprojection helpers. No duplicated ray code.""" from __future__ import annotations import json from typing import Optional, Tuple import torch import torch.nn as nn import torch.nn.functional as F from config import ModelConfig, CONFIG _DEFAULT_FX_FY: float = 910.0 _DEFAULT_HEIGHT_OFFSET: float = 1.65 # metres above ground def get_camera_intrinsics( calibration_file: Optional[str] = None, config: ModelConfig = CONFIG, ) -> Tuple[torch.Tensor, torch.Tensor]: """ Returns (K_rgb, K_photon), each shape (num_cameras, 3, 3). Falls back to reasonable default values when no calibration file is given. """ device = config.device if calibration_file: with open(calibration_file, "r") as f: cal = json.load(f) K_rgb = torch.tensor(cal["intrinsics_rgb"], dtype=torch.float32, device=device) K_photon = torch.tensor(cal["intrinsics_photon"], dtype=torch.float32, device=device) return K_rgb, K_photon fx = fy = _DEFAULT_FX_FY cx_rgb, cy_rgb = config.img_size_rgb[1] / 2.0, config.img_size_rgb[0] / 2.0 cx_ph, cy_ph = config.img_size_photon[1] / 2.0, config.img_size_photon[0] / 2.0 fx_ph = fx * (config.img_size_photon[1] / config.img_size_rgb[1]) fy_ph = fy * (config.img_size_photon[0] / config.img_size_rgb[0]) def _make_K(fx_, fy_, cx_, cy_): K = torch.eye(3, device=device).unsqueeze(0).repeat(config.num_cameras, 1, 1) K[:, 0, 0] = fx_; K[:, 1, 1] = fy_ K[:, 0, 2] = cx_; K[:, 1, 2] = cy_ return K return _make_K(fx, fy, cx_rgb, cy_rgb), _make_K(fx_ph, fy_ph, cx_ph, cy_ph) def get_camera_extrinsics( calibration_file: Optional[str] = None, config: ModelConfig = CONFIG, ) -> torch.Tensor: """ Returns extrinsic matrices (num_cameras, 4, 4) mapping camera→ego. Falls back to evenly-spaced yaw angles at roof height. """ device = config.device if calibration_file: with open(calibration_file, "r") as f: cal = json.load(f) return torch.tensor(cal["extrinsics"], dtype=torch.float32, device=device) angles_deg = torch.tensor( [0, 45, -45, 180, 135, -135, 90, -90], device=device, dtype=torch.float32 ) rad = torch.deg2rad(angles_deg) cos, sin = rad.cos(), rad.sin() R = torch.zeros(config.num_cameras, 3, 3, device=device) R[:, 0, 0] = cos; R[:, 0, 2] = sin R[:, 1, 1] = 1.0 R[:, 2, 0] = -sin; R[:, 2, 2] = cos t = torch.zeros(config.num_cameras, 3, device=device) t[:, 2] = _DEFAULT_HEIGHT_OFFSET Rt = torch.eye(4, device=device).unsqueeze(0).repeat(config.num_cameras, 1, 1) Rt[:, :3, :3] = R Rt[:, :3, 3] = t return Rt def create_pixel_grid( h: int, w: int, orig_h: int, orig_w: int, device: torch.device ) -> torch.Tensor: """Returns (h, w, 3) homogeneous pixel coords in original image space.""" scale_h, scale_w = orig_h / h, orig_w / w xs = torch.arange(w, device=device, dtype=torch.float32) * scale_w + scale_w / 2 ys = torch.arange(h, device=device, dtype=torch.float32) * scale_h + scale_h / 2 yy, xx = torch.meshgrid(ys, xs, indexing="ij") return torch.stack([xx, yy, torch.ones_like(xx)], dim=-1) def backproject_to_3d( pixels: torch.Tensor, depth: torch.Tensor, K: torch.Tensor, ) -> torch.Tensor: """ Lift 2-D pixels to 3-D camera-frame points. Args: pixels: (B, N, H*W, 3) homogeneous pixel coords depth: (B, N, H*W, 1) K: (N, 3, 3) Returns: cam_points: (B, N, H*W, 3) """ fx = K[:, 0, 0].view(1, -1, 1, 1) fy = K[:, 1, 1].view(1, -1, 1, 1) cx = K[:, 0, 2].view(1, -1, 1, 1) cy = K[:, 1, 2].view(1, -1, 1, 1) rays = torch.cat( [(pixels[..., 0:1] - cx) / fx, (pixels[..., 1:2] - cy) / fy, pixels[..., 2:3]], dim=-1, ) return rays * depth class OptimizedLSS(nn.Module): """ Lift-Splat-Shoot with: - Gaussian depth prediction (mu, sigma) - Uncertainty-weighted scatter-add into 3-D BEV voxels - Reuses create_pixel_grid / backproject_to_3d helpers (no duplicated ray code) """ def __init__( self, in_channels: int, img_h: int, img_w: int, K_base: torch.Tensor, calibration_file: Optional[str] = None, config: ModelConfig = CONFIG, ): super().__init__() self.config = config self.img_h = img_h self.img_w = img_w self.depth_net = nn.Sequential( nn.Conv2d(in_channels, in_channels, 3, padding=1), nn.GroupNorm(32, in_channels), nn.ReLU(inplace=True), nn.Conv2d(in_channels, 2, 1), # [mu_logit, log_sigma] ) self.K = nn.Parameter(K_base.float(), requires_grad=False) self.cam2ego = nn.Parameter( get_camera_extrinsics(calibration_file, config).float(), requires_grad=False, ) def forward(self, feats: torch.Tensor) -> torch.Tensor: """ Args: feats: (B, N, C, Hf, Wf) Returns: bev: (B, C, Z, H_bev, W_bev) """ B, N, C, Hf, Wf = feats.shape device = feats.device cfg = self.config # Depth prediction d = self.depth_net(feats.flatten(0, 1)).view(B, N, 2, Hf, Wf) mu = F.softplus(d[:, :, 0]) + cfg.near sigma = F.softplus(d[:, :, 1]) + 1e-6 depth = torch.clamp(mu, cfg.near, cfg.far) weight = 1.0 / sigma # 3-D points via helpers pixels = create_pixel_grid(Hf, Wf, self.img_h, self.img_w, device) pixels = pixels.view(1, 1, Hf * Wf, 3).expand(B, N, -1, -1) cam_pts = backproject_to_3d(pixels, depth.view(B, N, Hf * Wf, 1), self.K) # Camera → ego R = self.cam2ego[None, :N, :3, :3] t = self.cam2ego[None, :N, :3, 3:4].transpose(-1, -2) ego_pts = cam_pts @ R.transpose(-1, -2) + t x_idx = ((ego_pts[..., 0] - cfg.x_bound[0]) / cfg.x_bound[2]).long() y_idx = ((ego_pts[..., 1] - cfg.y_bound[0]) / cfg.y_bound[2]).long() z_idx = ((ego_pts[..., 2] - cfg.z_bound[0]) / cfg.z_bound[2]).long() mask = ( (x_idx >= 0) & (x_idx < cfg.bev_w) & (y_idx >= 0) & (y_idx < cfg.bev_h) & (z_idx >= 0) & (z_idx < cfg.bev_size_z) ) feat_flat = feats.permute(0, 1, 3, 4, 2).reshape(B, N, Hf * Wf, C) wfeat = feat_flat * weight.view(B, N, Hf * Wf, 1) valid = mask.reshape(-1).nonzero(as_tuple=False).squeeze(-1) valid_b = valid // (N * Hf * Wf) valid_x = x_idx.reshape(-1)[valid].clamp(0, cfg.bev_w - 1) valid_y = y_idx.reshape(-1)[valid].clamp(0, cfg.bev_h - 1) valid_z = z_idx.reshape(-1)[valid].clamp(0, cfg.bev_size_z - 1) valid_f = wfeat.reshape(-1, C)[valid] V = cfg.bev_size_z * cfg.bev_h * cfg.bev_w vox = valid_z * (cfg.bev_h * cfg.bev_w) + valid_y * cfg.bev_w + valid_x gidx = valid_b * V + vox bev = torch.zeros(B * V, C, device=device, dtype=feats.dtype) cnt = torch.zeros(B * V, 1, device=device, dtype=feats.dtype) bev.scatter_add_(0, gidx.unsqueeze(1).expand(-1, C), valid_f) cnt.scatter_add_(0, gidx.unsqueeze(1), torch.ones_like(valid_f[:, :1])) bev = bev.view(B, V, C).permute(0, 2, 1).view(B, C, cfg.bev_size_z, cfg.bev_h, cfg.bev_w) cnt = cnt.view(B, 1, cfg.bev_size_z, cfg.bev_h, cfg.bev_w) return bev / (cnt + 1e-6)

#!/usr/bin/env python3 """attention.py — Swin-style window attention and bidirectional cross-attention. Bug fix applied: window_reverse is rewritten with correct reshape/permute order that passes round-trip tests (window_reverse(window_partition(x)) == x). """ from __future__ import annotations from typing import Optional, Tuple import torch import torch.nn as nn import torch.nn.functional as F from config import ModelConfig, CONFIG # ── Window helpers ────────────────────────────────────────────────────────────── def window_partition(x: torch.Tensor, window_size: int) -> torch.Tensor: """ Partition a (B, C, H, W) feature map into non-overlapping windows. Returns: (num_windows * B, window_size * window_size, C) """ B, C, H, W = x.shape pad_h = (window_size - H % window_size) % window_size pad_w = (window_size - W % window_size) % window_size if pad_h > 0 or pad_w > 0: x = F.pad(x, (0, pad_w, 0, pad_h)) Hp, Wp = H + pad_h, W + pad_w x = x.view(B, C, Hp // window_size, window_size, Wp // window_size, window_size) windows = x.permute(0, 2, 4, 3, 5, 1).contiguous() # B, nH, nW, ws, ws, C return windows.view(-1, window_size * window_size, C) def window_reverse( windows: torch.Tensor, window_size: int, H: int, W: int ) -> torch.Tensor: """ Reconstruct (B, C, H, W) from windowed tokens. Corrected implementation — round-trip tested. Args: windows: (num_windows * B, ws*ws, C) window_size: window side length H, W: original spatial dims (before padding) Returns: (B, C, H, W) """ pad_h = (window_size - H % window_size) % window_size pad_w = (window_size - W % window_size) % window_size Hp = H + pad_h Wp = W + pad_w num_h = Hp // window_size num_w = Wp // window_size B = windows.shape[0] // (num_h * num_w) C = windows.shape[-1] x = windows.view(B, num_h, num_w, window_size, window_size, C) x = x.permute(0, 5, 1, 3, 2, 4).contiguous() # B, C, num_h, ws, num_w, ws x = x.view(B, C, Hp, Wp) return x[:, :, :H, :W] def compute_mask( H: int, W: int, window_size: int, shift_size: int, device: torch.device ) -> torch.Tensor: """Compute shifted-window attention mask.""" pad_h = (window_size - H % window_size) % window_size pad_w = (window_size - W % window_size) % window_size Hp, Wp = H + pad_h, W + pad_w img_mask = torch.zeros(1, 1, Hp, Wp, device=device) cnt = 0 slices = ( slice(-window_size, None), slice(-window_size, -shift_size), slice(-shift_size, None), ) for h_s in slices: for w_s in slices: img_mask[:, :, h_s, w_s] = cnt cnt += 1 mask_windows = window_partition(img_mask, window_size).squeeze(-1) attn_mask = mask_windows.unsqueeze(1) - mask_windows.unsqueeze(2) attn_mask = attn_mask.masked_fill(attn_mask != 0, -100.0).masked_fill(attn_mask == 0, 0.0) return attn_mask # ── WindowAttention ───────────────────────────────────────────────────────────── class WindowAttention(nn.Module): """ Swin-style windowed self-/cross-attention with relative position bias. Supports cross-attention: q_input and kv_input may differ. """ def __init__(self, dim: int, win_size: int, num_heads: int): super().__init__() self.num_heads = num_heads head_dim = dim // num_heads self.scale = head_dim ** -0.5 self.q = nn.Linear(dim, dim) self.kv = nn.Linear(dim, dim * 2) self.proj = nn.Linear(dim, dim) self.relative_position_bias_table = nn.Parameter( torch.zeros((2 * win_size - 1) ** 2, num_heads) ) coords = torch.stack(torch.meshgrid( torch.arange(win_size), torch.arange(win_size), indexing="ij" )) coords_flat = coords.reshape(2, -1) rel = coords_flat[:, :, None] - coords_flat[:, None, :] rel = rel.permute(1, 2, 0).contiguous() rel[:, :, 0] += win_size - 1 rel[:, :, 1] += win_size - 1 rel[:, :, 0] *= 2 * win_size - 1 self.register_buffer("relative_position_index", rel.sum(-1)) def forward( self, q_input: torch.Tensor, kv_input: torch.Tensor, mask: Optional[torch.Tensor] = None, ) -> torch.Tensor: B_, N, C = q_input.shape H, D = self.num_heads, C // self.num_heads q = self.q(q_input).reshape(B_, N, H, D).permute(0, 2, 1, 3) kv = self.kv(kv_input).reshape(B_, N, 2, H, D).permute(2, 0, 3, 1, 4) k, v = kv.unbind(0) attn = (q @ k.transpose(-2, -1)) * self.scale rpb = ( self.relative_position_bias_table[self.relative_position_index.reshape(-1)] .reshape(N, N, -1).permute(2, 0, 1).contiguous().unsqueeze(0) ) attn = attn + rpb if mask is not None: nW = mask.shape[0] attn = attn.view(B_ // nW, nW, H, N, N) + mask.unsqueeze(1).unsqueeze(0) attn = attn.view(-1, H, N, N) attn = F.softmax(attn, dim=-1) x = (attn @ v).transpose(1, 2).reshape(B_, N, C) return self.proj(x) # ── BidirectionalWindowCrossAttention ────────────────────────────────────────── class BidirectionalWindowCrossAttention(nn.Module): """ Bidirectional windowed cross-attention between two BEV slices (visual ↔ photon). Pre-LN added; regular + shifted pass consolidated into a single forward. """ def __init__(self, dim: int, win_size: int, num_heads: int): super().__init__() self.win_size = win_size self.shift_size = win_size // 2 self.attn = WindowAttention(dim, win_size, num_heads) self.norm_v = nn.LayerNorm(dim) self.norm_p = nn.LayerNorm(dim) def _apply( self, v: torch.Tensor, p: torch.Tensor, shift: bool, ) -> Tuple[torch.Tensor, torch.Tensor]: H, W = v.shape[2], v.shape[3] ss = self.shift_size if shift else 0 mask = compute_mask(H, W, self.win_size, ss, v.device) if shift else None if shift: v = torch.roll(v, shifts=(-ss, -ss), dims=(2, 3)) p = torch.roll(p, shifts=(-ss, -ss), dims=(2, 3)) vw = window_partition(v, self.win_size) pw = window_partition(p, self.win_size) ve = self.attn(self.norm_v(vw), self.norm_p(pw), mask) + vw pe = self.attn(self.norm_p(pw), self.norm_v(vw), mask) + pw v_out = window_reverse(ve, self.win_size, H, W) p_out = window_reverse(pe, self.win_size, H, W) if shift: v_out = torch.roll(v_out, shifts=(ss, ss), dims=(2, 3)) p_out = torch.roll(p_out, shifts=(ss, ss), dims=(2, 3)) return v_out, p_out def forward( self, v: torch.Tensor, p: torch.Tensor ) -> Tuple[torch.Tensor, torch.Tensor]: """Run one regular pass followed by one shifted-window pass.""" v, p = self._apply(v, p, shift=False) v, p = self._apply(v, p, shift=True) return v, p

Order: ZYX.""" cy, sy = yaw.cos(), yaw.sin() cp, sp = pitch.cos(), pitch.sin() cr, sr = roll.cos(), roll.sin() R = torch.stack([ torch.stack([ cy*cp, cy*sp*sr - sy*cr, cy*sp*cr + sy*sr], dim=-1), torch.stack([ sy*cp, sy*sp*sr + cy*cr, sy*sp*cr - cy*sr], dim=-1), torch.stack([ -sp, cp*sr, cp*cr ], dim=-1), ], dim=-2) return R def _motion_to_SE3(motion: torch.Tensor) -> torch.Tensor: """Convert 6-DoF vector (B, 6) → (B, 4, 4) SE(3) matrix.""" B = motion.shape[0] dx, dy, dz = motion[:, 0], motion[:, 1], motion[:, 2] roll, pitch, yaw = motion[:, 3], motion[:, 4], motion[:, 5] R = _euler_to_rot(roll, pitch, yaw) t = torch.stack([dx, dy, dz], dim=-1) T = torch.eye(4, device=motion.device, dtype=motion.dtype).unsqueeze(0).expand(B, -1, -1).clone() T[:, :3, :3] = R T[:, :3, 3] = t return T def compose_SE3(T1: torch.Tensor, T2: torch.Tensor) -> torch.Tensor: """(B, 4, 4) @ (B, 4, 4) → (B, 4, 4) — proper SE(3) composition.""" return torch.bmm(T1, T2) def SE3_to_motion(T: torch.Tensor) -> torch.Tensor: """Approximate inverse of _motion_to_SE3 — for logging only.""" dx = T[:, 0, 3]; dy = T[:, 1, 3]; dz = T[:, 2, 3] pitch = -torch.asin(T[:, 2, 0].clamp(-1, 1)) roll = torch.atan2(T[:, 2, 1], T[:, 2, 2]) yaw = torch.atan2(T[:, 1, 0], T[:, 0, 0]) return torch.stack([dx, dy, dz, roll, pitch, yaw], dim=-1) # ── BEV volume warp ───────────────────────────────────────────────────────────── def warp_bev_volume( prev_bev: torch.Tensor, motion: torch.Tensor, config: ModelConfig = CONFIG, ) -> torch.Tensor: """ Warp a BEV volume by an SE(3) ego-motion. import torch.nn.functional as F def compute_total_loss(outputs, targets): # --- Existing losses --- occ_loss = F.cross_entropy(outputs["occupancy"], targets["occ_labels"]) # Negative log-likelihood under Gaussian with predicted log-variance vel_var = torch.exp(outputs["velocity_logvar"]).clamp(1e-6, 1e3) vel_loss = (0.5 * (outputs["velocity"] - targets["velocity"]) ** 2 / vel_var + 0.5 * outputs["velocity_logvar"]).mean() det_loss = ( F.cross_entropy(outputs["detection_cls"].flatten(0, 1), targets["det_cls"].flatten(0, 1)) + F.l1_loss(outputs["detection_boxes"], targets["det_boxes"]) + F.binary_cross_entropy(outputs["detection_iou"].squeeze(-1), targets["det_iou"]) ) ctrl_var = outputs["control_var"].clamp(1e-6, 1e3) ctrl_loss = (0.5 * (outputs["control"] - targets["control"]) ** 2 / ctrl_var + 0.5 * ctrl_var.log()).mean() # --- NEW: Occupancy flow loss (HybridFSD only) --- flow_loss = torch.tensor(0.0, device=occ_loss.device) if "occupancy_flow" in outputs and "gt_flow" in targets: flow_loss = F.l1_loss(outputs["occupancy_flow"], targets["gt_flow"]) # Optional: add endpoint error term # flow_loss += F.mse_loss(outputs["occupancy_flow"], targets["gt_flow"]).sqrt().mean() total = occ_loss + vel_loss + det_loss + ctrl_loss + 0.5 * flow_loss return total, { "occ": occ_loss.item(), "vel": vel_loss.item(), "det": det_loss.item(), "ctrl": ctrl_loss.item(), "flow": flow_loss.item(), }
