""" proposal_generator.py --------------------- v11 Changes: 1. ConsistencyDistillationGenerator: single-step inference wrapper. DynamicBicycleModel gains an EKFTireObserver that continuously self-calibrates mu from real IMU feedback, replacing the fixed VehicleParameterHead mu estimate for safety-critical scenarios. """ import torch import torch.nn as nn import torch.nn.functional as F from torch import Tensor from typing import Optional, Tuple, Dict, Any from config import ModelConfig # ───────────────────────────────────────────────────────────────────────────── # EKF Tire Observer (Improvement #8) # ───────────────────────────────────────────────────────────────────────────── class EKFTireObserver(nn.Module): """ Extended Kalman Filter that self-calibrates the road friction coefficient (mu) by comparing predicted vs. actual IMU accelerations. State vector: [mu] (scalar friction coefficient) Observation: [a_x_meas, a_y_meas] (IMU accelerations, last 500ms) At each inference step we run one EKF predict-update cycle. The corrected mu_hat is then fed forward into the DynamicBicycleModel, replacing the static VehicleParameterHead estimate for the friction term. This is the difference between a car that slides on black ice versus one that catches itself immediately. """ def __init__(self, config: ModelConfig): super().__init__() self.q = config.ekf_process_noise_q # Process noise self.r = config.ekf_obs_noise_r # Observation noise # Learned residual corrector: maps (mu_prior, a_pred_err) -> delta_mu # Small 2-layer net; runs at every timestep so must be lightweight. self.corrector = nn.Sequential( nn.Linear(3, 32), nn.SiLU(), nn.Linear(32, 1), nn.Tanh() ) # Running EKF state (mu_hat, P) — treated as buffers, not parameters self.register_buffer('mu_hat', torch.tensor(0.85)) self.register_buffer('P', torch.tensor(0.1)) def _predict(self): """EKF predict step: mu evolves slowly, so F=1, add process noise.""" # mu_{k+1|k} = mu_{k|k} (constant model) P_pred = self.P + self.q return self.mu_hat.clone(), P_pred def _update(self, mu_pred: Tensor, P_pred: Tensor, a_pred: Tensor, a_meas: Tensor) -> Tuple[Tensor, Tensor]: """ EKF update step. H = d(a_pred)/d(mu) approximated via the neural corrector residual. Innovation y = a_meas - a_pred. """ # Innovation innov = a_meas - a_pred # (B, 2) innov_norm = innov.norm(dim=-1, keepdim=True) # (B, 1) # Neural residual correction: learn the nonlinear H mapping inp = torch.cat([mu_pred.expand(innov.shape[0], 1), innov_norm, (a_pred.norm(dim=-1, keepdim=True))], dim=-1) delta_mu = self.corrector(inp).squeeze(-1) # (B,) # Scalar Kalman gain: K = P * H^T / (H * P * H^T + R) H_approx = 1.0 # linearisation of the mu->a mapping S = H_approx * P_pred * H_approx + self.r K = P_pred * H_approx / S # Updated state mu_upd = mu_pred + K * delta_mu.mean() mu_upd = mu_upd.clamp(0.1, 1.1) # physical bounds [ice, dry+rubber] P_upd = (1 - K * H_approx) * P_pred return mu_upd, P_upd def forward(self, a_pred: Tensor, a_meas: Tensor) -> Tensor: """ Args: a_pred: (B, 2) predicted [ax, ay] from DBM at last timestep a_meas: (B, 2) measured [ax, ay] from IMU (last 500ms window) Returns: mu_hat: (B,) updated friction estimate """ mu_pred, P_pred = self._predict() mu_upd, P_upd = self._update(mu_pred, P_pred, a_pred, a_meas) # Persist state (single-device; stateful across inference calls) self.mu_hat = mu_upd.detach().mean() self.P = P_upd.detach() return mu_upd.expand(a_pred.shape[0]) # ───────────────────────────────────────────────────────────────────────────── # Dynamic Bicycle Model (v10 base + EKF mu override) # ───────────────────────────────────────────────────────────────────────────── class DynamicBicycleModel(nn.Module): """ v11 DBM: Pacejka tire physics (unchanged from v10) + optional EKF mu. When use_ekf_tire_observer=True, the mu value from dynamic_params is replaced by the EKF's online calibrated estimate before each rollout. """ def __init__(self, config: ModelConfig): super().__init__() self.cfg = config self.dt = config.dbm_dt if config.use_ekf_tire_observer: self.ekf = EKFTireObserver(config) def compute_derivatives(self, state: Tensor, controls: Tensor, dynamic_params: Optional[Dict[str, Tensor]] = None) -> Tensor: a = controls[..., 0].clamp(self.cfg.dbm_max_decel, self.cfg.dbm_max_accel) delta = controls[..., 1].clamp(-self.cfg.dbm_max_steer, self.cfg.dbm_max_steer) x, y, psi, v, beta = (state[..., i] for i in range(5)) v = v.clamp(min=1.0) m = dynamic_params['mass'] if dynamic_params else torch.full_like(v, self.cfg.dbm_mass_base) Iz = dynamic_params['Iz'] if dynamic_params else torch.full_like(v, self.cfg.dbm_Iz_base) Caf = dynamic_params['Caf'] if dynamic_params else torch.full_like(v, self.cfg.dbm_Caf_base) Car = dynamic_params['Car'] if dynamic_params else torch.full_like(v, self.cfg.dbm_Car_base) # v11: mu may be overridden by EKF observer (set in forward) mu = dynamic_params.get('mu_ekf', dynamic_params['mu']) \ if dynamic_params else torch.full_like(v, self.cfg.dbm_mu_base) lf, lr = self.cfg.dbm_lf, self.cfg.dbm_lr psi_dot = v * torch.tan(delta) / (lf + lr) alpha_f = delta - (psi_dot * lf + v * beta) / v alpha_r = -(psi_dot * lr - v * beta) / v if self.cfg.use_pacejka_tire_model: B, C, E = self.cfg.pacejka_B, self.cfg.pacejka_C, self.cfg.pacejka_E F_z_f = m * 9.81 * (lr / (lf + lr)) F_z_r = m * 9.81 * (lf / (lf + lr)) D_f, D_r = mu * F_z_f, mu * F_z_r F_yf = D_f * torch.sin(C * torch.atan(B * alpha_f - E * (B * alpha_f - torch.atan(B * alpha_f)))) F_yr = D_r * torch.sin(C * torch.atan(B * alpha_r - E * (B * alpha_r - torch.atan(B * alpha_r)))) else: cap = mu * m * 9.81 / 2 F_yf = (Caf * alpha_f).clamp(-cap, cap) F_yr = (Car * alpha_r).clamp(-cap, cap) beta_dot = (F_yf + F_yr) / (m * v) - psi_dot psi_ddot = (lf * F_yf - lr * F_yr) / Iz dx = v * torch.cos(psi + beta) dy = v * torch.sin(psi + beta) return torch.stack([dx, dy, psi_dot + psi_ddot * self.dt * 0.5, a, beta_dot], dim=-1) def forward(self, controls: Tensor, ego_state: Tensor, dynamic_params: Optional[Dict[str, Tensor]] = None, imu_accel: Optional[Tensor] = None) -> Tensor: """ Args: controls: (B, T, 2) [accel, steer] ego_state: (B, 5) [x, y, psi, v, beta] dynamic_params: dict of per-scene vehicle params imu_accel: (B, 2) optional real IMU [ax, ay] for EKF update """ # v11: update mu via EKF if IMU data is available if self.cfg.use_ekf_tire_observer and imu_accel is not None and dynamic_params is not None: # Predict accel from current state for innovation with torch.no_grad(): dz_pred = self.compute_derivatives(ego_state, controls[:, 0], dynamic_params) a_pred = dz_pred[..., 3:5] # (ax_pred, beta_dot) as proxy a_pred_2d = torch.stack([dz_pred[..., 3], ego_state[:, 3] * dz_pred[..., 2]], dim=-1) # [ax, v*psi_dot] mu_ekf = self.ekf(a_pred_2d, imu_accel) dynamic_params = dict(dynamic_params) dynamic_params['mu_ekf'] = mu_ekf B, T, _ = controls.shape states, curr = [], ego_state.clone() for t in range(T): dz = self.compute_derivatives(curr, controls[:, t], dynamic_params) curr = curr + dz * self.dt states.append(curr) return torch.stack(states, dim=1) # ───────────────────────────────────────────────────────────────────────────── # Lane Prior Utility # ───────────────────────────────────────────────────────────────────────────── def extract_lane_prior(ego_state: Tensor, lane_centrelines: Tensor) -> Tensor: B, L, P, _ = lane_centrelines.shape ego_xy = ego_state[:, :2].unsqueeze(1).unsqueeze(1) nearest_idx = (lane_centrelines - ego_xy).norm(dim=-1).min(dim=-1).values.argmin(dim=-1) idx = nearest_idx[:, None, None, None].expand(-1, 1, P, 2) return lane_centrelines.gather(1, idx).squeeze(1) # ───────────────────────────────────────────────────────────────────────────── # Manifold Flow Matching Generator (v10 teacher — used at training time) # ───────────────────────────────────────────────────────────────────────────── class ManifoldFlowMatchingGenerator(nn.Module): """ v10 Full Riemannian Flow Matching on DBM Manifold (unchanged). """ def __init__(self, config: ModelConfig): super().__init__() self.config = config self.T = config.trajectory_horizon hd = config.flow_matching_hidden self.encoder = nn.Sequential( nn.AdaptiveAvgPool2d((8, 8)), nn.Flatten(), nn.Linear(config.bev_channels * 64, hd), nn.SiLU(), nn.Linear(hd, hd), ) self.lane_prior_enc = nn.Sequential( nn.Linear(self.T * 2, hd), nn.SiLU(), nn.Linear(hd, hd) ) self.velocity_net = nn.Sequential( nn.Linear(5 + 1 + hd + hd, hd * 2), nn.SiLU(), nn.Linear(hd * 2, hd * 2), nn.SiLU(), nn.Linear(hd * 2, 2) ) self.dbm = DynamicBicycleModel(config) self.n = config.num_proposals self.steps = config.guided_flow_steps def _build_lane_prior(self, lane_centrelines, ego_state, B, dev): nearest = extract_lane_prior(ego_state, lane_centrelines) P = nearest.shape[1] indices = torch.linspace(0, P - """ heads.py -------- v11 Changes: 1. MambaSSM block replaces GRU in TemporalTrackMemory (Improvement #2). 2. FoveatedPerceptionHead uses epistemic uncertainty to crop and re-process high-uncertainty BEV patches at 4× resolution (Improvement #5). 3. """ safety.py --------- v11 Changes: 1. Neural Control Barrier Function (NCBF) replaces the soft QP projection (Improvement #3). CBF provides a mathematical guarantee that if the system starts in the safe set h(x) >= 0, it NEVER leaves it given the control constraints. This is a hard safety guarantee, not a soft penalty. 2. STL Differentiable Rule Layer for training (Improvement #7). 3. Legacy DifferentiableQPSafetyFilter retained as fallback. The CBF h(x) is a neural network mapping state → scalar. Safety set: S = {x | h(x) >= 0}. Condition for forward invariance (Nagumo's theorem): dh/dt >= -gamma * h(x) i.e., if h(x) > 0 the barrier can decrease, but only at rate gamma*h(x), so it cannot cross zero. v11 implementation: 1. h_net: neural network that scores how "safe" a trajectory point is. Trained so h(x) > 0 inside free space, h(x) < 0 inside obstacles. 2. For each proposed trajectory, we evaluate h(x_t) and dh/dt along it. 3. If the CBF constraint is violated (dh/dt < -gamma*h), we apply the minimum-norm correction to the control input to restore safety. 4. This is a HARD filter — it physically prevents the control from exiting the safe invariant set, even if flow matching proposes something reckless. """ def __init__(self, config: ModelConfig): super().__init__() self.cfg = config self.gamma = config.ncbf_gamma self.margin = config.ncbf_safety_margin hd = config.ncbf_hidden_dim # h(x): state [x, y, psi, v, beta] + context -> safety score self.h_net = nn.Sequential( nn.Linear(5 + 2, hd), nn.Tanh(), # +2 for nearest obstacle dist nn.Linear(hd, hd), nn.Tanh(), nn.Linear(hd, 1) ) # Correction network: maps CBF violation -> minimum-norm control correction self.correction_net = nn.Sequential( nn.Linear(5 + 2, hd), nn.SiLU(), nn.Linear(hd, 2) # delta accel, delta steer ) def _get_nearest_obstacle_context(self, state: Tensor, occupancy: Tensor, epistemic_unc: Tensor) -> Tensor: """ Compute distance-to-nearest-obstacle and local uncertainty for each trajectory waypoint.
