#!/usr/bin/env python3 """ backbones.py — Visual (ConvNeXt) and photon backbone networks. """ from __future__ import annotations from typing import Optional, Tuple import torch import torch.nn as nn from torchvision.models import convnext_base from config import ModelConfig, CONFIG class VisualBackbone(nn.Module): """ ConvNeXt-Base backbone with 1×1 projection to 256 channels. Processes all cameras in a single batched forward pass. """ def __init__(self, out_channels: int = 256, config: ModelConfig = CONFIG): super().__init__() self.config = config convnext = convnext_base(weights="DEFAULT") self.net = nn.Sequential( convnext.features, nn.Conv2d(1024, out_channels, 1), ) self.out_channels = out_channels def forward(self, rgb: torch.Tensor) -> torch.Tensor: """ Args: rgb: (B, N, 3, H, W) Returns: feats: (B, N, out_channels, H//32, W//32) """ B, N, C, H, W = rgb.shape out = self.net(rgb.flatten(0, 1)) return out.view(B, N, self.out_channels, H // 32, W // 32) class PhotonBackbone(nn.Module): """ Lightweight photon (single-channel) backbone. Three stride-2 conv stages → stride-8 feature map. """ def __init__(self, out_channels: int = 128, config: ModelConfig = CONFIG): super().__init__() self.config = config self.net = nn.Sequential( nn.Conv2d(1, 64, 7, stride=2, padding=3), nn.BatchNorm2d(64), nn.SiLU(inplace=True), nn.Conv2d(64, 96, 5, stride=2, padding=2), nn.BatchNorm2d(96), nn.SiLU(inplace=True), nn.Conv2d(96, out_channels, 3, stride=2, padding=1), nn.BatchNorm2d(out_channels), nn.SiLU(inplace=True), ) self.out_channels = out_channels def forward(self, x: torch.Tensor) -> torch.Tensor: """ Args: x: (B, N, 1, H, W) Returns: feats: (B, N, out_channels, H//8, W//8) """ B, N, C, H, W = x.shape out = self.net(x.flatten(0, 1)) return out.view(B, N, self.out_channels, H // 8, W // 8)

#!/usr/bin/env python3 """ attention.py — Swin-style window attention and bidirectional cross-attention. Bug fix applied: window_reverse is rewritten with correct reshape/permute order that passes round-trip tests (window_reverse(window_partition(x)) == x). """ from __future__ import annotations from typing import Optional, Tuple import torch import torch.nn as nn import torch.nn.functional as F from config import ModelConfig, CONFIG # ── Window helpers ──────────────────────────────────────────────────────────── def window_partition(x: torch.Tensor, window_size: int) -> torch.Tensor: """ Partition a (B, C, H, W) feature map into non-overlapping windows. Returns: (num_windows * B, window_size * window_size, C) """ B, C, H, W = x.shape pad_h = (window_size - H % window_size) % window_size pad_w = (window_size - W % window_size) % window_size if pad_h > 0 or pad_w > 0: x = F.pad(x, (0, pad_w, 0, pad_h)) Hp, Wp = H + pad_h, W + pad_w x = x.view(B, C, Hp // window_size, window_size, Wp // window_size, window_size) windows = x.permute(0, 2, 4, 3, 5, 1).contiguous() # B, nH, nW, ws, ws, C windows = windows.view(-1, window_size * window_size, C) return windows def window_reverse( windows: torch.Tensor, window_size: int, H: int, W: int ) -> torch.Tensor: """ Reconstruct (B, C, H, W) from windowed tokens. Corrected implementation — round-trip tested: window_reverse(window_partition(x, ws), ws, H, W) == x Args: windows: (num_windows * B, ws*ws, C) window_size: window side length H, W: original spatial dimensions (before any padding) Returns: (B, C, H, W) """ pad_h = (window_size - H % window_size) % window_size pad_w = (window_size - W % window_size) % window_size Hp = H + pad_h Wp = W + pad_w num_h = Hp // window_size num_w = Wp // window_size # Derive exact batch size from the window count B = windows.shape[0] // (num_h * num_w) C = windows.shape[-1] # windows: (B*num_h*num_w, ws*ws, C) x = windows.view(B, num_h, num_w, window_size, window_size, C) # → (B, C, num_h, ws, num_w, ws) x = x.permute(0, 5, 1, 3, 2, 4).contiguous() x = x.view(B, C, Hp, Wp) return x[:, :, :H, :W] def compute_mask( H: int, W: int, window_size: int, shift_size: int, device: torch.device ) -> torch.Tensor: """Compute shifted-window attention mask (same Hp/Wp logic as window_reverse).""" pad_h = (window_size - H % window_size) % window_size pad_w = (window_size - W % window_size) % window_size Hp = H + pad_h Wp = W + pad_w img_mask = torch.zeros(1, 1, Hp, Wp, device=device) cnt = 0 slices = ( slice(-window_size, None), slice(-window_size, -shift_size), slice(-shift_size, None), ) for h_s in slices: for w_s in slices: img_mask[:, :, h_s, w_s] = cnt cnt += 1 mask_windows = window_partition(img_mask, window_size).squeeze(-1) # nW, N attn_mask = mask_windows.unsqueeze(1) - mask_windows.unsqueeze(2) # nW, N, N attn_mask = attn_mask.masked_fill(attn_mask != 0, -100.0).masked_fill(attn_mask == 0, 0.0) return attn_mask # ── WindowAttention ─────────────────────────────────────────────────────────── class WindowAttention(nn.Module): """ Swin-style windowed self-/cross-attention with relative position bias. Supports cross-attention: q_input and kv_input may differ. """ def __init__(self, dim: int, win_size: int, num_heads: int): super().__init__() self.num_heads = num_heads head_dim = dim // num_heads self.scale = head_dim ** -0.5 self.q = nn.Linear(dim, dim) self.kv = nn.Linear(dim, dim * 2) self.proj = nn.Linear(dim, dim) # Learnable relative position bias table self.relative_position_bias_table = nn.Parameter( torch.zeros((2 * win_size - 1) ** 2, num_heads) ) coords = torch.stack(torch.meshgrid( torch.arange(win_size), torch.arange(win_size), indexing="ij" )) # 2, ws, ws coords_flat = coords.reshape(2, -1) rel = coords_flat[:, :, None] - coords_flat[:, None, :] # 2, N, N rel = rel.permute(1, 2, 0).contiguous() rel[:, :, 0] += win_size - 1 rel[:, :, 1] += win_size - 1 rel[:, :, 0] *= 2 * win_size - 1 self.register_buffer("relative_position_index", rel.sum(-1)) # N, N def forward( self, q_input: torch.Tensor, kv_input: torch.Tensor, mask: Optional[torch.Tensor] = None, ) -> torch.Tensor: B_, N, C = q_input.shape H = self.num_heads D = C // H q = self.q(q_input).reshape(B_, N, H, D).permute(0, 2, 1, 3) kv = self.kv(kv_input).reshape(B_, N, 2, H, D).permute(2, 0, 3, 1, 4) k, v = kv.unbind(0) attn = (q @ k.transpose(-2, -1)) * self.scale rpb = self.relative_position_bias_table[ self.relative_position_index.reshape(-1) ].reshape(N, N, -1).permute(2, 0, 1).contiguous().unsqueeze(0) attn = attn + rpb if mask is not None: nW = mask.shape[0] attn = attn.view(B_ // nW, nW, H, N, N) + mask.unsqueeze(1).unsqueeze(0) attn = attn.view(-1, H, N, N) attn = F.softmax(attn, dim=-1) x = (attn @ v).transpose(1, 2).reshape(B_, N, C) return self.proj(x) # ── BidirectionalWindowCrossAttention ──────────────────────────────────────── class BidirectionalWindowCrossAttention(nn.Module): """ Bidirectional windowed cross-attention between two BEV slices (visual ↔ photon). Supports optional shifted-window pass. """ def __init__(self, dim: int, win_size: int, num_heads: int): super().__init__() self.win_size = win_size self.shift_size = win_size // 2 self.attn = WindowAttention(dim, win_size, num_heads) self.norm_v = nn.LayerNorm(dim) self.norm_p = nn.LayerNorm(dim) def _apply( self, v: torch.Tensor, p: torch.Tensor, shift: bool, ) -> Tuple[torch.Tensor, torch.Tensor]: H, W = v.shape[2], v.shape[3] device = v.device ss = self.shift_size if shift else 0 mask = compute_mask(H, W, self.win_size, ss, device) if shift else None if shift: v = torch.roll(v, shifts=(-ss, -ss), dims=(2, 3)) p = torch.roll(p, shifts=(-ss, -ss), dims=(2, 3)) vw = window_partition(v, self.win_size) # (B*nW, N, C) pw = window_partition(p, self.win_size) # Layer-norm before attention B_C = vw.shape[0] ve = self.attn(self.norm_v(vw), self.norm_p(pw), mask) + vw pe = self.attn(self.norm_p(pw), self.norm_v(vw), mask) + pw v_out = window_reverse(ve, self.win_size, H, W) p_out = window_reverse(pe, self.win_size, H, W) if shift: v_out = torch.roll(v_out, shifts=(ss, ss), dims=(2, 3)) p_out = torch.roll(p_out, shifts=(ss, ss), dims=(2, 3)) return v_out, p_out def forward( self, v: torch.Tensor, p: torch.Tensor ) -> Tuple[torch.Tensor, torch.Tensor]: """Run one regular pass followed by one shifted-window pass.""" v, p = self._apply(v, p, shift=False) v, p = self._apply(v, p, shift=True) return v, p

#!/usr/bin/env python3 """ heads.py — Detection, occupancy, velocity, and dual control heads. """ def __init__(self, in_channels: int = 128, num_outputs: int = 5): super().__init__() def _make_head(): return nn.Sequential( nn.Conv2d(in_channels, 64, 3, padding=1), nn.ReLU(inplace=True), nn.AdaptiveAvgPool2d(1), nn.Flatten(), nn.Linear(64, num_outputs * 2), # mean + log-var ) self.head_a = _make_head() self.head_b = _make_head() self.num_outputs = num_outputs def forward(self, x: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]: """ Returns: control: (B, num_outputs) uncertainty-weighted ensemble mean total_var: (B, num_outputs) total variance (epistemic + aleatoric) """ out_a = self.head_a(x) out_b = self.head_b(x) mu_a, lv_a = out_a.chunk(2, dim=-1) mu_b, lv_b = out_b.chunk(2, dim=-1) var_a = torch.exp(lv_a).clamp(1e-6, 1e3) var_b = torch.exp(lv_b).clamp(1e-6, 1e3) # Precision-weighted combination prec_a = 1.0 / var_a prec_b = 1.0 / var_b total_prec = prec_a + prec_b control = (prec_a * mu_a + prec_b * mu_b) / total_prec total_var = 1.0 / total_prec + ((mu_a - mu_b) ** 2) / 4.0 return torch.tanh(control).clamp(-1, 1), total_var

#!/usr/bin/env python3 """ temporal.py — Temporal state containers, ConvGRU, DeformAlign, and BEV warping with proper SE(3) motion composition. """ from __future__ import annotations from typing import NamedTuple, Optional, Tuple import torch import torch.nn as nn import torch.nn.functional as F from config import ModelConfig, CONFIG # ── Frame history container ─────────────────────────────────────────────────── class FrameHistory(NamedTuple): visual_bev: torch.Tensor # (T-1, B, C_vis, Z, H, W) photon_bev: torch.Tensor # (T-1, B, C_ph, Z, H, W) motions: torch.Tensor # (T-1, B, 6) [dx,dy,dz,roll,pitch,yaw] gru_state: torch.Tensor # (B, C_fused, Z, H, W) @classmethod def empty( cls, bsz: int, c_vis: int = 256, c_ph: int = 128, c_gru: int = 512, config: ModelConfig = CONFIG, ) -> "FrameHistory": Z = config.bev_size_z H, W = config.bev_h, config.bev_w T = config.temporal_frames d, dt = config.device, config.dtype return cls( visual_bev=torch.zeros(T - 1, bsz, c_vis, Z, H, W, device=d, dtype=dt), photon_bev=torch.zeros(T - 1, bsz, c_ph, Z, H, W, device=d, dtype=dt), motions=torch.zeros(T - 1, bsz, 6, device=d, dtype=dt), gru_state=torch.zeros(bsz, c_gru, Z, H, W, device=d, dtype=dt), ) # ── SE(3) motion composition ────────────────────────────────────────────────── def _euler_to_rot(roll: torch.Tensor, pitch: torch.Tensor, yaw: torch.Tensor) -> torch.Tensor: """ Build (B, 3, 3) rotation matrices from per-batch Euler angles (radians). Rotation order: yaw → pitch → roll (extrinsic ZYX). """ cy, sy = yaw.cos(), yaw.sin() cp, sp = pitch.cos(), pitch.sin() cr, sr = roll.cos(), roll.sin() # Row-major: R = Rz(yaw) @ Ry(pitch) @ Rx(roll) R = torch.stack([ torch.stack([ cy*cp, cy*sp*sr - sy*cr, cy*sp*cr + sy*sr], dim=-1), torch.stack([ sy*cp, sy*sp*sr + cy*cr, sy*sp*cr - cy*sr], dim=-1), torch.stack([-sp, cp*sr, cp*cr ], dim=-1), ], dim=-2) # (B, 3, 3) return R def _motion_to_SE3(motion: torch.Tensor) -> torch.Tensor: """ Convert 6-DoF motion vector (B, 6) → (B, 4, 4) SE(3) matrix. motion[:, :3] = (dx, dy, dz), motion[:, 3:] = (roll, pitch, yaw) """ B = motion.shape[0] dx, dy, dz = motion[:, 0], motion[:, 1], motion[:, 2] roll, pitch, yaw = motion[:, 3], motion[:, 4], motion[:, 5] R = _euler_to_rot(roll, pitch, yaw) # (B, 3, 3) t = torch.stack([dx, dy, dz], dim=-1) # (B, 3) T = torch.eye(4, device=motion.device, dtype=motion.dtype).unsqueeze(0).expand(B, -1, -1).clone() T[:, :3, :3] = R T[:, :3, 3] = t return T def compose_SE3(T1: torch.Tensor, T2: torch.Tensor) -> torch.Tensor: """(B, 4, 4) @ (B, 4, 4) → (B, 4, 4) — proper SE(3) composition.""" return torch.bmm(T1, T2) def SE3_to_motion(T: torch.Tensor) -> torch.Tensor: """ Approximate inverse of _motion_to_SE3: extract (dx,dy,dz,roll,pitch,yaw). Only used for inspection / logging — warping uses the matrix directly. """ dx = T[:, 0, 3]; dy = T[:, 1, 3]; dz = T[:, 2, 3] pitch = -torch.asin(T[:, 2, 0].clamp(-1, 1)) roll = torch.atan2(T[:, 2, 1], T[:, 2, 2]) yaw = torch.atan2(T[:, 1, 0], T[:, 0, 0]) return torch.stack([dx, dy, dz, roll, pitch, yaw], dim=-1) # ── BEV volume warp ─────────────────────────────────────────────────────────── def warp_bev_volume(prev_bev: torch.Tensor, motion: torch.Tensor, config: ModelConfig = CONFIG) -> torch.Tensor: """ Warp a BEV volume by an SE(3) ego-motion. Args: prev_bev: (B, C, Z, H, W) motion: (B, 6) [dx,dy,dz,roll,pitch,yaw] or (B,4,4) SE3 matrix Returns: warped: (B, C, Z, H, W) """ B, C, Z, H, W = prev_bev.shape device = prev_bev.device dtype_orig = prev_bev.dtype cfg = config # Accept either a 6-DoF vector or a pre-built SE(3) matrix if motion.ndim == 2 and motion.shape[-1] == 6: T = _motion_to_SE3(motion.float()) else: T = motion.float() R = T[:, :3, :3] # (B, 3, 3) t = T[:, :3, 3] # (B, 3) # Metric grid gz = torch.linspace(cfg.z_bound[0] + cfg.z_bound[2] / 2, cfg.z_bound[1] - cfg.z_bound[2] / 2, Z, device=device) gy = torch.linspace(cfg.y_bound[0] + cfg.y_bound[2] / 2, cfg.y_bound[1] - cfg.y_bound[2] / 2, H, device=device) gx = torch.linspace(cfg.x_bound[0] + cfg.x_bound[2] / 2, cfg.x_bound[1] - cfg.x_bound[2] / 2, W, device=device) gz, gy, gx = torch.meshgrid(gz, gy, gx, indexing="ij") # Z,H,W each pts = torch.stack([gx, gy, gz], dim=-1).view(1, Z * H * W, 3).float() # (1,N,3) # Apply rotation and translation: p' = R @ p + t pts_t = pts @ R.transpose(-1, -2) + t.unsqueeze(1) # (B, N, 3) # Normalise to [-1, 1] def norm(v, lo, hi): return (v - lo) / (hi - lo) * 2.0 - 1.0 gx_n = norm(pts_t[..., 0], cfg.x_bound[0], cfg.x_bound[1]) gy_n = norm(pts_t[..., 1], cfg.y_bound[0], cfg.y_bound[1]) gz_n = norm(pts_t[..., 2], cfg.z_bound[0], cfg.z_bound[1]) grid = torch.stack([gx_n, gy_n, gz_n], dim=-1).view(B, Z, H, W, 3) warped = F.grid_sample( prev_bev.float(), grid, mode="bilinear", padding_mode="border", align_corners=True, ) return warped.to(dtype_orig) def compose_cumulative_motion(motions: torch.Tensor, idx: int) -> torch.Tensor: """ Properly compose motions[0..idx] using SE(3) multiplication. Args: motions: (T-1, B, 6) idx: index up to which to compose (inclusive) Returns: (B, 4, 4) composed SE(3) transform """ B = motions.shape[1] device = motions.device T_cum = torch.eye(4, device=device, dtype=torch.float32).unsqueeze(0).expand(B, -1, -1).clone() for i in range(idx + 1): T_i = _motion_to_SE3(motions[i].float()) T_cum = compose_SE3(T_cum, T_i) return T_cum # ── ConvGRU ─────────────────────────────────────────────────────────────────── class ConvGRU(nn.Module): """3D Convolutional GRU for long-term BEV memory.""" def __init__(self, channels: int): super().__init__() self.convz = nn.Conv3d(2 * channels, channels, 3, padding=1) self.convr = nn.Conv3d(2 * channels, channels, 3, padding=1) self.convn = nn.Conv3d(2 * channels, channels, 3, padding=1) def forward(self, h: torch.Tensor, x: torch.Tensor) -> torch.Tensor: """ Args: h: (B, C, Z, H, W) — previous hidden state x: (B, C, Z, H, W) — current input Returns: h_new: (B, C, Z, H, W) """ hx = torch.cat([h, x], dim=1) z = torch.sigmoid(self.convz(hx)) r = torch.sigmoid(self.convr(hx)) n = torch.tanh(self.convn(torch.cat([r * h, x], dim=1))) return (1 - z) * h + z * n # ── DeformAlign ─────────────────────────────────────────────────────────────── class DeformAlign(nn.Module): """ Learned deformable alignment of a warped BEV volume to the current frame. (B, N, 256, Hf, Wf) ph_feat = self.photon_backbone(photon) # (B, N, 128, Hf, Wf) aux_out = ( self.aux_head(vis_feat.flatten(0, 1)) if is_training else None ) return vis_feat, ph_feat, aux_out def lift_to_bev( self, vis_feat: torch.Tensor, ph_feat: torch.Tensor, ) -> Tuple[torch.Tensor, torch.Tensor]: return self.visual_lss(vis_feat), self.photon_lss(ph_feat) def warp_and_align_history( self, curr_vis: torch.Tensor, curr_ph: torch.Tensor, history: FrameHistory, ego_motion: torch.Tensor, ) -> Tuple[list, list]: """ Warp historical BEV frames to the current frame using proper SE(3) cumulative composition (rotation-safe, not simple vector addition). """ warped_vis = [curr_vis] warped_photon = [curr_ph] for i in range(self.config.temporal_frames - 1): # Compose all motions from i=0..i with full SE(3) multiplication T_cum = compose_cumulative_motion(history.motions, i) warped_v = warp_bev_volume(history.visual_bev[i], T_cum, self.config) aligned_v = self.deform_align_vis(curr_vis, warped_v) warped_vis.append(aligned_v) warped_p = warp_bev_volume(history.photon_bev[i], T_cum, self.config) aligned_p = self.deform_align_photon(curr_ph, warped_p) warped_photon.append(aligned_p) return warped_vis, warped_photon def fuse_multi_frame( self, warped_vis: list, warped_photon: list, ) -> torch.Tensor: """Per-Z-slice bidirectional cross-attention fusion across all frames.""" fused_list = [] for v, p in zip(warped_vis, warped_photon): v_proj = self.vis_proj(v) # (B, 128, Z, H, W) fused_slices = [] for z in range(self.config.bev_size_z): v_s = v_proj[:, :, z, :, :] # (B, 128, H, W) p_s = p[:, :, z, :, :] v_e, p_e = self.bi_cross_attn(v_s, p_s) fused_slices.append(torch.cat([v_e, p_e], dim=1)) # (B, 256, H, W) fused_list.append(torch.stack(fused_slices, dim=2)) # (B, 256, Z, H, W) stacked = torch.stack(fused_list, dim=0) fused = torch.cat( [fused_list[0], stacked.max(0)[0][:, :128], stacked.mean(0)[:, :128]], dim=1, ) # (B, 512, Z, H, W) return self.temporal_fusion(fused) + fused def apply_long_term_memory( self, fused: torch.Tensor, prev_gru: torch.Tensor, ego_motion: torch.Tensor, ) -> torch.Tensor: warped_gru = warp_bev_volume(prev_gru, ego_motion, self.config) return self.gru(warped_gru, fused) def modulate_with_language_3d( self, bev_volume: torch.Tensor, language_prompt: str, ) -> torch.Tensor: """ True 3D language modulation: cross-attention between all BEV voxels and language token embeddings (not just max-projected 2D slice). """ B, C, Z, H, W = bev_volume.shape device = bev_volume.device tokens = self.tokenizer( language_prompt, return_tensors="pt", truncation=True, max_length=64 ).to(device) with torch.no_grad(): lang_emb = self.lang_encoder(**tokens).last_hidden_state # (1, L, 384) lang_proj = self.lang_proj(lang_emb).expand(B, -1, -1) # (B, L, 128) bev_flat = bev_volume.permute(0, 2, 3, 4, 1).reshape(B, Z * H * W, C) # (B, N, 128) mod, _ = self.lang_attn_3d(query=bev_flat, key=lang_proj, value=lang_proj) mod_bev = (bev_flat + mod).reshape(B, Z, H, W, C).permute(0, 4, 1, 2, 3) return self.lang_proj_out(mod_bev) def run_heads(self, final_bev: torch.Tensor) -> Dict[str, torch.Tensor]: occ, occ_lv = self.occ_head(final_bev) vel, vel_lv = self.velocity_head(final_bev) cls_logits, boxes, iou, conf = self.detection_head(final_bev) return { "occupancy": occ, "occupancy_logvar": occ_lv, "velocity": vel, "velocity_logvar": vel_lv, "detection_cls": cls_logits, "detection_boxes": boxes, "detection_iou": iou, "detection_conf": conf, } # ── Forward ────────────────────────────────────────────────────────────── def forward( self, rgb: torch.Tensor, photon: torch.Tensor, ego_motion: torch.Tensor, is_training: bool = False, history: Optional[FrameHistory] = None, language_prompt: Optional[str] = None, ) -> Dict[str, Any]: cfg = self.config rgb = rgb.to(dtype=cfg.dtype) photon = photon.to(dtype=cfg.dtype) vis_feats, ph_feats, aux_out = self.extract_features(rgb, photon, is_training) curr_vis_bev, curr_ph_bev = self.lift_to_bev(vis_feats, ph_feats) history = history or FrameHistory.empty(rgb.shape[0], config=cfg) warped_vis, warped_ph = self.warp_and_align_history( curr_vis_bev, curr_ph_bev, history, ego_motion ) fused_bev = self.fuse_multi_frame(warped_vis, warped_ph) # Optional 3D language modulation on fused BEV (true 3D conditioning) if language_prompt: # Project fused_bev down to 128-ch for language modulation, then back ph_vol = self.vis_proj(curr_ph_bev) # (B,128,Z,H,W) — photon volume ph_vol_mod = self.modulate_with_language_3d(ph_vol, language_prompt) # Inject back into fused_bev via addition on photon channel slice fused_bev[:, 256:384] = fused_bev[:, 256:384] + ph_vol_mod final_bev = self.apply_long_term_memory(fused_bev, history.gru_state, ego_motion) outputs = self.run_heads(final_bev) outputs["bev_fused"] = final_bev # Control bev_2d = final_bev.max(2)[0][:, :128, :, :] # rough 2D summary for control control, control_var = self.control_head(bev_2d) outputs["control"] = control outputs["control_var"] = control_var if aux_out is not None: outputs["aux_out"] = aux_out # Update history if cfg.temporal_frames > 1: new_hist = FrameHistory( visual_bev=torch.cat([curr_vis_bev.detach().unsqueeze(0), history.visual_bev[:-1]], dim=0), photon_bev=torch.cat([curr_ph_bev.detach().unsqueeze(0), history.photon_bev[:-1]], dim=0), motions=torch.cat([ego_motion.detach().unsqueeze(0), history.motions[:-1]], dim=0), gru_state=final_bev.detach(), ) outputs["new_history"] = new_hist return outputs
