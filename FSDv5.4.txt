#!/usr/bin/env python3 """safety.py — Safety-critical planning stack for FSD v5.3. (B, K) # ── Mask rejected trajectories ───────────────────────────────────── total = total.masked_fill(~valid_mask, float("inf")) best_idx = total.argmin(dim=1) # (B,) best_traj = trajectories[torch.arange(B), best_idx] # (B, T, 4) # ── min_safe_margin guard ────────────────────────────────────────── if w.min_safe_margin_warn > 0.0: for b in range(B): worst_prox = col_margin[b, best_idx[b]].item() threshold = w.min_safe_margin_warn * self.cfg.safety_occ_threshold if worst_prox > threshold: log.warning( "[CostPlannerV2] Batch %d: selected trajectory worst-proximity " "occupancy %.3f exceeds warn threshold %.3f. " "Consider increasing cost_weights.collision_margin.", b, worst_prox, threshold, ) diag_list = [] for b in range(B): idx = best_idx[b].item() diag_list.append({ "selected_idx": idx, "collision_margin_cost": float(col_margin[b, idx].item()), "lane_cost": float(lane[b, idx].item()), "interaction_cost": float(interaction[b, idx].item()), "comfort_cost": float(comfort[b, idx].item()), "smoothness_cost": float(smoothness[b, idx].item()), "progress_cost": float(progress[b, idx].item()), "route_cost": float(route_c[b, idx].item()), "total_cost": float(total[b, idx].item()), }) return best_traj, total, diag_list # ── Cost 1: Collision Margin (soft — NOT binary) ─────────────────────── def _collision_margin_cost( self, trajectories: torch.Tensor, # (B, K, T, 4) occupancy: torch.Tensor, # (B, C, Z, H, W) ) -> torch.Tensor: # (B, K) """Penalise worst proximity to occupied cells across all timesteps. Uses max over time (not mean) so a single close moment is caught. Even valid (non-colliding) trajectories passing near obstacles get a higher cost, encouraging the planner to prefer clearance. """ B, K, T, _ = trajectories.shape cfg = self.cfg occ_bev = occupancy.max(dim=2).values # (B, C, H, W) occ_2d = occ_bev[:, 1:].sum(dim=1, keepdim=True) # (B, 1, H, W) traj_xy = trajectories[..., :2] norm_x = 2.0 * (traj_xy[..., 0] - cfg.x_bound[0]) / (cfg.x_bound[1] - cfg.x_bound[0]) - 1.0 norm_y = 2.0 * (traj_xy[..., 1] - cfg.y_bound[0]) / (cfg.y_bound[1] - cfg.y_bound[0]) - 1.0 grid = torch.stack([norm_x, norm_y], dim=-1).view(B, K * T, 1, 2) sampled = F.grid_sample( occ_2d.float(), grid, mode="bilinear", padding_mode="border", align_corners=False, ).view(B, K, T) return sampled.max(dim=-1).values # (B, K) worst-proximity # ── Cost 2: Lane Alignment ───────────────────────────────────────────── def _lane_cost( self, trajectories: torch.Tensor, # (B, K, T, 4) polylines: Optional[torch.Tensor], # (B, P, pts, 2) ) -> torch.Tensor: # (B, K) """Penalise distance of the final waypoint from the nearest lane centre. Falls back to zero if no polyline data is available (e.g. parking). """ if polylines is None: B, K = trajectories.shape[:2] return torch.zeros(B, K, device=trajectories.device) B, K = trajectories.shape[:2] final_pos = trajectories[:, :, -1, :2] # (B, K, 2) pts_flat = polylines.view(B, -1, 2) # (B, P*pts, 2) dist = torch.cdist(final_pos.float(), pts_flat.float()) # (B, K, P*pts) return dist.min(dim=-1).values # (B, K) # ── Cost 3: Interaction Cost ─────────────────────────────────────────── def _interaction_cost( self, trajectories: torch.Tensor, # (B, K, T, 4) agent_preds: Optional[torch.Tensor], # (B, A, T, 2) ) -> torch.Tensor: # (B, K) """Exponential repulsion from predicted agent positions. Unlike the binary SafetyFilterV2 agent check (reject if < radius), this cost is a smooth function of distance, encouraging the planner to prefer trajectories that are far from all agents at all times. """ if agent_preds is None: B, K = trajectories.shape[:2] return torch.zeros(B, K, device=trajectories.device) ego = trajectories[..., :2].unsqueeze(2) # (B, K, 1, T, 2) agents = agent_preds.unsqueeze(1) # (B, 1, A, T, 2) dist = torch.norm(ego - agents, dim=-1) # (B, K, A, T) cost = torch.exp(-dist).mean(dim=(-1, -2)) # (B, K) return cost # ── Cost 4: Comfort (Improved) ───────────────────────────────────────── def _comfort_cost( self, trajectories: torch.Tensor, # (B, K, T, 4) ) -> torch.Tensor: # (B, K) """Penalise acceleration and jerk. Jerk is weighted 0.5 relative to acceleration — physically, jerk is what passengers feel as a 'lurch'. Both are averaged over the horizon so longer horizons are not penalised unfairly. """ dt = self.cfg.trajectory_dt pos = trajectories[..., :2] vel = (pos[:, :, 1:] - pos[:, :, :-1]) / dt # (B, K, T-1, 2) acc = (vel[:, :, 1:] - vel[:, :, :-1]) / dt # (B, K, T-2, 2) jerk = (acc[:, :, 1:] - acc[:, :, :-1]) / dt # (B, K, T-3, 2) return ( acc.norm(dim=-1).mean(dim=-1) + 0.5 * jerk.norm(dim=-1).mean(dim=-1) ) # (B, K) # ── Cost 5: Temporal Consistency ────────────────────────────────────── def _smoothness_cost( self, trajectories: torch.Tensor, # (B, K, T, 4) prev_traj: Optional[torch.Tensor], # (B, T, 4) ) -> torch.Tensor: # (B, K) """Penalise deviation from the previously selected trajectory. Prevents jitter, oscillation, and indecision across planning frames. Returns zero on the first frame (no previous trajectory). """ if prev_traj is None: B, K = trajectories.shape[:2] return torch.zeros(B, K, device=trajectories.device) diff = trajectories - prev_traj.unsqueeze(1) # (B, K, T, 4) return torch.norm(diff, dim=-1).mean(dim=-1) # (B, K) # ── Cost 6: Progress (Improved) ──────────────────────────────────────── def _progress_cost( self, trajectories: torch.Tensor, # (B, K, T, 4) ) -> torch.Tensor: # (B, K) """Reward forward displacement along the ego x-axis. Negative cost = reward. Uses only the x (forward) component of the final waypoint so lateral movement is not rewarded. """ forward = trajectories[:, :, -1, 0] # (B, K) — x at final timestep return -forward # negative = reward progress # ── Cost 7: Route Following ──────────────────────────────────────────── def _route_cost( self, trajectories: torch.Tensor, # (B, K, T, 4) route: Optional[torch.Tensor], # (B, R, 2) ) -> torch.Tensor: # (B, K) """Penalise distance of the final waypoint from the nearest route point. Falls back to zero if no route is provided. """ if route is None: B, K = trajectories.shape[:2] return torch.zeros(B, K, device=trajectories.device) final_pos = trajectories[:, :, -1, :2] # (B, K, 2) dist = torch.cdist(final_pos.float(), route.float()) # (B, K, R) return dist.min(dim=-1).values # (B, K) # ── EmergencyBrake (unchanged from v5.2) ────────────────────────────────────── class EmergencyBrake(nn.Module): """Last-resort fallback when ALL K proposals are rejected. Generates a straight-line kinematic deceleration trajectory that brings the vehicle to a full stop over trajectory_horizon timesteps. Output shape: (1, T, 4) to match a single selected trajectory. """ def __init__(self, config: ModelConfig = CONFIG): super().__init__() self.cfg = config def forward(self, current_speed: float = 1.0) -> torch.Tensor: T = self.cfg.trajectory_horizon dt = self.cfg.trajectory_dt device = self.cfg.device speeds = torch.linspace(current_speed, 0.0, T, device=device) x_pos = torch.cumsum(speeds * dt, dim=0) y_pos = torch.zeros(T, device=device) heading = torch.zeros(T, device=device) traj = torch.stack([x_pos, y_pos, heading, speeds], dim=-1) # (T, 4) return traj.unsqueeze(0) # (1, T, 4)

#!/usr/bin/env python3 """model.py — FSD v5.3 top-level model. Improvements over original v5.3: - Emergency brake engages with explicit warnings.warn (RuntimeWarning) so calling code can catch or log the event. Previously silent. - CostPlannerV2 receives config.cost_weights (CostWeights instance). - reset_prev_trajectory() called at epoch boundaries in train.py. """ from __future__ import annotations import warnings from typing import Dict, Optional import torch import torch.nn as nn import torch.nn.functional as F from config import ModelConfig, CONFIG from lss import GaussianLSS, BEVCalibHead, OnlineExtrinsicRefinement, CalibrationManager from heads import ( MotionFormer, MapFormer, TrajectoryHeadV2, FlowMatchingTrajectoryHead, UncertaintyFusionV2, ) from safety import SafetyFilterV2, CostPlannerV2, EmergencyBrake, SafetyDiagnostics class FSDv5(nn.Module): """Full Self-Driving model v5.3. Safety outputs in forward(): outputs["best_trajectory"] : (B, T, 4) — final safe trajectory outputs["trajectory_costs"] : (B, K) — cost per proposal (+inf if rejected) outputs["safety_valid_mask"]: (B, K) — bool mask from SafetyFilterV2 outputs["safety_diagnostics"]: list[SafetyDiagnostics] outputs["emergency_used"] : (B,) bool — True if emergency brake engaged outputs["cost_diagnostics"] : list[dict] — 7-term breakdown per batch outputs["agent_predictions"]: (B, A, T, 2) outputs["drivable_map"] : (B, 1, H, W) or None outputs["map_polylines"] : (B, P, pts, 2) """ def __init__( self, config: ModelConfig = CONFIG, calibration_manager: Optional[CalibrationManager] = None, ): super().__init__() self.cfg = config # ── Backbone ────────────────────────────────────────────────────── self.backbone = nn.Sequential( nn.Conv2d(3, 64, 7, padding=3), nn.GroupNorm(32, 64), nn.ReLU(inplace=True), ) # ── BEV lifting ─────────────────────────────────────────────────── K_base = torch.eye(3).unsqueeze(0).repeat(config.num_cameras, 1, 1) self.lss = GaussianLSS( in_channels=64, img_h=config.img_size_rgb[0], img_w=config.img_size_rgb[1], K_base=K_base, calibration_manager=calibration_manager, config=config, ) # ── Calibration heads ───────────────────────────────────────────── self.calib_head = BEVCalibHead(in_channels=64, num_cameras=config.num_cameras) self.extrinsic_refine = OnlineExtrinsicRefinement( in_channels=64 * config.bev_size_z, num_cameras=config.num_cameras, ) # ── Scene understanding ─────────────────────────────────────────── self.motion_former = MotionFormer(config) self.map_former = MapFormer(config) self.occ_head = nn.Sequential( nn.Conv3d(64, 64, 3, padding=1), nn.GroupNorm(8, 64), nn.ReLU(inplace=True), nn.Conv3d(64, config.num_classes + 1, 1), nn.Sigmoid(), ) self.drivable_head = nn.Sequential( nn.Conv2d(64, 32, 3, padding=1), nn.ReLU(inplace=True), nn.Conv2d(32, 1, 1), nn.Sigmoid(), ) # ── Trajectory proposals ────────────────────────────────────────── if config.use_flow_matching: self.trajectory_head: nn.Module = FlowMatchingTrajectoryHead(config) else: self.trajectory_head = TrajectoryHeadV2(config) # ── Safety stack (v5.3) ─────────────────────────────────────────── self.safety_filter = SafetyFilterV2(config) self.cost_planner = CostPlannerV2(config, cost_weights=config.cost_weights) self.emergency_brake = EmergencyBrake(config) # ── Uncertainty ─────────────────────────────────────────────────── self.uncertainty = UncertaintyFusionV2(config) # ── Internal state: previous trajectory for smoothness cost ─────── self._prev_traj: Optional[torch.Tensor] = None def forward( self, rgb: torch.Tensor, # (B, N, 3, H, W) photon: Optional[torch.Tensor] = None, extrinsics: Optional[torch.Tensor] = None, range_prior: Optional[torch.Tensor] = None, route: Optional[torch.Tensor] = None, # (B, R, 2) current_speed: float = 1.0, use_uncertainty: bool = False, ) -> Dict[str, torch.Tensor]: B, N, C, H, W = rgb.shape outputs: Dict = {} # ── Feature extraction ───────────────────────────────────────────── img_flat = rgb.view(B * N, C, H, W) feats_flat = self.backbone(img_flat) # (B*N, 64, H, W) feats = feats_flat.view(B, N, 64, H, W) # ── BEV lifting ──────────────────────────────────────────────────── bev = self.lss(feats, range_prior) # (B, 64, Z, Hb, Wb) # ── Occupancy ────────────────────────────────────────────────────── occupancy = self.occ_head(bev) # (B, C+1, Z, Hb, Wb) outputs["occupancy"] = occupancy # ── Uncertainty-adjusted occupancy (optional) ────────────────────── uncertainty = None if use_uncertainty: bev_mean, bev_var = self.uncertainty(bev) uncertainty = self.occ_head(bev_var) outputs["occupancy_uncertainty"] = uncertainty # ── Drivable area map ─────────────────────────────────────────────── bev_2d = bev.mean(2) # (B, 64, Hb, Wb) drivable_map = self.drivable_head(bev_2d) # (B, 1, Hb, Wb) outputs["drivable_map"] = drivable_map # ── Scene understanding ───────────────────────────────────────────── agent_tokens, agent_trajs, agent_scores = self.motion_former(bev_2d) map_tokens, map_polylines = self.map_former(bev_2d) outputs["agent_trajectories"] = agent_trajs outputs["agent_scores"] = agent_scores outputs["map_polylines"] = map_polylines outputs["agent_predictions"] = agent_trajs # (B, A, T, 2) # ── Safety-integrated trajectory selection ────────────────────────── outputs = self._run_trajectory_head( outputs, agent_tokens=agent_tokens, map_tokens=map_tokens, bev=bev, agent_preds=agent_trajs, drivable_map=drivable_map, uncertainty=uncertainty, map_polylines=map_polylines, route=route, current_speed=current_speed, ) return outputs def _run_trajectory_head( self, outputs: Dict, agent_tokens: torch.Tensor, map_tokens: torch.Tensor, bev: torch.Tensor, agent_preds: Optional[torch.Tensor], drivable_map: Optional[torch.Tensor], uncertainty: Optional[torch.Tensor], map_polylines: Optional[torch.Tensor], route: Optional[torch.Tensor], current_speed: float, ) -> Dict: """v5.3 safety-integrated trajectory selection. Pipeline: 1. NN head generates K proposals 2. SafetyFilterV2 hard-rejects (footprint, agents, drivable, kinematics) 3. CostPlannerV2 selects best via 7-term multi-objective cost 4. EmergencyBrake activates (with warning) if all proposals rejected 5. Store best_traj as prev_traj for next-frame smoothness cost """ B = agent_tokens.shape[0] occupancy = outputs["occupancy"] # ── Step 1: NN proposals ─────────────────────────────────────────── traj_out = self.trajectory_head(agent_tokens, map_tokens, bev) trajectories = traj_out["trajectories"] # (B, K, T, 4) outputs["trajectory_proposals"] = trajectories outputs["trajectory_scores"] = traj_out.get("scores") # ── Step 2: SafetyFilterV2 ───────────────────────────────────────── valid_mask, safety_diag = self.safety_filter( trajectories, occupancy, agent_preds=agent_preds, drivable_map=drivable_map, uncertainty=uncertainty, ) outputs["safety_valid_mask"] = valid_mask outputs["safety_diagnostics"] = safety_diag # ── Step 3: CostPlannerV2 (multi-objective) ──────────────────────── best_traj, costs, cost_diag = self.cost_planner( trajectories, occupancy, valid_mask, map_polylines=map_polylines, agent_preds=agent_preds, prev_traj=self._prev_traj, route=route, ) outputs["trajectory_costs"] = costs outputs["cost_diagnostics"] = cost_diag # ── Step 4: Emergency brake fallback (explicit warning) ──────────── emergency_used = torch.zeros(B, dtype=torch.bool, device=trajectories.device) for b in range(B): if not valid_mask[b].any(): brake_traj = self.emergency_brake(current_speed).to(trajectories.device) best_traj[b] = brake_traj[0] emergency_used[b] = True safety_diag[b].emergency_used = True warnings.warn( f"[FSD v5.3] Batch {b}: ALL {self.cfg.num_trajectories} proposals " f"rejected — emergency brake engaged (speed={current_speed:.1f} m/s).", RuntimeWarning, stacklevel=2, ) outputs["best_trajectory"] = best_traj # (B, T, 4) outputs["emergency_used"] = emergency_used # ── Step 5: Store for next-frame smoothness cost ─────────────────── self._prev_traj = best_traj.detach() outputs["prev_trajectory"] = self._prev_traj return outputs def reset_prev_trajectory(self) -> None: """Call at the start of each new episode to clear temporal state.""" self._prev_traj = None

#!/usr/bin/env python3 """export.py — TorchScript / ONNX export for FSD v5.3. Improvement over original: - export_onnx now clones the model to CPU instead of mutating the caller's instance in-place.
