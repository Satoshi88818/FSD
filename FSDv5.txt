targets["control"]) ** 2 / ctrl_var + 0.5 * ctrl_var.log() ).mean() # v5: Hungarian agent forecasting loss agent_loss = torch.tensor(0.0, device=device) if "agent_trajs" in outputs and "gt_agent_trajs" in targets: agent_loss = compute_agent_loss(outputs["agent_trajs"], targets["gt_agent_trajs"]) # v5: Chamfer map loss map_loss = torch.tensor(0.0, device=device) if "map_polylines" in outputs and "gt_polylines" in targets: map_loss = compute_map_loss( outputs["map_polylines"], targets["gt_polylines"], pred_cls=outputs.get("map_cls"), gt_cls=targets.get("gt_map_cls"), ) # v5: Flow-matching loss (Improvement 4) flow_matching_loss = torch.tensor(0.0, device=device) if "flow_loss" in outputs: flow_matching_loss = outputs["flow_loss"] # Occupancy flow occ_flow_loss = torch.tensor(0.0, device=device) if "occupancy_flow" in outputs and "gt_flow" in targets: occ_flow_loss = F.l1_loss(outputs["occupancy_flow"], targets["gt_flow"]) total = ( occ_loss + vel_loss + det_loss + planning_weight * ctrl_loss + agent_loss + map_loss + planning_weight * flow_matching_loss + flow_weight * occ_flow_loss ) return total, { "total": total.item(), "occ": occ_loss.item(), "vel": vel_loss.item(), "det": det_loss.item(), "ctrl": ctrl_loss.item(), "agent": agent_loss.item(), "map": map_loss.item(), "flow_match": flow_matching_loss.item(), "occ_flow": occ_flow_loss.item(), } def build_scheduler(optimizer, config: ModelConfig): def lr_lambda(step): if step < config.warmup_steps: return step / max(1, config.warmup_steps) progress = (step - config.warmup_steps) / max(1, config.total_steps - config.warmup_steps) return 0.5 * (1.0 + math.cos(math.pi * progress)) return torch.optim.lr_scheduler.LambdaLR(optimizer, lr_lambda) def flow_weight_ramp(step, config): return min(1.0, step / max(1, config.flow_loss_rampup_steps)) # ── NuScenesDataset (v4, unchanged structure) ───────────────────────────────── class NuScenesDataset(Dataset): """ nuScenes-compatible dataset loader stub. v5 addition: gt_map_cls field for map class supervision. Hardware Notes

PureE2EFSD: ~10–12 GB VRAM with AdaptiveToMe merging (T=6)

HybridFSD: ~8–10 GB VRAM; recommended for 16 GB cards

FSD: ~6 GB VRAM; auto-selected on smaller cards

Flow-matching overhead: flow_matching_steps Euler steps at inference (each is a small MLP forward pass); negligible vs. BEV lifting. Set flow_matching_steps=1 for fastest inference.

Hungarian matching: adds ~1–2 ms per batch during training (CPU-bound scipy call). For large batches, consider GPU-based assignment (e.g., torch_linear_sum_assignment).

Scene-token attention: scene_token_rounds=2 with num_agent_queries + map_num_polylines = 128 tokens adds ~3% overhead to TrajectoryHeadV2.
