#!/usr/bin/env python3 """rules.py — FSD v6.0 Traffic Rule Injector. Converts semantic traffic rules (red lights, stop signs, crosswalks) into dynamic virtual obstacles injected directly into the BEV occupancy grid. This allows the existing SafetyFilterV2 and CostPlannerV2 to automatically obey traffic laws without any rule-specific planning logic. Design principle: treat traffic rules as Dynamic Virtual Obstacles. A red-light stop line becomes a wall in occupancy space. """ from __future__ import annotations from dataclasses import dataclass from enum import Enum from typing import Dict, List, Optional import torch import torch.nn.functional as F from config import ModelConfig, TrafficRuleConfig # ── Traffic State Enums ─────────────────────────────────────────────────────── class LightState(Enum): RED = "RED" YELLOW = "YELLOW" GREEN = "GREEN" UNKNOWN = "UNKNOWN" class SignType(Enum): STOP = "STOP" YIELD = "YIELD" GIVE_WAY = "GIVE_WAY" @dataclass class TrafficElement: """A single traffic rule element in the scene.""" element_type: str # "light", "stop_sign", "yield_sign", "crosswalk" light_state: Optional[LightState] = None sign_type: Optional[SignType] = None # Stop-line or crosswalk polygon in BEV coordinates (N_pts, 2) — ego frame metres geometry_m: Optional[torch.Tensor] = None distance_m: float = 0.0 # ── Rasterisation Utilities ─────────────────────────────────────────────────── def _metres_to_grid( xy_m: torch.Tensor, # (..., 2) in ego-frame metres x_bound: tuple, # (x_min, x_max, resolution) y_bound: tuple, H: int, W: int, ) -> torch.Tensor: """Convert ego-frame metre coordinates to BEV grid pixel indices.""" x_min, x_max, x_res = x_bound y_min, y_max, y_res = y_bound px = ((xy_m[..., 0] - x_min) / x_res).long().clamp(0, W - 1) py = ((xy_m[..., 1] - y_min) / y_res).long().clamp(0, H - 1) return torch.stack([px, py], dim=-1) def _render_line_to_grid( grid: torch.Tensor, # (H, W) line_pts_m: torch.Tensor, # (N, 2) in metres value: float, x_bound: tuple, y_bound: tuple, half_width_px: int = 2, ) -> torch.Tensor: """Rasterise a polyline into a 2-D occupancy grid with given half-width.""" H, W = grid.shape px_pts = _metres_to_grid(line_pts_m, x_bound, y_bound, H, W) # Draw each segment for i in range(len(px_pts) - 1): x0, y0 = px_pts[i].tolist() x1, y1 = px_pts[i + 1].tolist() # Bresenham-style: sample at 2× resolution steps = max(abs(x1 - x0), abs(y1 - y0), 1) * 2 for s in range(steps + 1): t = s / steps xi = int(x0 + t * (x1 - x0)) yi = int(y0 + t * (y1 - y0)) xi_lo = max(0, xi - half_width_px) xi_hi = min(W, xi + half_width_px + 1) yi_lo = max(0, yi - half_width_px) yi_hi = min(H, yi + half_width_px + 1) grid[yi_lo:yi_hi, xi_lo:xi_hi] = max( grid[yi_lo:yi_hi, xi_lo:xi_hi].max().item(), value ) return grid # ── Traffic Rule Injector ───────────────────────────────────────────────────── class TrafficRuleInjector: """Injects traffic rules as virtual obstacles into the BEV occupancy grid. Usage (inside model.py forward()): injector = TrafficRuleInjector(config) occupancy = injector(occupancy, traffic_elements) The occupancy grid's obstacle channel (channel 1) is modified in-place (on a clone — original is not mutated). """ def __init__(self, config: ModelConfig): self.cfg = config self.rc: TrafficRuleConfig = config.traffic_rule_config x_min, x_max, x_res = config.x_bound y_min, y_max, y_res = config.y_bound self.H = int((y_max - y_min) / y_res) self.W = int((x_max - x_min) / x_res) self.half_width_px = max(1, int(self.rc.stop_line_width_m / x_res)) def __call__( self, occupancy: torch.Tensor, # (B, C, Z, H, W) traffic_elements: List[List[TrafficElement]], # outer: batch, inner: elements ) -> torch.Tensor: """Return a new occupancy tensor with traffic rules injected. Args: occupancy: BEV occupancy (B, C, Z, H, W). Channel 1 = obstacle. traffic_elements: Per-batch list of TrafficElement objects detected by perception (TrafficLightHead + MapFormer). Returns: Modified occupancy tensor, same shape. """ occ = occupancy.clone() B = occ.shape[0] for b in range(B): for elem in traffic_elements[b]: if elem.geometry_m is None: continue value = self._element_value(elem) if value <= 0.0: continue # Inject into all Z layers (stop line is vertical obstacle) for z in range(occ.shape[2]): occ[b, 1, z] = _render_line_to_grid( occ[b, 1, z], elem.geometry_m.to(occ.device), value=value, x_bound=self.cfg.x_bound, y_bound=self.cfg.y_bound, half_width_px=self.half_width_px, ) return occ def _element_value(self, elem: TrafficElement) -> float: """Map a traffic element to its occupancy injection value.""" if elem.element_type == "light": if elem.light_state == LightState.RED: return self.rc.stop_line_occupancy_value elif elem.light_state == LightState.YELLOW and self.rc.enable_yellow_slow: return self.rc.yellow_occupancy_value else: return 0.0 # Green: no obstacle elif elem.element_type == "stop_sign": return self.rc.stop_line_occupancy_value elif elem.element_type == "yield_sign": return self.rc.yield_sign_occupancy_value elif elem.element_type == "crosswalk": return self.rc.crosswalk_pedestrian_value return 0.0 # ── TrafficLightHead (perception module) ───────────────────────────────────── class TrafficLightHead(torch.nn.Module): """Lightweight head to classify traffic light state from BEV features. Outputs per-detected-light: state probabilities (Red / Yellow / Green / Unknown). In a full system this would attend to specific image crops of traffic lights. Here it runs on the 2-D BEV summary as a learned classifier. """ def __init__(self, in_channels: int = 64, num_lights: int = 4): super().__init__() self.num_lights = num_lights self.classifier = torch.nn.Sequential( torch.nn.AdaptiveAvgPool2d(8), torch.nn.Flatten(), torch.nn.Linear(in_channels * 64, 128), torch.nn.ReLU(inplace=True), torch.nn.Linear(128, num_lights * 4), # 4 classes per light ) def forward(self, bev_2d: torch.Tensor) -> torch.Tensor: """ Args: bev_2d: (B, C, H, W) 2-D BEV feature map. Returns: light_logits: (B, num_lights, 4) — [RED, YELLOW, GREEN, UNKNOWN] logits. """ B = bev_2d.shape[0] logits = self.classifier(bev_2d) # (B, num_lights * 4) return logits.view(B, self.num_lights, 4)

#!/usr/bin/env python3 """behavior.py — FSD v6.0 High-Level Behavior Planner. Implements a discrete intention state machine: KEEP_LANE | CHANGING_LEFT | CHANGING_RIGHT The BehaviorPlanner: 1. Observes the ego trajectory proposals to infer requested manoeuvre. 2. Locks-in an intention once committed (hysteresis prevents flip-flopping). 3. Returns an intention_cost tensor penalising proposals that contradict the locked intention — injected as the 8th cost term in CostPlannerV2. 4. Resets intention when manoeuvre completion is detected. """ from __future__ import annotations from enum import Enum from typing import Optional import torch from config import ModelConfig, BehaviorConfig # ── Intention Enum ──────────────────────────────────────────────────────────── class Intention(Enum): KEEP_LANE = "KEEP_LANE" CHANGING_LEFT = "CHANGING_LEFT" CHANGING_RIGHT = "CHANGING_RIGHT" # ── Behavior Planner ────────────────────────────────────────────────────────── class BehaviorPlanner: """Stateful behavior planner. Not an nn.Module — it holds frame-level discrete state and produces differentiable cost tensors for the neural cost planner. Example usage inside model.py: self.behavior_planner = BehaviorPlanner(config) # In forward(): intention_cost = self.behavior_planner.compute_intention_cost( trajectories, # (B, K, T, 4) map_polylines, # (B, P, pts, 2) or None ) # Pass intention_cost to CostPlannerV2 as extra cost term. """ def __init__(self, config: ModelConfig): self.cfg = config self.bc: BehaviorConfig = config.behavior_config # Per-batch state — reset when batch size changes or reset() called self._intention: list[Intention] = [] self._frames_in_intention: list[int] = [] self._pending_intention: list[Optional[Intention]] = [] self._pending_frames: list[int] = [] def reset(self, batch_size: int = 1) -> None: """Reset planner state. Call at episode boundaries.""" self._intention = [Intention.KEEP_LANE] * batch_size self._frames_in_intention = [0] * batch_size self._pending_intention = [None] * batch_size self._pending_frames = [0] * batch_size def _ensure_state(self, B: int) -> None: if len(self._intention) != B: self.reset(B) def current_intention(self, batch_idx: int = 0) -> Intention: self._ensure_state(max(batch_idx + 1, len(self._intention))) return self._intention[batch_idx] def compute_intention_cost( self, trajectories: torch.Tensor, # (B, K, T, 4) map_polylines: Optional[torch.Tensor] = None, # (B, P, pts, 2) ) -> torch.Tensor: """Compute per-proposal intention consistency cost. Returns: cost: (B, K) — high cost for proposals contradicting locked intention. """ B, K, T, _ = trajectories.shape self._ensure_state(B) costs = torch.zeros(B, K, device=trajectories.device) for b in range(B): # ── Infer requested manoeuvre from proposals ────────────────── lateral_disp = trajectories[b, :, -1, 1] # final lateral y for each proposal mean_lat = lateral_disp.mean().item() requested = self._classify_manoeuvre(mean_lat) # ── Hysteresis: require consistent signal before committing ──── if requested != self._intention[b]: if requested == self._pending_intention[b]: self._pending_frames[b] += 1 else: self._pending_intention[b] = requested self._pending_frames[b] = 1 if self._pending_frames[b] >= self.bc.hysteresis_frames: # Check that minimum lock-in time has elapsed if self._frames_in_intention[b] >= self.bc.lane_change_lock_frames \ or self._intention[b] == Intention.KEEP_LANE: self._intention[b] = requested self._frames_in_intention[b] = 0 self._pending_intention[b] = None self._pending_frames[b] = 0 else: self._frames_in_intention[b] += 1 self._pending_intention[b] = None self._pending_frames[b] = 0 # ── Assign cost per proposal ────────────────────────────────── intention = self._intention[b] for k in range(K): lat_k = trajectories[b, k, -1, 1].item() costs[b, k] = self._proposal_deviation(lat_k, intention) return costs * self.bc.intention_cost_weight def _classify_manoeuvre(self, mean_lateral_disp_m: float) -> Intention: """Map a mean lateral displacement to a coarse manoeuvre category.""" if mean_lateral_disp_m > 1.0: return Intention.CHANGING_LEFT elif mean_lateral_disp_m < -1.0: return Intention.CHANGING_RIGHT else: return Intention.KEEP_LANE def _proposal_deviation(self, lat_disp_m: float, intention: Intention) -> float: """Cost for a single proposal's lateral displacement given locked intention.""" if intention == Intention.KEEP_LANE: # Penalise lateral movement return abs(lat_disp_m) elif intention == Intention.CHANGING_LEFT: # Reward moving left (negative deviation from desired left offset) return max(0.0, 2.8 - lat_disp_m) elif intention == Intention.CHANGING_RIGHT: return max(0.0, 2.8 + lat_disp_m) return 0.0 def check_completion( self, ego_lateral_m: float, batch_idx: int = 0, ) -> bool: """Check whether the current lane-change manoeuvre is complete. Returns True if the intention was reset to KEEP_LANE. """ self._ensure_state(batch_idx + 1) intention = self._intention[batch_idx] threshold = self.bc.lane_change_complete_lat_m if intention == Intention.CHANGING_LEFT and ego_lateral_m >= threshold: self._intention[batch_idx] = Intention.KEEP_LANE self._frames_in_intention[batch_idx] = 0 return True elif intention == Intention.CHANGING_RIGHT and ego_lateral_m <= -threshold: self._intention[batch_idx] = Intention.KEEP_LANE self._frames_in_intention[batch_idx] = 0 return True return False 

optimizer.py (NEW)

#!/usr/bin/env python3 """optimizer.py — FSD v6.0 Gradient-Based Trajectory Optimizer. Implements Receding Horizon Optimisation: once CostPlannerV2 selects the best proposal out of K candidates, this module fine-tunes the winning trajectory's coordinates using PyTorch autograd. Because CostPlannerV2 uses only differentiable operations (F.grid_sample, torch.cdist, torch.norm, smooth cost functions), we can backpropagate through the cost directly into the trajectory coordinates — no approximate gradients. Latency budget: configured at 5 steps by default; on a 3090 this adds ~2–5 ms. Disable via OptimizerConfig(enable=False) for latency-critical deployments. """ from __future__ import annotations from typing import Callable, Optional import torch import torch.optim from config import ModelConfig, OptimizerConfig class TrajectoryOptimizer: """Gradient-based post-hoc trajectory refinement. This is NOT an nn.Module — it holds no trainable weights. It is instantiated once and called on each frame with the best trajectory selected by CostPlannerV2. Args: config: Full model config. cost_fn: A callable (traj: Tensor[B,T,4]) -> Tensor[B] that returns a scalar differentiable cost per batch item. Typically a curried call to CostPlannerV2's soft cost evaluation. """ def __init__(self, config: ModelConfig, cost_fn: Optional[Callable] = None): self.cfg = config self.oc: OptimizerConfig = config.optimizer_config self.cost_fn = cost_fn def optimize( self, best_traj: torch.Tensor, # (B, T, 4) cost_fn: Optional[Callable] = None, # override stored cost_fn ) -> torch.Tensor: """Refine trajectory coordinates using gradient descent. Args: best_traj: Best trajectory from CostPlannerV2. Shape (B, T, 4). cost_fn: Differentiable cost function (traj) -> scalar. If None, uses self.cost_fn set at construction. Returns: Refined trajectory, same shape (B, T, 4). Detached from graph. """ fn = cost_fn or self.cost_fn if fn is None or not self.oc.enable: return best_traj # Clone and enable gradients on trajectory coordinates only optimized = best_traj.clone().detach().requires_grad_(True) if self.oc.optimizer_type == "lbfgs": optim = torch.optim.LBFGS( [optimized], lr=self.oc.lr, max_iter=self.oc.num_steps, ) def closure(): optim.zero_grad() loss = fn(optimized).sum() loss = loss + self._smoothness_reg(optimized) loss.backward() torch.nn.utils.clip_grad_norm_([optimized], self.oc.clip_grad_norm) return loss optim.step(closure) else: # adam (default) optim = torch.optim.Adam([optimized], lr=self.oc.lr) for _ in range(self.oc.num_steps): optim.zero_grad() loss = fn(optimized).sum() loss = loss + self._smoothness_reg(optimized) loss.backward() torch.nn.utils.clip_grad_norm_([optimized], self.oc.clip_grad_norm) optim.step() return optimized.detach() def _smoothness_reg(self, traj: torch.Tensor) -> torch.Tensor: """L2 regularisation on trajectory second differences (acceleration).""" if traj.shape[1] < 3: return torch.tensor(0.0, device=traj.device) accel = traj[:, 2:, :2] - 2 * traj[:, 1:-1, :2] + traj[:, :-2, :2] return self.oc.smoothness_lambda * accel.pow(2).mean()

#!/usr/bin/env python3 """heads.py — FSD v6.0 perception and trajectory heads. v6.0 change: MotionFormer gains ego-conditioned forecasting support. forward_conditional(bev_2d, ego_proposals) → agent_preds (B, K, A, T, 2) All other heads (MapFormer, TrajectoryHeadV2, FlowMatchingTrajectoryHead, UncertaintyFusionV2) are unchanged from v5.3. """ from __future__ import annotations from typing import Optional, Tuple import torch import torch.nn as nn import torch.nn.functional as F from config import ModelConfig # ── MotionFormer ────────────────────────────────────────────────────────────── class MotionFormer(nn.Module): """Agent motion forecasting. Standard forward(): predicts one set of futures per agent. Unchanged from v5.3.""" def __init__(self, config: ModelConfig): super().__init__() self.cfg = config def forward(self, current_speed: float = 1.0) -> torch.Tensor: T = self.cfg.trajectory_horizon dt = self.cfg.trajectory_dt device = self.cfg.device traj = torch.zeros(1, T, 4, device=device) speed = max(current_speed, 0.0) x = 0.0 for t in range(T): decel = min(speed / dt, self.cfg.vehicle_params.max_decel_ms2 if hasattr(self.cfg, 'vehicle_params') else 3.0) speed = max(speed - decel * dt, 0.0) x += speed * dt traj[0, t, 0] = x traj[0, t, 3] = speed return traj

#!/usr/bin/env python3 """train_irl.py — FSD v6.0 MaxEnt IRL offline cost-weight training. Learns CostWeights by maximising the probability that the human expert's trajectory is assigned the lowest cost among K+1 candidates. Loss: Cross-Entropy(softmax(-costs), expert_index) where expert trajectory is appended as the (K+1)-th proposal. Usage: python train_irl.py --data /path/to/expert_demos --output checkpoints/irl/ Expert demo format (per sample): { "proposals": (K, T, 4) tensor — NN-generated proposals, "expert_traj": (T, 4) tensor — human ground-truth trajectory, "occupancy": (C, Z, H, W) tensor, "map_polylines":(P, pts, 2) tensor or None, "route": (R, 2) tensor or None, } """ from __future__ import annotations import argparse import os from typing import Optional import torch import torch.nn as nn import torch.nn.functional as F from torch.utils.data import DataLoader, Dataset from config import ModelConfig, CONFIG, IRLConfig # ── Differentiable Cost Planner (IRL version) ──────────────────────────────── class DifferentiableCostPlanner(nn.Module): """Cost planner with learnable weights (nn.Parameters). Identical cost formulas to CostPlannerV2 but weights are trainable. Used only during offline IRL training — not in inference. """ def __init__(self, config: ModelConfig): super().__init__() cw = config.cost_weights # Convert static floats to learnable parameters self.w_collision = nn.Parameter(torch.tensor(cw.collision_margin)) self.w_lane = nn.Parameter(torch.tensor(cw.lane)) self.w_interaction= nn.Parameter(torch.tensor(cw.interaction)) self.w_comfort = nn.Parameter(torch.tensor(cw.comfort)) self.w_smoothness = nn.Parameter(torch.tensor(cw.smoothness)) self.w_progress = nn.Parameter(torch.tensor(cw.progress)) self.w_route = nn.Parameter(torch.tensor(cw.route)) self.w_intention = nn.Parameter(torch.tensor(cw.intention)) self.cfg = config def forward( self, proposals: torch.Tensor, # (B, K, T, 4) expert_traj: torch.Tensor, # (B, T, 4) occupancy: torch.Tensor, # (B, C, Z, H, W) map_polylines: Optional[torch.Tensor] = None, route: Optional[torch.Tensor] = None, temperature: float = 1.0, ) -> torch.Tensor: """Compute cross-entropy IRL loss. The expert trajectory is appended as the (K+1)-th proposal. We want the model to assign it the lowest cost (highest reward). Returns: loss: scalar tensor """ B, K, T, _ = proposals.shape # Append expert as K+1-th proposal all_trajs = torch.cat([proposals, expert_traj.unsqueeze(1)], dim=1) # (B, K+1, T, 4) expert_idx = K # index of expert trajectory # Compute costs costs = self._total_cost(all_trajs, occupancy, map_polylines, route) # (B, K+1) # Convert costs to log-probs (lower cost = higher reward = higher prob) logits = -costs / temperature target = torch.full((B,), expert_idx, dtype=torch.long, device=costs.device) loss = F.cross_entropy(logits, target) return loss def _total_cost(self, trajs, occupancy, map_polylines, route): B, K, T, _ = trajs.shape device = trajs.device # Collision margin occ_2d = occupancy[:, 1:].max(1).values.max(1).values # (B, H, W) x_min, x_max, _ = self.cfg.x_bound y_min, y_max, _ = self.cfg.y_bound c_col = torch.zeros(B, K, device=device) for b in range(B): for k in range(K): worst = 0.0 for t in range(T): xn = 2 * (trajs[b, k, t, 0] - x_min) / (x_max - x_min) - 1 yn = 2 * (trajs[b, k, t, 1] - y_min) / (y_max - y_min) - 1 gp = torch.tensor([[[[xn, yn]]]], device=device) v = F.grid_sample(occ_2d[b].unsqueeze(0).unsqueeze(0), gp, align_corners=True)[0, 0, 0, 0].item() worst = max(worst, v) c_col[b, k] = worst # Lane cost if map_polylines is not None: ep = trajs[:, :, -1, :2] pc = map_polylines.mean(2) c_lane = torch.cdist(ep, pc).min(-1).values else: c_lane = torch.zeros(B, K, device=device) # Comfort (acceleration) if T >= 3: dt = self.cfg.trajectory_dt pos = trajs[..., :2] vel = (pos[:, :, 1:] - pos[:, :, :-1]) / dt acc = (vel[:, :, 1:] - vel[:, :, :-1]) / dt c_comfort = acc.norm(dim=-1).mean(-1) else: c_comfort = torch.zeros(B, K, device=device) # Progress c_progress = -(trajs[:, :, -1, 0] - trajs[:, :, 0, 0]) # Route if route is not None: ep = trajs[:, :, -1, :2] c_route = torch.cdist(ep, route).min(-1).values else: c_route = torch.zeros(B, K, device=device) return ( F.softplus(self.w_collision) * c_col + F.softplus(self.w_lane) * c_lane + F.softplus(self.w_comfort) * c_comfort + F.softplus(self.w_progress) * c_progress + F.softplus(self.w_route) * c_route ) def export_weights(self) -> dict: """Export learned weights as a plain dict for loading into CostWeights.""" return { "collision_margin": F.softplus(self.w_collision).item(), "lane": F.softplus(self.w_lane).item(), "interaction": F.softplus(self.w_interaction).item(), "comfort": F.softplus(self.w_comfort).item(), "smoothness": F.softplus(self.w_smoothness).item(), "progress": F.softplus(self.w_progress).item(), "route": F.softplus(self.w_route).item(), "intention": F.softplus(self.w_intention).item(), } # ── IRL Training Loop ───────────────────────────────────────────────────────── def train_irl( config: ModelConfig = CONFIG, data_path: Optional[str] = None, output_dir: str = "checkpoints/irl/", num_samples: int = 1000, ) -> DifferentiableCostPlanner: irl_cfg: IRLConfig = config.irl_config device = config.device planner = DifferentiableCostPlanner(config).to(device) optim = torch.optim.Adam( planner.parameters(), lr=irl_cfg.learning_rate, weight_decay=irl_cfg.weight_decay, ) # Dataset — use synthetic if no real data path provided dataset = SyntheticExpertDataset(num_samples, config) loader = DataLoader( dataset, batch_size=irl_cfg.batch_size, shuffle=True, collate_fn=collate_fn, ) os.makedirs(output_dir, exist_ok=True) print(f"[IRL] Starting training: {irl_cfg.num_epochs} epochs, " f"{len(dataset)} samples, batch={irl_cfg.batch_size}") for epoch in range(irl_cfg.num_epochs): epoch_loss = 0.0 for step, batch in enumerate(loader): proposals = batch["proposals"].to(device) expert_traj = batch["expert_traj"].to(device) occupancy = batch["occupancy"].to(device) map_polylines= batch["map_polylines"].to(device) route = batch["route"].to(device) optim.zero_grad() loss = planner( proposals, expert_traj, occupancy, map_polylines=map_polylines, route=route, temperature=irl_cfg.temperature, ) loss.backward() torch.nn.utils.clip_grad_norm_(planner.parameters(), 1.0) optim.step() epoch_loss += loss.item() avg_loss = epoch_loss / len(loader) if (epoch + 1) % irl_cfg.log_interval == 0: weights = planner.export_weights() print(f"[IRL] Epoch {epoch+1:3d}/{irl_cfg.num_epochs} | " f"Loss: {avg_loss:.4f} | Weights: {weights}") # Save learned weights torch.save(planner.state_dict(), os.path.join(output_dir, "irl_weights.pt")) weights = planner.export_weights() import json with open(os.path.join(output_dir, "learned_cost_weights.json"), "w") as f: json.dump(weights, f, indent=2) print(f"[IRL] Training complete. Weights saved to {output_dir}") print(f"[IRL] Learned weights: {weights}") return planner def main(): parser = argparse.ArgumentParser(description="FSD v6.0 MaxEnt IRL training") parser.add_argument("--data", type=str, default=None) parser.add_argument("--output", type=str, default="checkpoints/irl/") parser.add_argument("--samples", type=int, default=1000) args = parser.parse_args() train_irl(output_dir=args.output, num_samples=args.samples) if __name__ == "__main__": main()

Design Decisions & Engineering Notes

Traffic Rules — Why Virtual Obstacles?

Treating a red light's stop line as a 1.0 occupancy value is the simplest safe integration: zero new planner logic, full reuse of existing hard rejection and soft cost machinery. The stop line becomes physically indistinguishable from a wall. The only downside is that it cannot express "slow down but proceed" (yield) without a fractional value — addressed by yellow_occupancy_value=0.5.

Conditional Predictions — Dimensionality Increase

Changing agent_preds from (B, A, T, 2) to (B, K, A, T, 2) is the most invasive change. The SafetyFilterV2 and CostPlannerV2 both accept either shape (detected via dim() check) for backwards compatibility with v5.3 checkpoints and unit tests.

Computational note: Conditional forecasting scales as O(K × A). With K=8, A=32, the cross-attention adds ~0.5 ms. Reduce K or A if latency is tight.

Multi-Modal Commitment — Hysteresis Design

The BehaviorPlanner uses two-stage hysteresis:

Signal stability: intention switches only after hysteresis_frames consecutive consistent proposals.

Lock-in duration: once a lane change starts, it cannot be aborted until lane_change_lock_frames have elapsed (prevents mid-manoeuvre reversal).

This prevents the "wobble" problem without requiring any learned parameters.

Gradient Optimisation — Safety Constraint

The TrajectoryOptimizer optimises over the soft cost (comfort, smoothness, route) not the hard safety constraints. Hard constraints (occupancy > threshold, drivable area) remain enforced by SafetyFilterV2 before the optimizer runs. The optimizer can only make an already-safe trajectory smoother — it cannot make an unsafe trajectory pass safety.

IRL Weights — Softplus Constraint

Cost weights are passed through F.softplus during IRL training to ensure positivity. This is critical: a negative collision weight would actively seek collisions. After training, export_weights() returns the softplus-transformed values for use in the static CostWeights dataclass.
