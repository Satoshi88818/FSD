proposal_generator.py

""" proposal_generator.py --------------------- Learned trajectory proposal generator for FSD v7. v7 NEW MODULE addressing the #1 bottleneck in v6.1: proposals were passed in from outside, creating a hard dependency on an external (non-differentiable) planner. Two generation strategies: A. ConditionalVAE: latent-space generation conditioned on MapFormer outputs + occupancy. Enables diversity via sampling. Trained end-to-end with a coverage loss so the expert trajectory is always well-represented. B. DiverseProposalSampler: fast, deterministic fallback. - Mode 1: Top-K from cost map + Gaussian noise injection - Mode 2: Explicit behavioural modes (lane-follow, lane-change-left, lane-change-right, yield, stop) Both generators are pluggable through the CostFunctionRegistry interface. Coverage Loss ───────────── L_coverage = max_k [ ||expert_traj - proposal_k||_2 ] Penalises the generator when no proposal is close to the expert. Ensures the proposal set "covers" the training distribution. """ import math import torch import torch.nn as nn import torch.nn.functional as F from torch import Tensor from typing import Optional, Tuple, List from config import ModelConfig # ────────────────────────────────────────────────────────────────────────────── # Encoder / Decoder building blocks # ────────────────────────────────────────────────────────────────────────────── class _BEVContextEncoder(nn.Module): """Compress BEV + occupancy into a fixed-size context vector.""" def __init__(self, bev_channels: int, hidden_dim: int): super().__init__() self.net = nn.Sequential( nn.Conv2d(bev_channels + 1, hidden_dim, 3, stride=2, padding=1), nn.GroupNorm(8, hidden_dim), nn.GELU(), nn.Conv2d(hidden_dim, hidden_dim, 3, stride=2, padding=1), nn.GroupNorm(8, hidden_dim), nn.GELU(), nn.AdaptiveAvgPool2d(1), ) self.proj = nn.Linear(hidden_dim, hidden_dim) def forward(self, bev: Tensor, occupancy: Tensor) -> Tensor: occ = occupancy.unsqueeze(1) # (B, 1, H, W) x = torch.cat([bev, occ], dim=1) # (B, C+1, H, W) x = self.net(x).view(x.shape[0], -1) # (B, H) return self.proj(x) # (B, hidden_dim) class _TrajectoryEncoder(nn.Module): """Encode a trajectory (B, T, 4) → (B, latent_dim*2) [mu, log_var].""" def __init__(self, T: int, hidden_dim: int, latent_dim: int): super().__init__() self.net = nn.Sequential( nn.Linear(T * 4, hidden_dim), nn.GELU(), nn.Linear(hidden_dim, hidden_dim), nn.GELU(), ) self.mu_head = nn.Linear(hidden_dim, latent_dim) self.lv_head = nn.Linear(hidden_dim, latent_dim) def forward(self, traj: Tensor) -> Tuple[Tensor, Tensor]: B = traj.shape[0] h = self.net(traj.view(B, -1)) return self.mu_head(h), self.lv_head(h) class _TrajectoryDecoder(nn.Module): """Decode (B, latent_dim + hidden_dim) → (B, T, 4).""" def __init__(self, T: int, hidden_dim: int, latent_dim: int): super().__init__() self.T = T self.net = nn.Sequential( nn.Linear(latent_dim + hidden_dim, hidden_dim), nn.GELU(), nn.Linear(hidden_dim, hidden_dim), nn.GELU(), nn.Linear(hidden_dim, T * 4), ) def forward(self, z: Tensor, context: Tensor) -> Tensor: x = torch.cat([z, context], dim=-1) return self.net(x).view(z.shape[0], self.T, 4) # ────────────────────────────────────────────────────────────────────────────── # A. Conditional VAE Proposal Generator # ────────────────────────────────────────────────────────────────────────────── class ConditionalVAEProposalGenerator(nn.Module): """ cVAE conditioned on MapFormer context + occupancy. Training : encode expert trajectory → z, decode K samples around z. Inference : sample K latents from prior N(0,I), decode → K proposals. Args: config: ModelConfig """ def __init__(self, config: ModelConfig): super().__init__() H = config.hidden_dim L = config.proposal_latent_dim T = config.trajectory_horizon K = config.num_proposals self.K = K self.L = L self.context_encoder = _BEVContextEncoder(config.bev_channels, H) self.traj_encoder = _TrajectoryEncoder(T, H, L) self.decoder = _TrajectoryDecoder(T, H, L) def _reparameterise(self, mu: Tensor, log_var: Tensor) -> Tensor: std = torch.exp(0.5 * log_var) eps = torch.randn_like(std) return mu + eps * std def forward( self, bev: Tensor, # (B, C, H_bev, W_bev) occupancy: Tensor, # (B, H_bev, W_bev) expert_traj: Optional[Tensor] = None, # (B, T, 4) — only during training ) -> Tuple[Tensor, dict]: """ Returns: proposals : (B, K, T, 4) aux : dict with 'kl_loss', 'mu', 'log_var' (None at inference) """ B = bev.shape[0] context = self.context_encoder(bev, occupancy) # (B, H) if expert_traj is not None and self.training: # Posterior sampling during training mu, log_var = self.traj_encoder(expert_traj) z_base = self._reparameterise(mu, log_var) # (B, L) kl = -0.5 * (1 + log_var - mu.pow(2) - log_var.exp()).sum(dim=-1).mean() # Sample K-1 additional latents near the posterior mean noise = torch.randn(B, self.K - 1, self.L, device=bev.device) zs = torch.cat([z_base.unsqueeze(1), z_base.unsqueeze(1) + noise * 0.5], dim=1) zs = zs[:, :self.K] # (B, K, L) else: # Prior sampling at inference zs = torch.randn(B, self.K, self.L, device=bev.device) mu, log_var, kl = None, None, torch.tensor(0.0) # Decode: broadcast context to each of the K latents ctx_rep = context.unsqueeze(1).expand(B, self.K, -1).reshape(B * self.K, -1) zs_flat = zs.reshape(B * self.K, self.L) proposals = self.decoder(zs_flat, ctx_rep).view(B, self.K, -1, 4) return proposals, {"kl_loss": kl, "mu": mu, "log_var": log_var} # ────────────────────────────────────────────────────────────────────────────── # B. Diverse Proposal Sampler (fast, deterministic fallback) # ────────────────────────────────────────────────────────────────────────────── class DiverseProposalSampler: """ Fast proposal sampler that does not require training. Generates proposals from: 1. Top-K points on a precomputed cost map + Gaussian noise 2. Explicit behavioural anchors: lane-follow, lane-change-L/R, yield, stop Args: T : trajectory horizon noise_std : std of perturbation noise added to top-K proposals anchors : list of anchor trajectory templates (T, 4) """ def __init__( self, T: int, noise_std: float = 0.1, anchors: Optional[List[Tensor]] = None, ): self.T = T self.noise_std = noise_std self.anchors = anchors or [] def sample( self, cost_map: Tensor, # (B, H, W) — lower is better K: int, ego_state: Tensor, # (B, 4) — current [x, y, heading, speed] ) -> Tensor: """ Returns proposals: (B, K, T, 4). Simple heuristic: start from ego_state and project K random forward trajectories perturbed around a straight-line baseline. In production, this would use a motion primitive library. """ B, H, W = cost_map.shape device = cost_map.device # Straight-line baseline from ego state xy0 = ego_state[:, :2] # (B, 2) heading = ego_state[:, 2:3] # (B, 1) speed = ego_state[:, 3:4] # (B, 1) dt = 0.5 # seconds per step dx = speed * torch.cos(heading) * dt # (B, 1) dy = speed * torch.sin(heading) * dt # (B, 1) steps = torch.arange(1, self.T + 1, device=device, dtype=torch.float32) xs = xy0[:, 0:1] + dx * steps.unsqueeze(0) # (B, T) ys = xy0[:, 1:2] + dy * steps.unsqueeze(0) # (B, T) hs = heading.expand(B, self.T) ss = speed.expand(B, self.T) base = torch.stack([xs, ys, hs, ss], dim=-1) # (B, T, 4) # Replicate and perturb proposals = base.unsqueeze(1).expand(B, K, self.T, 4).clone() proposals += torch.randn_like(proposals) * self.noise_std # Overwrite first len(anchors) slots with explicit anchors if provided for i, anchor in enumerate(self.anchors[:K]): proposals[:, i] = anchor.to(device).unsqueeze(0).expand(B, self.T, 4) return proposals # (B, K, T, 4) # ────────────────────────────────────────────────────────────────────────────── # Coverage Loss # ────────────────────────────────────────────────────────────────────────────── def coverage_loss( expert_traj: Tensor, # (B, T, 4) proposals: Tensor, # (B, K, T, 4) ) -> Tensor: """ Coverage loss: penalise the generator when no proposal is close to the expert trajectory. L = min_k || expert_traj - proposals_k ||_2 (per batch element, then mean) Gradient flows into the proposal generator to encourage coverage. """ # (B, K, T, 4) - (B, 1, T, 4) → (B, K) diff = (proposals - expert_traj.unsqueeze(1)).norm(dim=(-1, -2)) min_dist = diff.min(dim=1).values # (B,) — distance to nearest proposal return min_dist.mean() 

cost_functions.py

""" cost_functions.py ----------------- Declarative cost function registry for FSD v7. v7 NEW MODULE: replaces the stub compute_costs() in v6.1 with a full suite of differentiable cost functions. Each function is registered in COST_REGISTRY so new costs can be added without modifying the planner. Registered costs ──────────────── soft_collision — proximity to occupied cells (occupancy grid) stop_line_violation — penalty for crossing active stop lines lane_center_deviation — penalise lateral deviation from lane centrelines kinematic_comfort — penalise jerk, lateral acceleration, curvature progress_towards_goal — encourage forward progress towards waypoint/goal rule_soft_penalty — soft penalty from TrafficRuleInjector Usage ───── from cost_functions import COST_REGISTRY, build_cost_fn cost_fn = build_cost_fn(config) costs = cost_fn(trajs, occupancy, map_out, goal, rule_injector) """ import torch import torch.nn as nn import torch.nn.functional as F from torch import Tensor from typing import Callable, Dict, Optional, Any from dataclasses import dataclass # ────────────────────────────────────────────────────────────────────────────── # Registry infrastructure # ────────────────────────────────────────────────────────────────────────────── _REGISTRY: Dict[str, Callable] = {} def register_cost(name: str): """Decorator: register a cost function by name.""" def _decorator(fn: Callable) -> Callable: _REGISTRY[name] = fn return fn return _decorator def list_costs() -> list: """Return names of all registered cost functions.""" return list(_REGISTRY.keys()) # ────────────────────────────────────────────────────────────────────────────── # Shared helpers # ────────────────────────────────────────────────────────────────────────────── def _sample_occupancy_at_traj(traj_xy: Tensor, occupancy: Tensor) -> Tensor: """ Bilinear-sample the occupancy map at each trajectory waypoint. Args: traj_xy : (B, T, 2) — x,y positions in BEV pixel coords occupancy: (B, H, W) — occupancy probability [0, 1] Returns: (B, T) — occupancy probability at each waypoint """ B, T, _ = traj_xy.shape _, H, W = occupancy.shape grid = traj_xy.clone() grid[..., 0] = 2.0 * grid[..., 0] / (W - 1) - 1.0 # x → col grid[..., 1] = 2.0 * grid[..., 1] / (H - 1) - 1.0 # y → row grid = grid.unsqueeze(2) # (B, T, 1, 2) sampled = F.grid_sample( occupancy.unsqueeze(1), grid, mode="bilinear", padding_mode="zeros", align_corners=True, ).squeeze(1).squeeze(-1) # (B, T) return sampled def _nearest_polyline_distance(traj_xy: Tensor, polylines: Tensor) -> Tensor: """ Approximate minimum distance from each traj waypoint to the nearest polyline. Args: traj_xy : (B, T, 2) polylines: (B, P, pts, 2) Returns: (B, T) — mean distance to nearest polyline segment """ B, T, _ = traj_xy.shape _, P, pts, _ = polylines.shape traj_exp = traj_xy.unsqueeze(2).unsqueeze(3) # (B, T, 1, 1, 2) poly_exp = polylines.unsqueeze(1) # (B, 1, P, pts, 2) dist = (traj_exp - poly_exp).norm(dim=-1) # (B, T, P, pts) min_dist = dist.min(dim=-1).values.min(dim=-1).values # (B, T) return min_dist # ────────────────────────────────────────────────────────────────────────────── # Registered cost functions # ────────────────────────────────────────────────────────────────────────────── @register_cost("soft_collision") def soft_collision_cost( trajs: Tensor, # (B, T, 4) or (B*K, T, 4) occupancy: Tensor, d_min: float = 1.0, **kwargs, ) -> Tensor: """ Soft collision cost: penalise proximity to occupied cells. Uses log-barrier approximation: log(1 + exp(occ / d_min)). Returns: (B,) scalar cost per trajectory. """ occ = _sample_occupancy_at_traj(trajs[..., :2], occupancy) # (B, T) cost = F.softplus(occ / d_min) return cost.mean(dim=-1) # (B,) @register_cost("stop_line_violation") def stop_line_violation_cost( trajs: Tensor, # (B, T, 4) stop_lines: Tensor, # (B, S, pts, 2) stop_conf: Tensor, # (B, S) — confidence [0, 1] activation_dist: float = 1.5, **kwargs, ) -> Tensor: """ Penalise trajectories that pass through active stop lines. The penalty is weighted by stop-line confidence. Returns: (B,) """ B, T, _ = trajs.shape dist = _nearest_polyline_distance(trajs[..., :2], stop_lines) # (B, T) # Per-stop-line min distance from trajectory _, S, pts, _ = stop_lines.shape traj_exp = trajs[:, :, :2].unsqueeze(2).unsqueeze(3) # (B, T, 1, 1, 2) stop_exp = stop_lines.unsqueeze(1) # (B, 1, S, pts, 2) stop_dist = (traj_exp - stop_exp).norm(dim=-1).min(dim=-1).values.min(dim=1).values # stop_dist: (B, S) # Soft violation: sigmoid that activates near stop line violation = torch.sigmoid(activation_dist - stop_dist) # (B, S) # Weight by confidence weighted = (violation * stop_conf).sum(dim=-1) # (B,) return weighted @register_cost("lane_center_deviation") def lane_center_deviation_cost( trajs: Tensor, # (B, T, 4) lane_centrelines: Tensor, # (B, L, pts, 2) **kwargs, ) -> Tensor: """ Penalise lateral deviation from the nearest lane centreline. Returns: (B,) """ dist = _nearest_polyline_distance(trajs[..., :2], lane_centrelines) # (B, T) return dist.mean(dim=-1) # (B,) @register_cost("kinematic_comfort") def kinematic_comfort_cost( trajs: Tensor, # (B, T, 4) jerk_weight: float = 0.5, curvature_weight: float = 1.0, lat_acc_weight: float = 1.0, **kwargs, ) -> Tensor: """ Penalise kinematically uncomfortable trajectories. Components: jerk magnitude, curvature, lateral acceleration. Returns: (B,) """ xy = trajs[..., :2] vel = torch.diff(xy, dim=1) # (B, T-1, 2) acc = torch.diff(vel, dim=1) # (B, T-2, 2) jerk = torch.diff(acc, dim=1) # (B, T-3, 2) acc_mag = acc.norm(dim=-1) # (B, T-2) jerk_mag = jerk.norm(dim=-1) # (B, T-3) # Lateral acceleration: component of acc perpendicular to velocity speed = vel.norm(dim=-1, keepdim=True).clamp(min=1e-4) vel_dir = vel[:, :-1] / speed[:, :-1] lat_acc = (acc * vel_dir).sum(dim=-1).abs() # dot then take magnitude # Curvature: |cross(d1, d2)| / (|d1| |d2|) d1 = vel[:, :-1] d2 = vel[:, 1:] cross = (d1[..., 0] * d2[..., 1] - d1[..., 1] * d2[..., 0]).abs() denom = (d1.norm(dim=-1) * d2.norm(dim=-1)).clamp(min=1e-6) curvature = (cross / denom).clamp(max=10.0) cost = ( jerk_weight * jerk_mag.mean(dim=-1) + lat_acc_weight * lat_acc.mean(dim=-1) + curvature_weight * curvature.mean(dim=-1) ) return cost # (B,) @register_cost("progress_towards_goal") def progress_towards_goal_cost( trajs: Tensor, # (B, T, 4) goal: Optional[Tensor], # (B, 2) or None **kwargs, ) -> Tensor: """ Encourage progress towards a goal waypoint. Cost = final distance to goal (negative progress = positive cost). Returns: (B,) """ if goal is None: return torch.zeros(trajs.shape[0], device=trajs.device) final_pos = trajs[:, -1, :2] # (B, 2) return (final_pos - goal).norm(dim=-1) # (B,) @register_cost("rule_soft_penalty") def rule_soft_penalty_cost( trajs: Tensor, rule_injector=None, occupancy: Optional[Tensor] = None, **kwargs, ) -> Tensor: """ Soft penalty from the TrafficRuleInjector occupancy grid. Samples the rule-modified occupancy at trajectory positions. Returns: (B,) """ if rule_injector is None or occupancy is None: return torch.zeros(trajs.shape[0], device=trajs.device) occ = _sample_occupancy_at_traj(trajs[..., :2], occupancy) return occ.mean(dim=-1) # (B,) # ────────────────────────────────────────────────────────────────────────────── # Learnable cost MLP (v7: optional weight MLP on top of raw features) # ────────────────────────────────────────────────────────────────────────────── class LearnableCostHead(nn.Module): """ Optional small MLP that takes the raw per-feature costs and produces a single scalar cost. Replaces fixed linear combination of weights, allowing the model to learn non-linear interactions between cost terms. """ def __init__(self, n_features: int, hidden: int = 64): super().__init__() self.net = nn.Sequential( nn.Linear(n_features, hidden), nn.GELU(), nn.Linear(hidden, 1), ) def forward(self, feature_costs: Tensor) -> Tensor: """ Args: feature_costs: (B, n_features) Returns: (B,) scalar cost """ return self.net(feature_costs).squeeze(-1) # ────────────────────────────────────────────────────────────────────────────── # Composite cost function builder # ────────────────────────────────────────────────────────────────────────────── class CompositeCostFunction(nn.Module): """ Combines all registered cost functions using learnable weights or a LearnableCostHead MLP. Args: config : ModelConfig with per-cost weights use_mlp : if True, use LearnableCostHead instead of fixed weights """ def __init__(self, config, use_mlp: bool = False): super().__init__() self.config = config self.use_mlp = use_mlp self.weights = nn.ParameterDict({ "soft_collision": nn.Parameter(torch.tensor(config.cost_w_collision)), "stop_line_violation": nn.Parameter(torch.tensor(config.cost_w_stop_line)), "lane_center_deviation": nn.Parameter(torch.tensor(config.cost_w_lane_deviation)), "kinematic_comfort": nn.Parameter(torch.tensor(config.cost_w_comfort)), "progress_towards_goal": nn.Parameter(torch.tensor(config.cost_w_progress)), "rule_soft_penalty": nn.Parameter(torch.tensor(config.cost_w_rules)), }) if use_mlp: self.mlp_head = LearnableCostHead(n_features=len(self.weights)) def forward( self, trajs: Tensor, # (B, T, 4) occupancy: Tensor, # (B, H, W) map_out=None, goal: Optional[Tensor] = None, rule_injector=None, ) -> Tensor: """Returns (B,) total cost per trajectory.""" feature_costs = [] stop_lines = map_out.stop_lines if map_out else None stop_conf = map_out.stop_conf if map_out else None lane_centrelines= map_out.lane_centrelines if map_out else None for name, fn in _REGISTRY.items(): c = fn( trajs=trajs, occupancy=occupancy, stop_lines=stop_lines, stop_conf=stop_conf, lane_centrelines=lane_centrelines, goal=goal, rule_injector=rule_injector, d_min=self.config.safety_d_min_base, ) feature_costs.append(c) stacked = torch.stack(feature_costs, dim=-1) # (B, n_features) if self.use_mlp: return self.mlp_head(stacked) else: w = torch.stack([self.weights[n] for n in _REGISTRY], dim=0).abs() return (stacked * w.unsqueeze(0)).sum(dim=-1) # Public alias COST_REGISTRY = _REGISTRY def build_cost_fn(config, use_mlp: bool = False) -> CompositeCostFunction: """Factory: create a CompositeCostFunction from config.""" return CompositeCostFunction(config, use_mlp=use_mlp) 

criticality.py

""" criticality.py -------------- Multi-factor criticality scoring for IRL sample weighting. v7 additions over v6.1: 1. Temporal criticality — penalise trajectories that become critical in the future (predictive, not just instantaneous). 2. Interaction criticality — uses actor prediction outputs to reason about future conflicts with other agents. 3. Surprise signal — prediction error from previous frames as a proxy for unexpected situations. """ import torch import torch.nn.functional as F from torch import Tensor from typing import Optional, Callable # ────────────────────────────────────────────────────────────────────────────── # Helper: minimum distance to occupied cells # ────────────────────────────────────────────────────────────────────────────── def compute_min_distance_to_occupancy( traj: Tensor, # (B, T, 2) — x,y positions in BEV grid coords occupancy: Tensor, # (B, H, W) — occupancy probability map [0, 1] sigma: float = 1.0, ) -> Tensor: """ Approximate minimum distance from each trajectory to any occupied cell. Strategy: convolve occupancy with a Gaussian distance-falloff kernel, then bilinear-sample the convolved map at trajectory positions. Fast on GPU; no KD-tree needed. Returns: (B,) — one approximate distance scalar per batch element. """ B, T, _ = traj.shape _, H, W = occupancy.shape k = 9 r = k // 2 ys, xs = torch.meshgrid( torch.arange(-r, r + 1, dtype=torch.float32, device=traj.device), torch.arange(-r, r + 1, dtype=torch.float32, device=traj.device), indexing="ij", ) kernel = torch.exp(-(xs ** 2 + ys ** 2) / (2 * sigma ** 2)) kernel = kernel / kernel.sum() kernel = kernel.view(1, 1, k, k) occ = occupancy.unsqueeze(1) # (B, 1, H, W) prox = F.conv2d(occ, kernel, padding=r).squeeze(1) # (B, H, W) grid = traj.clone() grid[..., 0] = 2.0 * grid[..., 0] / (W - 1) - 1.0 grid[..., 1] = 2.0 * grid[..., 1] / (H - 1) - 1.0 grid = grid.unsqueeze(2) # (B, T, 1, 2) sampled = F.grid_sample( prox.unsqueeze(1), grid, mode="bilinear", padding_mode="border", align_corners=True, ).squeeze(1).squeeze(-1) # (B, T) dist_approx = -torch.log(sampled.clamp(min=1e-6)) return dist_approx.mean(dim=1) # (B,) def compute_curvature(traj: Tensor) -> Tensor: """ Approximate signed curvature magnitude at each interior waypoint. Args: traj: (B, T, 2) Returns: (B, T-2) curvature magnitude """ p0, p1, p2 = traj[:, :-2], traj[:, 1:-1], traj[:, 2:] d1, d2 = p1 - p0, p2 - p1 cross = d1[..., 0] * d2[..., 1] - d1[..., 1] * d2[..., 0] n1 = d1.norm(dim=-1).clamp(min=1e-6) n2 = d2.norm(dim=-1).clamp(min=1e-6) return (cross.abs() / (n1 * n2)).clamp(max=10.0) # ────────────────────────────────────────────────────────────────────────────── # NEW v7: Temporal criticality # ────────────────────────────────────────────────────────────────────────────── def compute_temporal_criticality( expert_traj: Tensor, # (B, T, 4) occupancy: Tensor, # (B, H, W) horizon: int = 5, decay: float = 0.8, ) -> Tensor: """ Predictive criticality: compute proximity at each future sub-horizon and weight by a temporal decay. Trajectories that approach danger zones in the *near future* score higher than those that are currently safe but will become critical. Returns: (B,) temporal criticality score """ B, T, _ = expert_traj.shape score = torch.zeros(B, device=expert_traj.device) for t in range(1, min(horizon + 1, T)): weight = decay ** (horizon - t) # near future gets higher weight partial_traj = expert_traj[:, :t, :2] # (B, t, 2) dist = compute_min_distance_to_occupancy(partial_traj, occupancy) prox = torch.exp(-dist / 2.0) # (B,) score = score + weight * prox return score / (sum(decay ** (horizon - t) for t in range(1, min(horizon + 1, T))) + 1e-6) # ────────────────────────────────────────────────────────────────────────────── # NEW v7: Interaction criticality # ────────────────────────────────────────────────────────────────────────────── def compute_interaction_criticality( expert_traj: Tensor, # (B, T, 4) actor_predictions: Optional[Tensor], # (B, A, modes, T, 2) or None interaction_radius: float = 3.0, ) -> Tensor: """ Interaction criticality: measures how close the ego trajectory comes to predicted future positions of other agents. If actor predictions are unavailable, returns zero. Returns: (B,) interaction criticality score """ if actor_predictions is None: return torch.zeros(expert_traj.shape[0], device=expert_traj.device) B, T, _ = expert_traj.shape ego_xy = expert_traj[..., :2] # (B, T, 2) # actor_predictions: (B, A, modes, T, 2) # Take the most likely mode (index 0) for simplicity agent_xy = actor_predictions[:, :, 0, :, :] # (B, A, T, 2) # Distance: ego vs each agent at each timestep ego_exp = ego_xy.unsqueeze(1) # (B, 1, T, 2) dist = (ego_exp - agent_xy).norm(dim=-1) # (B, A, T) min_dist = dist.min(dim=1).values.min(dim=1).values # (B,) interaction_score = torch.exp(-min_dist / interaction_radius) return interaction_score # ────────────────────────────────────────────────────────────────────────────── # NEW v7: Surprise / prediction error signal # ────────────────────────────────────────────────────────────────────────────── def compute_surprise_score( prev_predicted_occ: Optional[Tensor], # (B, H, W) — occupancy predicted last frame current_occ: Optional[Tensor], # (B, H, W) — observed this frame ) -> Tensor: """ Surprise = L1 difference between predicted and observed occupancy. High surprise indicates an unexpected situation. Returns: (B,) surprise score, or zeros if inputs unavailable. """ if prev_predicted_occ is None or current_occ is None: return torch.zeros(current_occ.shape[0] if current_occ is not None else 1) surprise = (prev_predicted_occ - current_occ).abs().mean(dim=(-1, -2)) return surprise # ────────────────────────────────────────────────────────────────────────────── # Main: multi-factor criticality (v7) # ────────────────────────────────────────────────────────────────────────────── def compute_criticality( expert_traj: Tensor, # (B, T, 4) occupancy: Tensor, # (B, H, W) proposals: Tensor, # (B, K, T, 4) soft_cost_fn: Callable, actor_predictions: Optional[Tensor] = None, # (B, A, modes, T, 2) prev_predicted_occ: Optional[Tensor] = None, min_dist_threshold: float = 2.0, temporal_horizon: int = 5, w_proximity: float = 0.4, w_kinematic: float = 0.2, w_uniqueness: float = 0.2, w_temporal: float = 0.1, w_interaction: float = 0.1, ) -> Tensor: """ Multi-factor criticality (v7): - Instantaneous obstacle proximity - Kinematic stress - Expert uniqueness (vs. proposals) - Temporal criticality (predictive) - Interaction criticality (with agents) - Surprise signal (absorbed into proximity via occupancy) Returns: (B,) normalised criticality weights, mean ≈ 1 """ B, T, _ = expert_traj.shape xy = expert_traj[..., :2] # 1. Obstacle proximity ───────────────────────────────────────────────── dist = compute_min_distance_to_occupancy(xy, occupancy) proximity_score = torch.exp(-dist / min_dist_threshold) # 2. Kinematic stress ─────────────────────────────────────────────────── vel = torch.diff(xy, dim=1) acc = torch.diff(vel, dim=1) jerk = torch.diff(acc, dim=1) acc_mag = acc.norm(dim=-1).mean(dim=1) jerk_mag = jerk.norm(dim=-1).mean(dim=1) curvature = compute_curvature(xy).mean(dim=1) kinematic_score = acc_mag + 0.5 * jerk_mag + curvature # 3. Expert uniqueness ────────────────────────────────────────────────── expert_cost = soft_cost_fn(expert_traj, occupancy) K = proposals.shape[1] prop_flat = proposals.view(B * K, T, -1) occ_rep = occupancy.repeat_interleave(K, dim=0) prop_costs = soft_cost_fn(prop_flat, occ_rep).view(B, K) best_prop_cost = prop_costs.min(dim=1).values uniqueness = torch.relu(expert_cost - best_prop_cost) # 4. Temporal criticality ─────────────────────────────────────────────── temporal_score = compute_temporal_criticality(expert_traj, occupancy, horizon=temporal_horizon) # 5. Interaction criticality ──────────────────────────────────────────── interaction_score = compute_interaction_criticality(expert_traj, actor_predictions) interaction_score = interaction_score.to(expert_traj.device) # Surprise is additive to proximity surprise = compute_surprise_score(prev_predicted_occ, occupancy) surprise = surprise.to(expert_traj.device) proximity_score = proximity_score + 0.1 * surprise def _norm(x: Tensor) -> Tensor: return x / (x.mean() + 1e-6) criticality = ( w_proximity * _norm(proximity_score) + w_kinematic * _norm(kinematic_score) + w_uniqueness * _norm(uniqueness) + w_temporal * _norm(temporal_score) + w_interaction * _norm(interaction_score) ) return criticality / (criticality.mean() + 1e-6) # (B,), mean ≈ 1

""" refinement_net.py ----------------- Transformer-based amortized trajectory refiner. v7 additions over v6.1: 1. Cross-attention to map tokens / BEV occupancy features — the v6.1 refiner only had self-attention over trajectory tokens. 2. RoPE (Rotary Positional Encoding) instead of learned time embeddings for better length generalization. 3. Diffusion-style iterative refinement option (DDPM-lite): iterative denoising for higher quality at the cost of a few extra passes. 4. Multi-modal output: output M trajectory modes + mode scores. 5. CVaR-based mode selection at inference. """ import math import torch import torch.nn as nn import torch.nn.functional as F from torch import Tensor from typing import Optional, Tuple # ────────────────────────────────────────────────────────────────────────────── # Rotary Positional Encoding (RoPE) # ────────────────────────────────────────────────────────────────────────────── def _rope_freqs(dim: int, seq_len: int, device: torch.device) -> Tensor: """Precompute RoPE frequency matrix (seq_len, dim/2).""" inv_freq = 1.0 / (10000 ** (torch.arange(0, dim, 2, device=device).float() / dim)) t = torch.arange(seq_len, device=device).float() freqs = torch.outer(t, inv_freq) # (T, dim/2) return torch.cat([freqs, freqs], dim=-1) # (T, dim) def _apply_rope(x: Tensor, freqs: Tensor) -> Tensor: """ Apply RoPE to token sequence. Args: modes : (B, M, T, 4) scores : (B, M) log-softmax cost_fn : cost function callable alpha : tail fraction (0.2 = worst 20%) Returns: (B, T, 4) selected trajectory """ B, M, T, D = modes.shape probs = scores.exp() # (B, M) # Compute cost for each mode modes_flat = modes.view(B * M, T, D) # Replicate any extra kwargs that have batch dim costs_flat = cost_fn(modes_flat, **cost_kwargs) # (B*M,) costs = costs_flat.view(B, M) # (B, M) # Sort modes by cost (ascending) per batch element sorted_costs, sorted_idx = costs.sort(dim=1) sorted_probs = probs.gather(1, sorted_idx) # Find the CVaR threshold: smallest set of modes whose prob ≥ alpha cumprob = sorted_probs.cumsum(dim=1) cvar_mask = (cumprob <= alpha).float() # Add the first mode that crosses the alpha threshold cvar_mask[:, 1:] = cvar_mask[:, 1:] * (1 - cvar_mask[:, :-1]) cvar_mask = cvar_mask + (cvar_mask.sum(dim=1, keepdim=True) == 0).float() cvar_mask = cvar_mask.clamp(max=1.0) # Weighted average over the CVaR set weights = (cvar_mask * sorted_probs) weights = weights / (weights.sum(dim=1, keepdim=True) + 1e-6) # Re-index modes by sorted order idx_exp = sorted_idx.unsqueeze(-1).unsqueeze(-1).expand(B, M, T, D) sorted_modes = modes.gather(1, idx_exp) # (B, M, T, D) selected = (sorted_modes * weights.unsqueeze(-1).unsqueeze(-1)).sum(dim=1) return selected # (B, T, D) def select_mode_minimax_regret( modes: Tensor, # (B, M, T, 4) scores: Tensor, # (B, M) cost_fn, **cost_kwargs, ) -> Tensor: """ Minimax-regret mode selection: pick the mode with minimum worst-case regret (cost - best_possible_cost). Returns: (B, T, 4) """ B, M, T, D = modes.shape costs = cost_fn(modes.view(B * M, T, D), **cost_kwargs).view(B, M) best = costs.min(dim=1, keepdim=True).values regret = costs - best # (B, M) best_mode = regret.argmin(dim=1) # (B,) return modes[torch.arange(B), best_mode] # (B, T, D) # ────────────────────────────────────────────────────────────────────────────── # Distillation loss # ────────────────────────────────────────────────────────────────────────────── def refiner_distillation_loss( refiner: TrajectoryRefiner, traj_init: Tensor, cost_grad: Tensor, traj_teacher: Tensor, map_tokens: Optional[Tensor] = None, perturbation_std: float = 0.05, ) -> dict: """ L1 distillation from teacher + consistency regularisation. """ modes, scores = refiner(traj_init, cost_grad, map_tokens) # (B, M, T, 4) # Find mode closest to teacher diff = (modes - traj_teacher.unsqueeze(1)).norm(dim=(-1, -2)) # (B, M) best_mode = diff.argmin(dim=1) # (B,) best_traj = modes[torch.arange(modes.shape[0]), best_mode] # (B, T, 4) l1_loss = F.l1_loss(best_traj, traj_teacher) # Consistency regularisation noise = torch.randn_like(traj_init) * perturbation_std modes_p, _ = refiner(traj_init + noise, cost_grad, map_tokens) best_traj_p = modes_p[torch.arange(modes.shape[0]), best_mode] consistency_loss = F.l1_loss(best_traj, best_traj_p.detach()) total = l1_loss + 0.1 * consistency_loss return { "total": total, "l1": l1_loss.item(), "consistency": consistency_loss.item(), }

""" safety.py --------- Differentiable safety filter. v7 additions over v6.1: 1. Adaptive d_min: clearance grows with speed (faster → needs more room). 2. Args: traj: (B, T, 4) where traj[..., 3] is speed Returns: (B, T) adaptive d_min per timestep """ speed = traj[..., 3].clamp(min=0.0) # (B, T) return self.d_min_base + self.d_min_speed_gain * speed # ────────────────────────────────────────────────────────────────────── # Barrier loss (differentiable) # ────────────────────────────────────────────────────────────────────── def barrier_loss( self, traj: Tensor, # (B, T, 4) occupancy: Tensor, # (B, H, W) ) -> Tensor: """ Differentiable barrier loss. Includes: 1. Soft-plus barrier at trajectory waypoints. 2. CBF term: penalise negative rate of change of safety margin. """ B, T, _ = traj.shape _, H, W = occupancy.shape d_min = self._adaptive_d_min(traj) # (B, T) xy = traj[..., :2] grid = xy.clone() grid[..., 0] = 2.0 * grid[..., 0] / (W - 1) - 1.0 grid[..., 1] = 2.0 * grid[..., 1] / (H - 1) - 1.0 grid = grid.unsqueeze(2) # (B, T, 1, 2) occ_at_traj = F.grid_sample( occupancy.unsqueeze(1), grid, mode="bilinear", padding_mode="zeros", align_corners=True, ).squeeze(1).squeeze(-1) # (B, T) # Per-timestep barrier barrier = F.softplus(occ_at_traj / d_min) # (B, T) # CBF term: safety margin h(x) = d_min - occ_at_traj # Constraint: Δh + γ·h ≥ 0 → penalise negative (Δh + γ·h) h = d_min - occ_at_traj # (B, T) if T > 1: dh = torch.diff(h, dim=1) # (B, T-1) h_mid = h[:, :-1] # (B, T-1) cbf_violation = torch.relu(-(dh + self.cbf_gamma * h_mid)) cbf_term = cbf_violation.mean() else: cbf_term = torch.tensor(0.0, device=traj.device) return self.barrier_weight * barrier.mean() + self.barrier_weight * cbf_term # ────────────────────────────────────────────────────────────────────── # Reachability-aware check (NEW) # ────────────────────────────────────────────────────────────────────── def reachability_loss( self, traj: Tensor, # (B, T, 4) occupancy: Tensor, # (B, H, W) speed_scale: float = 0.5, ) -> Tensor: """ Approximate forward reachable set at each waypoint as a disc of radius r = speed * dt. """ B, T, _ = traj.shape _, H, W = occupancy.shape dt = 0.5 # seconds per step speed = traj[..., 3].clamp(min=0.0) # (B, T) radius = speed * dt * speed_scale # (B, T) — reachable radius # For each waypoint, sample occupancy in a ring around it. # Approximation: sample at 8 points on the reachability circle. angles = torch.linspace(0, 2 * 3.14159, 8, device=traj.device)[:-1] total_violation = torch.zeros(B, device=traj.device) xy = traj[..., :2] # (B, T, 2) for angle in angles: cos_a, sin_a = angle.cos(), angle.sin() offset = torch.stack([cos_a * radius, sin_a * radius], dim=-1) # (B, T, 2) sample_pts = xy + offset # (B, T, 2) grid = sample_pts.clone() grid[..., 0] = 2.0 * grid[..., 0] / (W - 1) - 1.0 grid[..., 1] = 2.0 * grid[..., 1] / (H - 1) - 1.0 grid = grid.unsqueeze(2) occ = F.grid_sample( occupancy.unsqueeze(1), grid, mode="bilinear", padding_mode="zeros", align_corners=True, ).squeeze(1).squeeze(-1) # (B, T) total_violation = total_violation + occ.mean(dim=-1) return total_violation / len(angles) # (B,) # ────────────────────────────────────────────────────────────────────── # PGD hard projection (inference) # ────────────────────────────────────────────────────────────────────── @torch.no_grad() def _pgd_project(self, traj: Tensor, occupancy: Tensor) -> Tensor: t = traj.clone().requires_grad_(True) for _ in range(self.pgd_steps): loss = self.barrier_loss(t, occupancy) + self.reachability_loss( t, occupancy ).mean() grad = torch.autograd.grad(loss, t)[0] t = (t - self.pgd_lr * grad).detach().requires_grad_(True) return t.detach() # ────────────────────────────────────────────────────────────────────── def forward( self, traj: Tensor, occupancy: Tensor, hard_project: bool = False, ) -> Tensor: if hard_project or not self.training: return self._pgd_project(traj, occupancy) return traj

""" losses.py --------- All training losses for FSD v7. v7 additions over v6.1: 1. CombinedIRLLoss: weighted sum of FocalCriticalityLoss + CostSensitiveRankingLoss (both existed separately in v6.1 but were never combined — now they are used together with tunable weights). 2. TrajectoryForecastingAuxLoss: auxiliary loss on MapFormer actor predictions (encourages accurate agent future prediction). 3. CoverageAuxLoss: wraps coverage_loss from proposal_generator.py into a training-friendly module. 4. CurriculumLossWrapper: scales criticality weights based on epoch for curriculum learning. """ import torch import torch.nn as nn import torch.nn.functional as F from torch import Tensor from typing import Optional # ────────────────────────────────────────────────────────────────────────────── # Focal Criticality Loss (from v6.1, unchanged) # ────────────────────────────────────────────────────────────────────────────── class FocalCriticalityLoss(nn.Module): """ Focal cross-entropy with optional per-sample criticality weighting. L_i = w_i * (1 - p_t)^gamma * CE_i gamma=0 → standard cross-entropy. gamma=2 → strong down-weighting of easy examples. """ def __init__(self, gamma: float = 2.0, reduction: str = "mean"): super().__init__() self.gamma = gamma self.reduction = reduction def forward( self, logits: Tensor, # (B, K+1) targets: Tensor, # (B,) long weights: Optional[Tensor] = None, # (B,) criticality ) -> Tensor: log_probs = F.log_softmax(logits, dim=-1) probs = log_probs.exp() target_lp = log_probs.gather(1, targets.unsqueeze(1)).squeeze(1) target_p = probs.gather(1, targets.unsqueeze(1)).squeeze(1) focal_w = (1.0 - target_p) ** self.gamma loss = -focal_w * target_lp if weights is not None: loss = loss * weights if self.reduction == "mean": return loss.mean() elif self.reduction == "sum": return loss.sum() return loss # ────────────────────────────────────────────────────────────────────────────── # Cost-Sensitive Ranking Loss (from v6.1, unchanged) # ────────────────────────────────────────────────────────────────────────────── class CostSensitiveRankingLoss(nn.Module): """ Hinge ranking loss: expert cost < all proposal costs by a margin. L_total = w_focal * L_focal + w_ranking * L_ranking """ def __init__( self, gamma: float = 2.0, margin: float = 0.5, w_focal: float = 1.0, w_ranking: float = 0.5, ): super().__init__() self.focal = FocalCriticalityLoss(gamma=gamma, reduction="mean") self.ranking = CostSensitiveRankingLoss(margin=margin) self.w_focal = w_focal self.w_ranking = w_ranking def forward( self, logits: Tensor, # (B, K+1) — planner logits targets: Tensor, # (B,) long expert_costs: Tensor, # (B,) proposal_costs: Tensor, # (B, K) weights: Optional[Tensor] = None, ) -> dict: l_focal = self.focal(logits, targets, weights) l_ranking = self.ranking(expert_costs, proposal_costs, weights) total = self.w_focal * l_focal + self.w_ranking * l_ranking return { "total": total, "focal": l_focal.item(), "ranking": l_ranking.item(), } # ────────────────────────────────────────────────────────────────────────────── # NEW: Trajectory Forecasting Auxiliary Loss # ────────────────────────────────────────────────────────────────────────────── class TrajectoryForecastingAuxLoss(nn.Module): """ Auxiliary loss on MapFormer actor predictions. Given ground-truth agent future trajectories and the M predicted modes, use winner-takes-all (WTA) with the best mode: compute L1 to the closest predicted mode and minimise it. Also add a cross-entropy term on mode scores to push probability mass to the correct mode. Args: n_modes : number of predicted modes per agent """ def __init__(self, n_modes: int = 3): super().__init__() self.n_modes = n_modes def forward( self, pred_traj: Tensor, # (B, A, M, T, 2) — predicted modes pred_scores: Tensor, # (B, A, M) — log-softmax gt_traj: Tensor, # (B, A, T, 2) — ground truth gt_mask: Optional[Tensor] = None, # (B, A) bool — valid agents ) -> Tensor: """ Returns scalar loss. """ B, A, M, T, _ = pred_traj.shape # Winner-takes-all: find mode closest to GT for each agent gt_exp = gt_traj.unsqueeze(2).expand_as(pred_traj) # (B, A, M, T, 2) dist = (pred_traj - gt_exp).norm(dim=-1).mean(dim=-1) # (B, A, M) best_m = dist.argmin(dim=-1) # (B, A) # L1 loss on best mode best_pred = pred_traj[ torch.arange(B).unsqueeze(1).expand(B, A), torch.arange(A).unsqueeze(0).expand(B, A), best_m, ] # (B, A, T, 2) l1 = F.l1_loss(best_pred, gt_traj, reduction="none").mean(dim=(-1, -2)) # (B, A) # Mode classification loss: encourage high score for winning mode ce = F.nll_loss( pred_scores.view(B * A, M), best_m.view(B * A), reduction="none", ).view(B, A) # (B, A) loss = l1 + 0.1 * ce # (B, A) if gt_mask is not None: loss = loss * gt_mask.float() return loss.sum() / (gt_mask.float().sum() + 1e-6) return loss.mean() # ────────────────────────────────────────────────────────────────────────────── # NEW: Curriculum Loss Wrapper # ────────────────────────────────────────────────────────────────────────────── class CurriculumLossWrapper(nn.Module): """ Scales per-sample criticality weights based on training epoch. Strategy: - Early epochs: only use samples with criticality below curriculum_threshold (low-criticality curriculum). - Later epochs: gradually include harder (high-criticality) samples. This prevents the planner from immediately overfit to rare difficult cases before learning the common-case distribution. Args: base_loss : the underlying loss module to wrap start_epoch : epoch at which curriculum begins end_epoch : epoch at which full data is used crit_start : initial criticality cap (0.0–1.0 quantile) crit_end : final criticality cap (1.0 = all data) """ def __init__( self, base_loss: nn.Module, start_epoch: int = 0, end_epoch: int = 20, crit_start: float = 0.3, crit_end: float = 1.0, ): super().__init__() self.base_loss = base_loss self.start_epoch = start_epoch self.end_epoch = end_epoch self.crit_start = crit_start self.crit_end = crit_end self._epoch = 0 def set_epoch(self, epoch: int): self._epoch = epoch def _criticality_cap(self) -> float: if self._epoch <= self.start_epoch: return self.crit_start if self._epoch >= self.end_epoch: return self.crit_end progress = (self._epoch - self.start_epoch) / (self.end_epoch - self.start_epoch) return self.crit_start + progress * (self.crit_end - self.crit_start) def forward(self, weights: Tensor, *args, **kwargs): cap = self._criticality_cap() # Clamp criticality weights to [0, cap * max_weight] max_w = weights.max().clamp(min=1e-6) weights = weights.clamp(max=cap * max_w) return self.base_loss(*args, weights=weights, **kwargs)

────────────────────────────────────────────────────────────────────────────── # Prioritized Replay Buffer # ────────────────────────────────────────────────────────────────────────────── class PrioritizedReplayBuffer: """ Replay buffer with priority proportional to criticality × loss. Samples high-priority (hard, critical) examples more frequently. Uses proportional prioritization (no IS weights for simplicity). Args: max_size : maximum buffer capacity alpha : priority exponent (0 = uniform, 1 = full priority) """ def __init__(self, max_size: int = 10_000, alpha: float = 0.6): self.max_size = max_size self.alpha = alpha self._buffer: List[Dict[str, Any]] = [] self._priorities: List[float] = [] def add(self, sample: dict, priority: float): """Add a sample with given priority.""" if len(self._buffer) >= self.max_size: # Remove lowest-priority entry min_idx = self._priorities.index(min(self._priorities)) self._buffer.pop(min_idx) self._priorities.pop(min_idx) self._buffer.append(sample) self._priorities.append(max(priority, 1e-6)) def sample(self, n: int) -> List[dict]: """Sample n entries proportional to priority^alpha.""" if len(self._buffer) < n: return random.choices(self._buffer, k=n) priorities = torch.tensor(self._priorities, dtype=torch.float32) probs = (priorities ** self.alpha) probs = probs / probs.sum() idx = torch.multinomial(probs, n, replacement=False).tolist() return [self._buffer[i] for i in idx] def __len__(self) -> int: return len(self._buffer) # ────────────────────────────────────────────────────────────────────────────── # Adversarial Actor Augmentation # ────────────────────────────────────────────────────────────────────────────── def adversarial_actor_augment( occupancy: Tensor, # (B, H, W) n_actors: int = 2, actor_radius: int = 3, high_crit_mask: Optional[Tensor] = None, # (B,) bool — only augment these ) -> Tensor: """ Synthetic augmentation: randomly place adversarial actor blobs in the occupancy map for high-criticality samples. In production this would use a learned adversarial policy; here we use random placement near the ego path as a simple approximation. Returns: Modified occupancy (B, H, W) """ occ = occupancy.clone() B, H, W = occ.shape for b in range(B): if high_crit_mask is not None and not high_crit_mask[b]: continue for _ in range(n_actors): row = random.randint(actor_radius, H - actor_radius - 1) col = random.randint(actor_radius, W - actor_radius - 1) # Gaussian blob for dr in range(-actor_radius, actor_radius + 1): for dc in range(-actor_radius, actor_radius + 1): r2 = dr ** 2 + dc ** 2 v = torch.exp(torch.tensor(-r2 / (actor_radius ** 2), dtype=torch.float32)) rr, cc = row + dr, col + dc if 0 <= rr < H and 0 <= cc < W: occ[b, rr, cc] = (occ[b, rr, cc] + v).clamp(max=1.0) return occ # ────────────────────────────────────────────────────────────────────────────── # Latency budget tracker # ────────────────────────────────────────────────────────────────────────────── class LatencyTracker: """ Context manager that measures forward pass time and warns if it exceeds the budget. Usage: tracker = LatencyTracker(budget_ms=8.0) with tracker("forward_pass"): output = model(input) """ def __init__(self, budget_ms: float = 8.0): self.budget_ms = budget_ms self._start: Optional[float] = None self._label: str = "" def __call__(self, label: str) -> "LatencyTracker": self._label = label return self def __enter__(self): torch.cuda.synchronize() if torch.cuda.is_available() else None self._start = time.perf_counter() return self def __exit__(self, *args): torch.cuda.synchronize() if torch.cuda.is_available() else None elapsed_ms = (time.perf_counter() - self._start) * 1000 if elapsed_ms > self.budget_ms: import warnings warnings.warn( f"[LatencyTracker] '{self._label}' took {elapsed_ms:.2f} ms " f"(budget: {self.budget_ms:.1f} ms)", RuntimeWarning, ) return elapsed_ms # ────────────────────────────────────────────────────────────────────────────── # Training step (v7) # ────────────────────────────────────────────────────────────────────────────── def irl_train_step( planner: DifferentiableCostPlanner, optimizer: torch.optim.Optimizer, batch: dict, soft_cost_fn, replay_buffer: Optional[PrioritizedReplayBuffer] = None, actor_aux_loss: Optional[TrajectoryForecastingAuxLoss] = None, curriculum_wrapper: Optional[CurriculumLossWrapper] = None, config: Optional[ModelConfig] = None, latency_tracker: Optional[LatencyTracker] = None, ) -> dict: """ Full v7 IRL training step. batch keys: proposals : (B, K, T, 4) expert_traj : (B, T, 4) occupancy : (B, H, W) map_out : MapFormerOutput (optional) actor_traj : (B, A, M, T_a, 2) actor predictions (optional) gt_agent_traj: (B, A, T_a, 2) GT agent trajectories (optional) proposals_gen: dict from proposal generator (contains kl_loss) """ proposals = batch["proposals"] expert_traj = batch["expert_traj"] occupancy = batch["occupancy"] map_out = batch.get("map_out") proposals_info = batch.get("proposals_gen", {}) B = proposals.shape[0] #
