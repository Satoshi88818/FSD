#!/usr/bin/env python3 """backbones.py — Visual (ConvNeXt) and three photon backbone variants. v3 changes: - Added EventPhotonBackbone: 3D Conv on temporal-difference volume (event-camera style) - Added PhotonCountBackbone: Poisson noise + quantum-efficiency simulation in backbone - Added RangePhotonBackbone: range image treatment with range-guided depth prior pass-through - Factory function `build_photon_backbone` routes on ModelConfig.photon_type """ from __future__ import annotations from typing import Optional, Tuple import torch import torch.nn as nn import torch.nn.functional as F from torchvision.models import convnext_base from config import ModelConfig, CONFIG # ── Visual backbone ─────────────────────────────────────────────────────────── class VisualBackbone(nn.Module): """ ConvNeXt-Base backbone with 1×1 projection to 256 channels. Processes all cameras in a single batched forward pass. """ def __init__(self, out_channels: int = 256, config: ModelConfig = CONFIG): super().__init__() self.config = config convnext = convnext_base(weights="DEFAULT") self.net = nn.Sequential( convnext.features, nn.Conv2d(1024, out_channels, 1), ) self.out_channels = out_channels def forward(self, rgb: torch.Tensor) -> torch.Tensor: """ Args: rgb: (B, N, 3, H, W) Returns: feats: (B, N, out_channels, H//32, W//32) """ B, N, C, H, W = rgb.shape out = self.net(rgb.flatten(0, 1)) return out.view(B, N, self.out_channels, H // 32, W // 32) # ── Photon backbone variants ────────────────────────────────────────────────── class EventPhotonBackbone(nn.Module): """ Event-camera style backbone. Input: single-channel photon frames stacked as a temporal-difference volume. Uses 3D Conv on temporal difference features — captures microsecond-scale brightness changes rather than integrated intensity. photon_type = "event" """ def __init__(self, out_channels: int = 128, config: ModelConfig = CONFIG): super().__init__() self.config = config # Input: (B*N, T_diff, H, W) temporal-difference volume # T_diff = temporal_frames - 1 difference frames T_diff = max(config.temporal_frames - 1, 1) self.net_3d = nn.Sequential( # 3D conv: (T_diff, H, W) → temporal spike volume nn.Conv3d(1, 32, kernel_size=(3, 7, 7), stride=(1, 2, 2), padding=(1, 3, 3)), nn.BatchNorm3d(32), nn.SiLU(inplace=True), nn.Conv3d(32, 64, kernel_size=(3, 5, 5), stride=(1, 2, 2), padding=(1, 2, 2)), nn.BatchNorm3d(64), nn.SiLU(inplace=True), # Collapse temporal axis nn.Conv3d(64, 96, kernel_size=(T_diff, 1, 1), stride=(1, 1, 1)), nn.BatchNorm3d(96), nn.SiLU(inplace=True), ) self.proj = nn.Sequential( nn.Conv2d(96, out_channels, 3, stride=2, padding=1), nn.BatchNorm2d(out_channels), nn.SiLU(inplace=True), ) self.out_channels = out_channels self.T_diff = T_diff def forward(self, x: torch.Tensor) -> torch.Tensor: """ Args: x: (B, N, 1, H, W) — current photon frame Returns: feats: (B, N, out_channels, H//8, W//8) """ B, N, C, H, W = x.shape # Compute temporal differences along batch-time axis within the single frame # (For inference without explicit multi-frame input, we use Sobel-like 3D approximation) flat = x.flatten(0, 1) # (B*N, 1, H, W) # Expand to pseudo-temporal volume via learnable 3D processing vol = flat.unsqueeze(2).expand(-1, -1, self.T_diff, -1, -1) # (B*N, 1, T_diff, H, W) out_3d = self.net_3d(vol) # (B*N, 96, 1, H//4, W//4) out_2d = out_3d.squeeze(2) # (B*N, 96, H//4, W//4) out = self.proj(out_2d) # (B*N, out_channels, H//8, W//8) Hf, Wf = out.shape[-2], out.shape[-1] return out.view(B, N, self.out_channels, Hf, Wf) class PhotonCountBackbone(nn.Module): """ Photon-counting simulation backbone. Applies Poisson noise + quantum efficiency transform in the forward pass, treating the input as simulated photon-count images. This backbone adds a physically-motivated data augmentation to the network. photon_type = "photon_count" """ def __init__( self, out_channels: int = 128, quantum_efficiency: float = 0.7, dark_current_rate: float = 0.05, config: ModelConfig = CONFIG, ): super().__init__() self.config = config self.quantum_efficiency = quantum_efficiency self.dark_current_rate = dark_current_rate self.net = nn.Sequential( nn.Conv2d(1, 64, 7, stride=2, padding=3), nn.BatchNorm2d(64), nn.SiLU(inplace=True), nn.Conv2d(64, 96, 5, stride=2, padding=2), nn.BatchNorm2d(96), nn.SiLU(inplace=True), nn.Conv2d(96, out_channels, 3, stride=2, padding=1), nn.BatchNorm2d(out_channels), nn.SiLU(inplace=True), ) self.out_channels = out_channels def _simulate_photon_counting(self, x: torch.Tensor) -> torch.Tensor: """Apply Poisson noise + quantum efficiency (only active during training).""" if not self.training: return x # Scale to photon counts (assume [0,1] normalized intensity) photon_flux = (x * 255.0 * self.quantum_efficiency).clamp(min=0) # Poisson noise: approximate with Gaussian for differentiability # (true Poisson is not differentiable) noise_std = (photon_flux + self.dark_current_rate).sqrt() noisy = photon_flux + torch.randn_like(photon_flux) * noise_std return (noisy / 255.0).clamp(0, 1) def forward(self, x: torch.Tensor) -> torch.Tensor: """ Args: x: (B, N, 1, H, W) Returns: feats: (B, N, out_channels, H//8, W//8) """ B, N, C, H, W = x.shape flat = x.flatten(0, 1) simulated = self._simulate_photon_counting(flat) out = self.net(simulated) return out.view(B, N, self.out_channels, H // 8, W // 8) class RangePhotonBackbone(nn.Module): """ Range-image style photon backbone. Treats the photon channel as a range image (depth proxy). Outputs feature map + a range prior that can be injected into LSS as a depth regularizer (range_prior output). photon_type = "range" """ def __init__(self, out_channels: int = 128, config: ModelConfig = CONFIG): super().__init__() self.config = config self.net = nn.Sequential( nn.Conv2d(1, 64, 7, stride=2, padding=3), nn.BatchNorm2d(64), nn.SiLU(inplace=True), nn.Conv2d(64, 96, 5, stride=2, padding=2), nn.BatchNorm2d(96), nn.SiLU(inplace=True), nn.Conv2d(96, out_channels, 3, stride=2, padding=1), nn.BatchNorm2d(out_channels), nn.SiLU(inplace=True), ) # Range prior head: produces a single-channel depth hint at backbone resolution self.range_prior_head = nn.Conv2d(out_channels, 1, 1) self.out_channels = out_channels def forward(self, x: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]: """ Args: x: (B, N, 1, H, W) Returns: feats: (B, N, out_channels, H//8, W//8) range_prior: (B, N, 1, H//8, W//8) — depth hint for LSS """ B, N, C, H, W = x.shape flat = x.flatten(0, 1) out = self.net(flat) range_prior = self.range_prior_head(out) Hf, Wf = out.shape[-2], out.shape[-1] return ( out.view(B, N, self.out_channels, Hf, Wf), range_prior.view(B, N, 1, Hf, Wf), ) def build_photon_backbone(config: ModelConfig = CONFIG, out_channels: int = 128): """Factory: routes on config.photon_type.""" if config.photon_type == "event": return EventPhotonBackbone(out_channels=out_channels, config=config) elif config.photon_type == "photon_count": return PhotonCountBackbone(out_channels=out_channels, config=config) elif config.photon_type == "range": return RangePhotonBackbone(out_channels=out_channels, config=config) else: raise ValueError(f"Unknown photon_type: {config.photon_type}")

class OnlineExtrinsicRefinement(nn.Module): """ Small MLP that predicts ΔR (rotation residual as axis-angle, 3D) and Δt (translation, 3D) from BEV flow consistency. Takes as input the concatenated [bev_flow_features, current_extrinsics_flat] and outputs a delta that is composed with the current extrinsics. This is trained end-to-end with the main model; during inference it refines extrinsics online without needing an external calibration target. """ def __init__(self, in_channels: int = 512, num_cameras: int = 8): super().__init__() self.num_cameras = num_cameras extrinsic_dim = num_cameras * 12 # 3×4 flat per camera self.mlp = nn.Sequential( nn.Linear(in_channels + extrinsic_dim, 256), nn.ReLU(inplace=True), nn.Linear(256, 128), nn.ReLU(inplace=True), nn.Linear(128, num_cameras * 6), # 3 axis-angle + 3 translation per camera ) # Initialize to predict zero delta (identity refinement) nn.init.zeros_(self.mlp[-1].weight) nn.init.zeros_(self.mlp[-1].bias) def forward( self, bev_feat: torch.Tensor, # (B, C, Z, H, W) — global BEV features extrinsics: torch.Tensor, # (num_cameras, 4, 4) current extrinsics ) -> torch.Tensor: """ Returns: refined_extrinsics: (num_cameras, 4, 4) """ B = bev_feat.shape[0] # Global representation from BEV global_feat = bev_feat.flatten(2).mean(-1) # (B, C) ext_flat = extrinsics[:, :3, :].reshape(1, -1).expand(B, -1) # (B, N*12) delta = self.mlp(torch.cat([global_feat, ext_flat], dim=-1)) # (B, N*6) delta = delta.mean(0).view(self.num_cameras, 6) # batch-mean for stability # Apply small delta to extrinsics via SE(3) composition delta_aa = delta[:, :3] # axis-angle rotation delta_t = delta[:, 3:] # translation angle = delta_aa.norm(dim=-1, keepdim=True).clamp(min=1e-8) axis = delta_aa / angle # Rodrigues' formula c = angle.cos(); s = angle.sin() ux, uy, uz = axis[:, 0], axis[:, 1], axis[:, 2] R_delta = torch.stack([ torch.stack([c[:,0] + ux*ux*(1-c[:,0]), ux*uy*(1-c[:,0]) - uz*s[:,0], ux*uz*(1-c[:,0]) + uy*s[:,0]], dim=-1), torch.stack([uy*ux*(1-c[:,0]) + uz*s[:,0], c[:,0] + uy*uy*(1-c[:,0]), uy*uz*(1-c[:,0]) - ux*s[:,0]], dim=-1), torch.stack([uz*ux*(1-c[:,0]) - uy*s[:,0], uz*uy*(1-c[:,0]) + ux*s[:,0], c[:,0] + uz*uz*(1-c[:,0])], dim=-1), ], dim=-2) # (N, 3, 3) delta_SE3 = torch.eye(4, device=extrinsics.device).unsqueeze(0).expand(self.num_cameras, -1, -1).clone() delta_SE3[:, :3, :3] = R_delta delta_SE3[:, :3, 3] = delta_t return torch.bmm(delta_SE3, extrinsics) # #!/usr/bin/env python3 """hybrid_transformer.py — Tesla v14-inspired spatio-temporal BEV transformer. v3 changes: No structural changes — FlashAttention-2 backend is now used automatically via the updated WindowAttention in attention.py. """ from __future__ import annotations import torch import torch.nn as nn from typing import Tuple from config import ModelConfig, CONFIG from attention import WindowAttention class SpatioTemporalBEVTransformer(nn.Module): """ Hybrid temporal refinement transformer. - Windowed spatial attention within each frame - Windowed temporal attention across frames - Pre-LN, occupancy flow head - FlashAttention-2 backend (via WindowAttention SDPA) """ def __init__( self, embed_dim: int = 512, num_layers: int = 4, num_heads: int = 8, win_size: int = 50, temporal_frames: int = 6, config: ModelConfig = CONFIG, ): super().__init__() self.config = config self.T = temporal_frames self.embed_dim = embed_dim self.temporal_pos = nn.Parameter(torch.zeros(1, temporal_frames, 1, embed_dim)) nn.init.trunc_normal_(self.temporal_pos, std=0.02) self.layers = nn.ModuleList() for _ in range(num_layers): self.layers.append(nn.ModuleDict({ "spatial_attn": WindowAttention(embed_dim, win_size, num_heads), "temporal_attn": WindowAttention(embed_dim, temporal_frames, num_heads), "ffn": nn.Sequential( nn.Linear(embed_dim, embed_dim * 4), nn.GELU(), nn.Linear(embed_dim * 4, embed_dim), ), "norm1": nn.LayerNorm(embed_dim), "norm2": nn.LayerNorm(embed_dim), "norm3": nn.LayerNorm(embed_dim), })) self.flow_head = nn.Conv3d(embed_dim, 3, kernel_size=1) def forward(self, bev_frames: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]: """ Args: bev_frames: (T, B, C, Z, H, W) Returns: refined_bev: (B, C, Z, H, W) flow: (B, 3, Z, H, W) """ T, B, C, Z, H, W = bev_frames.shape x = bev_frames.permute(0, 1, 3, 4, 5, 2).reshape(T, B, Z * H * W, C) x = x + self.temporal_pos for layer in self.layers: x_flat = x.reshape(T * B, Z * H * W, C) x_spatial = layer["spatial_attn"](layer["norm1"](x_flat), layer["norm1"](x_flat)) + x_flat x = x_spatial.reshape(T, B, Z * H * W, C) x = x.permute(2, 1, 0, 3) x_t = x.reshape(Z * H * W * B, T, C) x_t_norm = layer["norm2"](x_t) x_temp = layer["temporal_attn"](x_t_norm, x_t_norm) + x_t x = x_temp.reshape(Z * H * W, B, T, C).permute(2, 1, 0, 3) x = x + layer["ffn"](layer["norm3"](x)) refined_bev = x.mean(dim=0).reshape(B, Z, H, W, C).permute(0, 4, 1, 2, 3) flow = self.flow_head(refined_bev) return refined_bev, flow

#!/usr/bin/env python3 """pure_e2e_transformer.py — Large ViT for PureE2EFSD with hierarchical token merging. v3 changes: - HierarchicalTokenMerger: reduces token count 4× after first 2 layers via Conv2d strided patch merging — drops ~60% VRAM with minimal accuracy loss - Gradient checkpointing on all ViT encoder layers via torch.utils.checkpoint.checkpoint_sequential - FlashAttention-2 backend via SDPA in nn.TransformerEncoderLayer (PyTorch ≥ 2.0 uses SDPA internally when norm_first=True and batch_first=True) - `hierarchical: bool` flag in __init__ to enable/disable merging """ from __future__ import annotations import torch import torch.nn as nn import torch.nn.functional as F from torch.utils.checkpoint import checkpoint_sequential from typing import Tuple from config import ModelConfig, CONFIG class HierarchicalTokenMerger(nn.Module): """ Reduces token sequence length by 4× via a strided Conv1d patch-merging step. Global mean pooling → BEV decoder global_feat = tokens.mean(dim=1) # (B, C) refined_bev = self.bev_decoder(global_feat) # (B, 512, Z, H_bev, W_bev) return refined_bev

#!/usr/bin/env python3 """main.py — Inference entry point for FSD v3. Usage: python main.py # original FSD python main.py --model hybrid # HybridFSD python main.py --model pure # PureE2EFSD python main.py --calibration cal.json # with calibration file python main.py --model hybrid --safe # enable safety speed reduction """ from __future__ import annotations import argparse import logging import time import torch from config import CONFIG from model import FSD, HybridFSD, PureE2EFSD from lss import CalibrationManager try: import torch_tensorrt TENSORRT_AVAILABLE = True except ImportError: TENSORRT_AVAILABLE = False logging.basicConfig(level=logging.INFO, format="%(levelname)s — %(message)s") logger = logging.getLogger(__name__) def build_dummy_inputs(config=CONFIG): d, dt = config.device, config.dtype rgb = torch.randn(1, config.num_cameras, 3, *config.img_size_rgb, device=d, dtype=dt) photon = torch.randn(1, config.num_cameras, 1, *config.img_size_photon, device=d, dtype=dt).abs() * 0.1 motion = torch.zeros(1, 6, device=d, dtype=dt) motion[:, :3] = torch.randn(1, 3) * 0.05 motion[:, 3:] = torch.randn(1, 3) * 0.01 return rgb, photon, motion def compile_tensorrt(model, rgb, photon, motion): try: logger.info("Compiling with Torch-TensorRT (INT8 PTQ)…") trt_model = torch_tensorrt.compile( model, inputs=[rgb, photon, motion], enabled_precisions={torch.int8}, truncate_long_and_double=True, use_fast_partitioner=True, ) logger.info("TensorRT INT8 compiled.") return trt_model except Exception as e: logger.warning(f"TensorRT compilation failed ({e}). Falling back to eager.") return model def main(): parser = argparse.ArgumentParser(description="FSD v3 inference") parser.add_argument("--calibration", type=str, default=None) parser.add_argument("--model", type=str, default="original", choices=["original", "hybrid", "pure"]) parser.add_argument("--safe", action="store_true", help="Enable safety speed reduction on high uncertainty") args = parser.parse_args() cfg = CONFIG cal_mgr = CalibrationManager.from_json(args.calibration, cfg) if args.calibration else None if args.model == "hybrid": model = HybridFSD(config=cfg, calibration_manager=cal_mgr).to(cfg.device) elif args.model == "pure": model = PureE2EFSD(config=cfg, calibration_manager=cal_mgr).to(cfg.device) else: model = FSD(config=cfg, calibration_manager=cal_mgr).to(cfg.device) logger.info(f"Model: {args.model}") model.eval() torch.backends.cudnn.benchmark = True rgb, photon, motion = build_dummy_inputs(cfg) inference_model = compile_tensorrt(model, rgb, photon, motion) if TENSORRT_AVAILABLE else model # Warm-up logger.info("Warming up (50 iterations)…") history = None for _ in range(50): with torch.no_grad(): out = inference_model(rgb, photon, motion, history=history, language_prompt="Drive cautiously") history = out.get("new_history") logger.info(f"FSD v3 ({args.model}) — inference loop started") t0 = time.perf_counter() count = 0 try: while True: with torch.no_grad(): out = inference_model(rgb, photon, motion, history=history, language_prompt="Drive cautiously") history = out.get("new_history") if torch.cuda.is_available(): torch.cuda.synchronize() # Safety margin: reduce speed if uncertainty is too high if args.safe and out.get("should_reduce_speed", False): logger.warning("High uncertainty detected — reducing commanded speed") # In real deployment: scale throttle down here count += 1 if count % 30 == 0: fps = 30.0 / (time.perf_counter() - t0) steer = out["control"][0, 0].item() throttle = out["control"][0, 1].item() ctrl_var = out["control_var"][0, 0].item() traj_info = "" if "trajectories" in out: best_score = out["trajectory_scores"][0].min().item() traj_info = f" │ Best traj score: {best_score:.3f}" flow_info = "" if "occupancy_flow" in out: flow_info = f" │ Flow ‖v‖: {out['occupancy_flow'].norm(dim=1).mean().item():.3f}" logger.info( f"FPS: {fps:5.1f} │ Steer: {steer:+.3f} │ Throttle: {throttle:+.3f}" f" │ CtrlVar: {ctrl_var:.4f}{traj_info}{flow_info}" ) t0 = time.perf_counter() except KeyboardInterrupt: logger.info("Inference stopped.") if __name__ == "__main__": main()

Quick Start

# Inference — original FSD python main.py # Inference — HybridFSD with safety margin python main.py --model hybrid --safe # Inference — PureE2EFSD (≥24 GB VRAM recommended) python main.py --model pure # Training — HybridFSD, single GPU python train.py --model hybrid --output checkpoints/ # Training — PureE2EFSD, 4×GPU with FSDP torchrun --nproc_per_node=4 train.py --model pure --fsdp --output checkpoints/ # Export to ONNX + TensorRT python export.py --model hybrid --output exports/ # Run unit tests pytest tests/ -v 

Hardware note: PureE2EFSD with hierarchical token merging (v3) reduces peak VRAM from ~30 GB to ~12–14 GB at T=6 frames, making it runnable on a single 24 GB card. Use photon_type="event" in config for best photon stream fidelity with event-style cameras.
