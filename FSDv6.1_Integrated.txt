criticality.py

""" criticality.py -------------- Multi-factor criticality scoring for IRL sample weighting. Replaces the flawed v6.0 proxy (torch.norm on trajectory magnitude) with three physically meaningful components: 1. True obstacle proximity via occupancy grid convolution 2. Kinematic stress (acceleration + curvature) 3. Expert uniqueness (how much better the expert is vs. proposals) """ import torch import torch.nn.functional as F from torch import Tensor # --------------------------------------------------------------------------- # Helper: minimum distance to occupied cells # --------------------------------------------------------------------------- def compute_min_distance_to_occupancy( traj: Tensor, # (B, T, 2) — x,y positions in BEV grid coords occupancy: Tensor, # (B, H, W) — occupancy probability map [0,1] sigma: float = 1.0, ) -> Tensor: """ Approximate minimum distance from each trajectory to any occupied cell. Strategy: convolve occupancy with a distance-falloff kernel, then sample the convolved map at trajectory positions. Fast on GPU; no KD-tree needed. Returns: (B,) — one scalar per batch element. """ B, T, _ = traj.shape _, H, W = occupancy.shape # Build a soft distance kernel (Gaussian falloff centred at 0) k = 9 r = k // 2 ys, xs = torch.meshgrid( torch.arange(-r, r + 1, dtype=torch.float32, device=traj.device), torch.arange(-r, r + 1, dtype=torch.float32, device=traj.device), indexing="ij", ) kernel = torch.exp(-(xs ** 2 + ys ** 2) / (2 * sigma ** 2)) kernel = kernel / kernel.sum() kernel = kernel.view(1, 1, k, k) # Convolve occupancy → proximity map occ = occupancy.unsqueeze(1) # (B,1,H,W) prox = F.conv2d(occ, kernel, padding=r) # (B,1,H,W) prox = prox.squeeze(1) # (B,H,W) # Sample proximity map at trajectory waypoints via grid_sample # Normalise traj coords to [-1, 1] for grid_sample grid = traj.clone() grid[..., 0] = 2.0 * grid[..., 0] / (W - 1) - 1.0 # x → col grid[..., 1] = 2.0 * grid[..., 1] / (H - 1) - 1.0 # y → row grid = grid.unsqueeze(2) # (B,T,1,2) sampled = F.grid_sample( prox.unsqueeze(1), grid, mode="bilinear", padding_mode="border", align_corners=True, ) # (B,1,T,1) sampled = sampled.squeeze(1).squeeze(-1) # (B,T) # Convert proximity score to approximate distance (inverse of Gaussian) # Higher prox → closer to obstacle → lower distance dist_approx = -torch.log(sampled.clamp(min=1e-6)) return dist_approx.mean(dim=1) # (B,) # --------------------------------------------------------------------------- # Helper: approximate curvature from waypoints # --------------------------------------------------------------------------- def compute_curvature(traj: Tensor) -> Tensor: """ Approximate signed curvature at each interior waypoint. traj: (B, T, 2) — x,y positions. Returns: (B, T-2) curvature magnitude. """ p0 = traj[:, :-2] # (B, T-2, 2) p1 = traj[:, 1:-1] p2 = traj[:, 2:] d1 = p1 - p0 d2 = p2 - p1 # Cross product (z-component) and norms cross = d1[..., 0] * d2[..., 1] - d1[..., 1] * d2[..., 0] norm1 = d1.norm(dim=-1).clamp(min=1e-6) norm2 = d2.norm(dim=-1).clamp(min=1e-6) return (cross.abs() / (norm1 * norm2)).clamp(max=10.0) # (B, T-2) # --------------------------------------------------------------------------- # Main: multi-factor criticality # --------------------------------------------------------------------------- def compute_criticality( expert_traj: Tensor, # (B, T, 4) — [x, y, heading, speed] occupancy: Tensor, # (B, H, W) proposals: Tensor, # (B, K, T, 4) soft_cost_fn, # callable(traj, occupancy) → (B,) or (B,K) min_dist_threshold: float = 2.0, w_proximity: float = 0.5, w_kinematic: float = 0.3, w_uniqueness: float = 0.2, ) -> Tensor: """ Criticality = f(obstacle_proximity, kinematic_stress, expert_uniqueness) Returns: (B,) normalised weights ≥ 0, mean ≈ 1. """ B, T, _ = expert_traj.shape xy = expert_traj[..., :2] # (B, T, 2) # 1. Obstacle proximity ------------------------------------------------ dist = compute_min_distance_to_occupancy(xy, occupancy) # (B,) proximity_score = torch.exp(-dist / min_dist_threshold) # (B,) # 2. Kinematic stress -------------------------------------------------- vel = torch.diff(xy, dim=1) # (B, T-1, 2) acc = torch.diff(vel, dim=1) # (B, T-2, 2) jerk = torch.diff(acc, dim=1) # (B, T-3, 2) acc_mag = acc.norm(dim=-1).mean(dim=1) # (B,) jerk_mag = jerk.norm(dim=-1).mean(dim=1) # (B,) curvature = compute_curvature(xy).mean(dim=1) # (B,) kinematic_score = acc_mag + 0.5 * jerk_mag + curvature # (B,) # 3. Expert uniqueness ------------------------------------------------- # How much harder is the expert's situation vs. the best proposal? expert_cost = soft_cost_fn(expert_traj, occupancy) # (B,) best_proposal_cost = soft_cost_fn( proposals.view(B * proposals.shape[1], T, -1), occupancy.repeat_interleave(proposals.shape[1], dim=0), ).view(B, -1).min(dim=1).values # (B,) uniqueness = torch.relu(expert_cost - best_proposal_cost) # (B,) # Normalise each component to unit mean before combining def norm_component(x: Tensor) -> Tensor: return x / (x.mean() + 1e-6) criticality = ( w_proximity * norm_component(proximity_score) + w_kinematic * norm_component(kinematic_score) + w_uniqueness * norm_component(uniqueness) ) return criticality / (criticality.mean() + 1e-6) # (B,), mean ≈ 1 

train_irl.py

""" train_irl.py ------------ Criticality-Weighted MaxEnt IRL training loop. v6.1 changes: - Uses multi-factor compute_criticality (see criticality.py). - Replaces plain cross-entropy with FocalCriticalityLoss (see losses.py). - Data engine hook: logs high-criticality samples for replay buffer. """ import torch import torch.nn as nn import torch.nn.functional as F from torch import Tensor from typing import Optional from criticality import compute_criticality from losses import FocalCriticalityLoss # --------------------------------------------------------------------------- # Differentiable Cost Planner (v6.1) # --------------------------------------------------------------------------- class DifferentiableCostPlanner(nn.Module): """ Learns cost weights via MaxEnt IRL. Supports criticality-weighted and focal loss variants. """ def __init__(self, config): super().__init__() self.config = config self.cost_weights = nn.Parameter(torch.ones(config.num_cost_features)) self.temperature = nn.Parameter(torch.tensor(1.0)) self.focal_loss = FocalCriticalityLoss(gamma=2.0, reduction="none") def compute_costs( self, proposals: Tensor, # (B, K, T, 4) occupancy: Tensor, # (B, H, W) ) -> Tensor: """Returns (B, K) cost for each proposal.""" raise NotImplementedError("Implement domain-specific cost features.") def forward( self, proposals: Tensor, # (B, K, T, 4) expert_traj: Tensor, # (B, T, 4) occupancy: Tensor, # (B, H, W) expert_traj_idx: int = 0, weights: Optional[Tensor] = None, use_focal: bool = True, ) -> Tensor: B, K, T, _ = proposals.shape # Stack expert with proposals → (B, K+1, T, 4) all_trajs = torch.cat([expert_traj.unsqueeze(1), proposals], dim=1) costs = self.compute_costs(all_trajs, occupancy) # (B, K+1) logits = -costs / self.temperature.clamp(min=0.1) target = torch.full((B,), expert_traj_idx, dtype=torch.long, device=logits.device) if use_focal: # Focal loss upweights hard (misclassified) examples loss = self.focal_loss(logits, target) # (B,) else: loss = F.cross_entropy(logits, target, reduction="none") # (B,) if weights is not None: # Apply criticality weights — rare critical maneuvers dominate return (loss * weights).mean() return loss.mean() # --------------------------------------------------------------------------- # Training step # --------------------------------------------------------------------------- def irl_train_step( planner: DifferentiableCostPlanner, optimizer: torch.optim.Optimizer, batch: dict, soft_cost_fn, data_engine_buffer: Optional[list] = None, criticality_log_threshold: float = 2.0, ) -> dict: """ Single IRL training step with criticality weighting. batch keys: proposals (B,K,T,4), expert_traj (B,T,4), occupancy (B,H,W) """ proposals = batch["proposals"] expert_traj = batch["expert_traj"] occupancy = batch["occupancy"] # Compute multi-factor criticality weights with torch.no_grad(): weights = compute_criticality( expert_traj=expert_traj, occupancy=occupancy, proposals=proposals, soft_cost_fn=soft_cost_fn, ) # Data engine: log high-criticality samples for replay if data_engine_buffer is not None: high_crit_mask = weights > criticality_log_threshold if high_crit_mask.any(): for idx in high_crit_mask.nonzero(as_tuple=True)[0]: data_engine_buffer.append({ "expert_traj": expert_traj[idx].detach().cpu(), "occupancy": occupancy[idx].detach().cpu(), "weight": weights[idx].item(), }) optimizer.zero_grad() loss = planner( proposals=proposals, expert_traj=expert_traj, occupancy=occupancy, weights=weights, use_focal=True, ) loss.backward() torch.nn.utils.clip_grad_norm_(planner.parameters(), max_norm=1.0) optimizer.step() return { "loss": loss.item(), "mean_criticality": weights.mean().item(), "max_criticality": weights.max().item(), } 

refinement_net.py

""" refinement_net.py ----------------- Transformer-based amortized trajectory refiner. v6.1 replaces the 5-step Adam loop AND the shallow MLP refiner from the initial v6.1 draft with a proper Transformer encoder that respects spatio-temporal structure. Latency target: ~0.2 ms vs ~5 ms for the iterative optimizer. Training protocol: - Teacher: full iterative optimizer (LBFGS) or human corrections. - Losses: L1 on delta + consistency loss across small perturbations. - Optional: DDPM-style diffusion for multimodal corrections. """ import torch import torch.nn as nn from torch import Tensor from typing import Optional class TrajectoryRefiner(nn.Module): """ Input: best trajectory (B, T, 4) + cost gradient (B, T, 4) + optional BEV context (B, T, C) Output: refined trajectory (B, T, 4) Architecture: learned time embeddings → Transformer encoder → delta head. The delta is bounded with tanh for training stability. """ def __init__( self, T: int = 12, d_model: int = 256, n_layers: int = 4, n_heads: int = 8, dim_ff: int = 1024, context_dim: int = 0, # set > 0 to condition on BEV context delta_scale: float = 0.15, ): super().__init__() self.T = T self.delta_scale = delta_scale # Input projection: (traj 4-dim + grad 4-dim + optional context) in_dim = 8 + context_dim self.input_proj = nn.Linear(in_dim, d_model) # Learnable time (positional) embeddings self.time_embed = nn.Parameter(torch.randn(T, d_model) * 0.02) # Transformer encoder — norm_first (Pre-LN) for training stability encoder_layer = nn.TransformerEncoderLayer( d_model=d_model, nhead=n_heads, dim_feedforward=dim_ff, activation="gelu", norm_first=True, batch_first=True, dropout=0.0, # disabled at inference; use in training ) self.encoder = nn.TransformerEncoder(encoder_layer, num_layers=n_layers) # Output delta head self.output_head = nn.Sequential( nn.LayerNorm(d_model), nn.Linear(d_model, 4), # dx, dy, dheading, dspeed ) self._init_weights() def _init_weights(self): for m in self.modules(): if isinstance(m, nn.Linear): nn.init.xavier_uniform_(m.weight, gain=0.1) if m.bias is not None: nn.init.zeros_(m.bias) def forward( self, traj: Tensor, # (B, T, 4) cost_grad: Tensor, # (B, T, 4) context: Optional[Tensor] = None, # (B, T, context_dim) ) -> Tensor: B, T, _ = traj.shape # Concatenate inputs x = torch.cat([traj, cost_grad], dim=-1) # (B, T, 8) if context is not None: x = torch.cat([x, context], dim=-1) # (B, T, 8+C) x = self.input_proj(x) # (B, T, d_model) x = x + self.time_embed.unsqueeze(0) # broadcast time embed x = self.encoder(x) # (B, T, d_model) delta = self.output_head(x) # (B, T, 4) delta = torch.tanh(delta) # bounded ∈ (-1, 1) return traj + self.delta_scale * delta # refined trajectory # --------------------------------------------------------------------------- # Distillation loss for training the refiner # --------------------------------------------------------------------------- def refiner_distillation_loss( refiner: TrajectoryRefiner, traj_init: Tensor, # (B, T, 4) — initial coarse trajectory cost_grad: Tensor, # (B, T, 4) traj_teacher: Tensor, # (B, T, 4) — teacher (iterative optimizer) context: Optional[Tensor] = None, perturbation_std: float = 0.05, ) -> dict: """ Combined distillation + consistency loss. 1. L1 between refined output and teacher trajectory. 2. Consistency: perturb input slightly → refined outputs should be close. """ refined = refiner(traj_init, cost_grad, context) # Primary distillation loss l1_loss = torch.nn.functional.l1_loss(refined, traj_teacher) # Consistency regularisation noise = torch.randn_like(traj_init) * perturbation_std refined_perturbed = refiner(traj_init + noise, cost_grad, context) consistency_loss = torch.nn.functional.l1_loss(refined, refined_perturbed.detach()) total = l1_loss + 0.1 * consistency_loss return { "total": total, "l1": l1_loss.item(), "consistency": consistency_loss.item(), } 

heads.py

""" heads.py -------- MapFormer with multi-task semantic prediction and uncertainty heads. v6.1 additions: - Stop-line confidence head (per-instance). - Lane centreline prediction head. - Drivable area uncertainty head (aleatoric). - Temporal GRU for fusing semantic elements across frames. """ import torch import torch.nn as nn import torch.nn.functional as F from torch import Tensor from typing import Tuple, Optional from dataclasses import dataclass from config import ModelConfig @dataclass class MapFormerOutput: map_tokens: Tensor # (B, N, D) map_polylines: Tensor # (B, P, pts, 2) stop_lines: Tensor # (B, S, pts, 2) stop_conf: Tensor # (B, S) — per-instance confidence [0,1] lane_centrelines: Tensor # (B, L, pts, 2) drivable_area: Tensor # (B, H, W) — BEV drivable probability drivable_unc: Tensor # (B, H, W) — aleatoric uncertainty class MapFormer(nn.Module): """ BEV-based map element prediction with uncertainty outputs. """ def __init__(self, config: ModelConfig): super().__init__() H = config.hidden_dim self.S = config.num_stop_lines self.L = config.num_lane_centrelines self.pts = config.map_pts_per_element # Shared BEV feature backbone (assumed pre-built upstream) self.bev_encoder = nn.Sequential( nn.Conv2d(config.bev_channels, H, 3, padding=1), nn.GroupNorm(8, H), nn.GELU(), nn.Conv2d(H, H, 3, padding=1), nn.GroupNorm(8, H), nn.GELU(), ) self.feat_pool = nn.AdaptiveAvgPool2d(1) # global BEV feature # Map polyline head self.map_head = nn.Linear(H, config.num_map_polylines * self.pts * 2) self.map_token_proj = nn.Linear(H, config.map_token_dim) # Stop-line heads self.stop_line_head = nn.Linear(H, self.S * self.pts * 2) self.stop_line_conf = nn.Linear(H, self.S) # Lane centreline head self.lane_head = nn.Linear(H, self.L * self.pts * 2) # Drivable area head (spatial — keep spatial resolution) self.drivable_head = nn.Conv2d(H, 1, 1) self.drivable_unc_head = nn.Conv2d(H, 1, 1) # log-variance # Temporal GRU for stop-line / traffic-light stability across frames self.temporal_gru = nn.GRUCell( input_size=self.S * 2, # (conf, x_mean) per stop-line hidden_size=self.S * 2, ) self._temporal_hidden: Optional[Tensor] = None def reset_temporal_state(self): self._temporal_hidden = None def forward( self, bev_2d: Tensor, # (B, C, H_bev, W_bev) temporal_reset: bool = False, ) -> MapFormerOutput: if temporal_reset: self.reset_temporal_state() B = bev_2d.shape[0] feat_map = self.bev_encoder(bev_2d) # (B, H, H_bev, W_bev) feat = self.feat_pool(feat_map).view(B, -1) # (B, H) # Polylines map_polylines = self.map_head(feat).view(B, -1, self.pts, 2) map_tokens = self.map_token_proj(feat).unsqueeze(1).expand( B, map_polylines.shape[1], -1) # Stop lines + confidence stop_lines = self.stop_line_head(feat).view(B, self.S, self.pts, 2) stop_conf_raw = torch.sigmoid(self.stop_line_conf(feat)) # (B, S) # Temporal smoothing on stop-line confidence conf_flat = stop_conf_raw.view(B, -1) if self._temporal_hidden is None: self._temporal_hidden = torch.zeros_like(conf_flat) self._temporal_hidden = self.temporal_gru(conf_flat, self._temporal_hidden) stop_conf = torch.sigmoid(self._temporal_hidden).view(B, self.S) # Lane centrelines lane_centrelines = self.lane_head(feat).view(B, self.L, self.pts, 2) # Drivable area + uncertainty drivable_logit = self.drivable_head(feat_map).squeeze(1) # (B, H_bev, W_bev) drivable_area = torch.sigmoid(drivable_logit) drivable_log_var = self.drivable_unc_head(feat_map).squeeze(1) drivable_unc = torch.exp(0.5 * drivable_log_var).clamp(max=5.0) return MapFormerOutput( map_tokens=map_tokens, map_polylines=map_polylines, stop_lines=stop_lines, stop_conf=stop_conf, lane_centrelines=lane_centrelines, drivable_area=drivable_area, drivable_unc=drivable_unc, ) 

rules.py

""" rules.py -------- Traffic rule injector with soft, uncertainty-aware HD Map + online fusion. v6.1 changes: - Replaces hard confidence threshold (> 0.8) with a soft cost multiplier. - Adds map confidence scalar to increase vision weight when HD map is stale. - Temporal fusion handled upstream in MapFormer (GRU); rules layer is stateless. """ import torch import torch.nn.functional as F from torch import Tensor from dataclasses import dataclass from enum import Enum from typing import List, Optional class LightState(Enum): RED = "red" YELLOW = "yellow" GREEN = "green" UNKNOWN = "unknown" @dataclass class TrafficElement: element_type: str # "light", "stop_sign", "crosswalk", ... light_state: LightState geometry_m: Tensor # (pts, 2) in metric BEV coords class TrafficRuleInjector: """ Merges HD Map traffic elements with online visual predictions into the BEV occupancy/cost grid. Soft fusion formula: weight_hd_map = 1.0 (always trusted if present) weight_online = alpha * online_conf where alpha ∈ [0, 1] map_conf_scalar = "how much to trust the HD map overall" derived from map staleness / localisation quality When map_conf is low, alpha → 1 (rely more on vision). When map_conf is high, alpha → 0.4 (moderate trust in vision as supplement). """ def __init__( self, grid_h: int, grid_w: int, resolution_m: float = 0.2, online_alpha_min: float = 0.4, online_alpha_max: float = 1.0, ): self.grid_h = grid_h self.grid_w = grid_w self.res = resolution_m self.alpha_min = online_alpha_min self.alpha_max = online_alpha_max # ------------------------------------------------------------------ # Internal helpers # ------------------------------------------------------------------ def _geometry_to_grid(self, geom_m: Tensor) -> Tensor: """Convert metric coordinates to grid indices.""" return (geom_m / self.res).long().clamp( min=torch.tensor([0, 0]), max=torch.tensor([self.grid_w - 1, self.grid_h - 1]), ) def _inject_element( self, occ: Tensor, batch_idx: int, elem: TrafficElement, weight: float = 1.0 ) -> Tensor: if elem.light_state not in (LightState.RED, LightState.YELLOW): return occ pts = self._geometry_to_grid(elem.geometry_m) # (pts, 2) → (col, row) for pt in pts: col, row = pt[0].item(), pt[1].item() if 0 <= row < self.grid_h and 0 <= col < self.grid_w: occ[batch_idx, int(row), int(col)] = min( 1.0, occ[batch_idx, int(row), int(col)] + weight ) return occ # ------------------------------------------------------------------ # Public API # ------------------------------------------------------------------ def __call__( self, occupancy: Tensor, # (B, H, W) traffic_elements: List[List[TrafficElement]], # [B][n_elems] online_stop_lines: Optional[Tensor] = None, # (B, S, pts, 2) online_conf: Optional[Tensor] = None, # (B, S) map_conf_scalar: Optional[Tensor] = None, # (B,) ∈ [0,1], HD map quality ) -> Tensor: occ = occupancy.clone() B = occ.shape[0] # 1. Inject HD Map elements (full weight) for b in range(B): for elem in traffic_elements[b]: occ = self._inject_element(occ, b, elem, weight=1.0) # 2. Soft-inject online visual predictions if online_stop_lines is not None and online_conf is not None: # Determine per-batch alpha based on map quality if map_conf_scalar is not None: # High map_conf → lower alpha (vision supplements) # Low map_conf → higher alpha (vision leads) alpha = self.alpha_max - ( (self.alpha_max - self.alpha_min) * map_conf_scalar ) # (B,) else: alpha = torch.full((B,), self.alpha_max, device=online_conf.device) for b in range(B): for s in range(online_stop_lines.shape[1]): conf = online_conf[b, s].item() weight = float(alpha[b].item()) * conf if weight < 0.05: continue # negligible — skip elem = TrafficElement( element_type="light", light_state=LightState.RED, # conservative default geometry_m=online_stop_lines[b, s], ) occ = self._inject_element(occ, b, elem, weight=weight) return occ

""" model.py -------- Unified FSD v6.1 forward pass. Pipeline: 1. Perception : MapFormer → road geometry + visual stop-lines + uncertainty 2. Rule Inject : TrafficRuleInjector → merged BEV occupancy 3. Planning : CostPlannerV2 → best proposal (criticality-weighted IRL) 4. Refinement : TrajectoryRefiner → smooth, optimal trajectory (1 fwd pass) 5. Safety : DifferentiableSafetyFilter → hard-stop guarantee (grad-through) """ import torch import torch.nn as nn from torch import Tensor from typing import Optional, Dict, Any from config import ModelConfig from heads import MapFormer, MapFormerOutput from rules import TrafficRuleInjector, TrafficElement from train_irl import DifferentiableCostPlanner from refinement_net import TrajectoryRefiner from safety import DifferentiableSafetyFilter class FSDV61(nn.Module): def __init__(self, config: ModelConfig): super().__init__() self.config = config # Perception self.map_former = MapFormer(config) # Rule injection self.rule_injector = TrafficRuleInjector( grid_h=config.bev_h, grid_w=config.bev_w, resolution_m=config.bev_resolution_m, ) # Planning backbone (IRL cost weights learned separately via train_irl.py) self.planner = DifferentiableCostPlanner(config) # Amortized refiner (single forward pass) self.refiner = TrajectoryRefiner( T=config.trajectory_horizon, d_model=256, n_layers=4, ) # Safety filter self.safety_filter = DifferentiableSafetyFilter(config) def forward( self, bev_2d: Tensor, # (B, C, H, W) proposals: Tensor, # (B, K, T, 4) traffic_elements, # List[List[TrafficElement]] map_conf_scalar: Optional[Tensor] = None, # (B,) HD map quality temporal_reset: bool = False, refine_trajectory: bool = True, ) -> Dict[str, Any]: B = bev_2d.shape[0] # ── 1. Perception ────────────────────────────────────────────── map_out: MapFormerOutput = self.map_former(bev_2d, temporal_reset) # ── 2. Rule Injection ───────────────────────────────────────── occupancy_base = self._build_base_occupancy(bev_2d) # (B, H, W) occupancy = self.rule_injector( occupancy=occupancy_base, traffic_elements=traffic_elements, online_stop_lines=map_out.stop_lines, online_conf=map_out.stop_conf, map_conf_scalar=map_conf_scalar, ) # ── 3. Planning ─────────────────────────────────────────────── costs = self.planner.compute_costs(proposals, occupancy) # (B, K) best_idx = costs.argmin(dim=-1) # (B,) best_traj = proposals[torch.arange(B), best_idx] # (B, T, 4) # ── 4. Amortized Refinement ─────────────────────────────────── if refine_trajectory: best_traj.requires_grad_(True) cost_scalar = self.planner.compute_costs( best_traj.unsqueeze(1), occupancy ).squeeze(1).sum() grad = torch.autograd.grad(cost_scalar, best_traj, create_graph=self.training)[0] best_traj = self.refiner(best_traj, grad) # ── 5. Safety Filter ────────────────────────────────────────── safe_traj = self.safety_filter(best_traj, occupancy) return { "trajectory": safe_traj, # (B, T, 4) "map_out": map_out, "occupancy": occupancy, "proposal_costs": costs, } def _build_base_occupancy(self, bev_2d: Tensor) -> Tensor: """Stub: derive initial occupancy from BEV features or sensor fusion.""" B = bev_2d.shape[0] return torch.zeros(B, self.config.bev_h, self.config.bev_w, device=bev_2d.device) 

losses.py

""" losses.py --------- Focal loss and cost-sensitive ranking loss for IRL training. FocalCriticalityLoss: Combines focal down-weighting of easy examples with per-sample criticality weights. Rare, hard maneuvers dominate the gradient. """ import torch import torch.nn as nn import torch.nn.functional as F from torch import Tensor from typing import Optional class FocalCriticalityLoss(nn.Module): """ Focal cross-entropy with optional per-sample criticality weighting. Loss_i = criticality_i * (1 - p_t)^gamma * CE_i where p_t is the model's probability on the correct class (expert trajectory). gamma=0 → standard cross-entropy; gamma=2 is a good default. """ def __init__(self, gamma: float = 2.0, reduction: str = "mean"): super().__init__() self.gamma = gamma self.reduction = reduction def forward( self, logits: Tensor, # (B, K+1) targets: Tensor, # (B,) long weights: Optional[Tensor] = None, # (B,) criticality ) -> Tensor: log_probs = F.log_softmax(logits, dim=-1) # (B, K+1) probs = log_probs.exp() # Gather probability on the correct class target_log_p = log_probs.gather(1, targets.unsqueeze(1)).squeeze(1) # (B,) target_p = probs.gather(1, targets.unsqueeze(1)).squeeze(1) # (B,) # Focal weight: (1 - p_t)^gamma focal_weight = (1.0 - target_p) ** self.gamma # (B,) loss = -focal_weight * target_log_p # (B,) if weights is not None: loss = loss * weights if self.reduction == "mean": return loss.mean() elif self.reduction == "sum": return loss.sum() return loss # "none" class CostSensitiveRankingLoss(nn.Module): """ Hinge-based ranking loss: enforces that the expert cost is lower than each proposal cost by a margin proportional to criticality. """ safety.py --------- Differentiable safety filter using barrier function projection. v6.1 upgrade: - v6.0 used a hard post-hoc clamp (non-differentiable, blocked gradients). - v6.1 uses a smooth barrier penalty that can be backpropagated through during end-to-end training. Two modes: 1. Training: add barrier loss term to the training objective. 2. Inference: projected gradient descent (PGD) for hard guarantee. """ import torch import torch.nn as nn import torch.nn.functional as F from torch import Tensor from typing import Optional from config import ModelConfig class DifferentiableSafetyFilter(nn.Module): """ Applies a log-barrier penalty to keep the trajectory away from occupied cells, while remaining differentiable. The barrier function is: B(d) = -log(d / d_min) for d > d_min B(d) = +∞ for d ≤ d_min Approximated as a smooth hinge for numerical stability. """ def __init__( self, config: ModelConfig, d_min: float = 0.5, # minimum clearance in metres barrier_weight: float = 10.0, pgd_steps: int = 5, # PGD steps for hard projection at inference pgd_lr: float = 0.01, ): super().__init__() self.d_min = d_min self.barrier_weight = barrier_weight self.pgd_steps = pgd_steps self.pgd_lr = pgd_lr self.res = config.bev_resolution_m def barrier_loss( self, traj: Tensor, # (B, T, 4) occupancy: Tensor, # (B, H, W) — values in [0, 1] ) -> Tensor: """ Compute differentiable barrier loss. Higher occupancy values closer to the trajectory → higher loss. Gradients flow back to the trajectory. """ B, T, _ = traj.shape _, H, W = occupancy.shape xy = traj[..., :2] grid = xy.clone() grid[..., 0] = 2.0 * grid[..., 0] / (W - 1) - 1.0 grid[..., 1] = 2.0 * grid[..., 1] / (H - 1) - 1.0 grid = grid.unsqueeze(2) # (B, T, 1, 2) occ_at_traj = F.grid_sample( occupancy.unsqueeze(1), grid, mode="bilinear", padding_mode="zeros", align_corners=True, ).squeeze(1).squeeze(-1) # (B, T) # Smooth barrier: penalise being in or near occupied cells # log-barrier approximation: loss = log(1 + exp(occ / d_min)) barrier = F.softplus(occ_at_traj / self.d_min) return self.barrier_weight * barrier.mean() @torch.no_grad() def _pgd_project( self, traj: Tensor, occupancy: Tensor, ) -> Tensor: """ Hard projection via PGD at inference time. Iteratively push the trajectory away from occupied regions. """ t = traj.clone().requires_grad_(True) for _ in range(self.pgd_steps): loss = self.barrier_loss(t, occupancy) grad = torch.autograd.grad(loss, t)[0] t = (t - self.pgd_lr * grad).detach().requires_grad_(True) return t.detach() def forward( self, traj: Tensor, occupancy: Tensor, hard_project: bool = False, ) -> Tensor: """ Training mode : return traj unchanged; call barrier_loss() separately. Inference mode : optionally apply PGD hard projection. """ if hard_project or not self.training: return self._pgd_project(traj, occupancy) return traj
